\chapter{Prerequisite Material}

\begin{remark}
    There is a huge amount of prerequisite material that is needed,
    and it is never possible to study algebra independently of other subjects.
    We need skills like doing proofs
    by induction, contradiction, and contraposition,
    and we also need a basic understanding of numbers in general,
    and perhaps a good deal of geometric intuition would be nice;
    occasionally some proofs might also involve a non-trivial amount of analysis.
    But we need a starting point,
    and for me that starting point
    is the material that perhaps was never formally covered in Science One math
    and second year linear algebra and calculus,
    or maybe was a topic that was last taught in high school.
\end{remark}


\section{Relations}

\begin{definition}
    A relation is a denotion on two elements of a set \(X\).
    More specifically,
    if a relation \(R \subset X \cross X\),
    then we write \(aRb\) if \((a,b) \in R\).
\end{definition}

\begin{definition}
    An equivalence relation \(\sim\)
    is a relation that has three properties:
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item reflexive \(\forall x \in X, x \sim x\);
        \item symmetric \(\forall x,y \in X, x \sim y \implies y \sim x\); and
        \item transitive \(\forall x,y,z \in X,
            x \sim y \land y \sim z \implies x \sim z\).
    \end{enumerate}
    We say \(x\) is related to \(y\) if \(x \sim y\).
\end{definition}

\begin{definition}
    Suppose we have an equivalence relation \(\sim\).
    The equivalence class of \(x \in X\) is
    \([x] = \{y \in X : y \sim x\}\).
\end{definition}
\begin{lemma}
    An element is in its own equivalence class; \(x \in [x]\).
\end{lemma}
\begin{proof}
    By reflexivity.
\end{proof}
\begin{theorem}
    \(x \sim y \iff [x] = [y]\);
    two related elements must have the same equivalence class.
\end{theorem}
\begin{proof}
    If \(x \sim y\),
    then \([x]\) consists of all elements \(z\) that are related to \(x\),
    so \(z \sim x\), and transitivity gives us \(z \sim y\),
    so \(z \in [y]\), and hence \([x] \subset [y]\).
    Reversing the argument gives us \([y] \subset [x]\),
    so \([x] = [y]\).

    If \([x] = [y]\), then \(y \in [x]\),
    so by definition \(y \sim x\).
\end{proof}
\begin{corollary}
    For all elements \(\{x,y\} \in X\),
    either \([x] = [y]\) or \([x] \cap [y] = \emptyset\).
\end{corollary}
\begin{proof}
    Suppose \([x] \cap [y] \neq \emptyset\).
    Then there exists some element \(z \in [x]\) and \(z \in [y]\).
    But then \(z \sim x\) and \(z \sim y\),
    so \(x \sim y\) by transitivity,
    which by our theorem gives \([x] = [y]\).
\end{proof}

\begin{theorem}\label{thm:equiv-class-partition}
    The set of equivalence classes \(X/{\sim}\) partition \(X\).
\end{theorem}
\begin{proof}
    From the lemma,
    each element must belong in an equivalence class.
    The above corollary tells us equivalence classes do not overlap.
\end{proof}


\section{Sets}

\begin{axiom}[Axiom of Choice]\label{ax:choice}
    Suppose we have a collection of nonempty sets \({\{S_i\}}_{i \in I}\)
    and \(I\) an index set;
    it is possible to form a set \({\{x_i\}}_{i \in I}\)
    such that \(x_i \in S_i\).
    In other words,
    we can choose an element from every set
    no matter how many sets we have.
\end{axiom}

\begin{definition}
    A partial order relation \(\leq\) is a relation that has three properties:
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item reflexive \(x \leq x\);
        \item antisymmetric \(x \leq y \land y \leq x \implies x = y\); and
        \item transitive \(x \leq y \land y \leq z \implies x \leq z\).
    \end{enumerate}
    Note that there possibly exists elements
    that are neither \(x \leq y\) or \(x \geq y\).
\end{definition}
\begin{definition}
    A partially ordered set, or a poset,
    is a tuple \((S,\leq)\),
    such that \(S\) is a set,
    and \(\leq\) is a partial order relation.
    A totally ordered set is a poset \((S,\leq)\)
    with the additional condition that every pair of element \(x,y\) is comparable,
    so it can be \(x \leq y\) or \(x \geq y\) (or both).
\end{definition}
\begin{definition}
    A chain is a subset of a poset \((S,\leq)\)
    such that under the same order, the subset is totally ordered.
\end{definition}
\begin{axiom}[Zorn's Lemma]\label{ax:zorn}
    A partially ordered set containing an upper bound for every chain
    must contain at least one maximal element.
\end{axiom}

\begin{axiom}[Well-Ordering Principle]\label{ax:well-order}
    Suppose \(S \neq \emptyset\) is a set.
    There exists a well ordering on \(S\),
    i.e.\ there exists a total order \(\leq\)
    such that every nonempty subset \(A \subseteq S\)
    has a smallest element \(a \in A\) where \(a \leq b\) for all \(b \in A\).
\end{axiom}

\begin{theorem}
    The above three axioms
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item the \hyperref[ax:choice]{axiom of choice};
        \item \hyperref[ax:zorn]{Zorn's lemma}; and
        \item the \hyperref[ax:well-order]{well-ordering principle}
    \end{enumerate}
    are equivalent in Zermelo-Fraenkel set theory,
    and are independent from any other ZF axiom.
\end{theorem}
\begin{proof}
    This is a result in set theory, and hence we shall omit the proof,
    and use the result as is.
    For more details see any reference in set theory,
    perhaps Chapter~5 in Jech's \textit{Set Theory}~\cite{jech}.
\end{proof}


\section{Functions}

\begin{definition}
    A function \(f\) that maps elements in \(A\) to \(B\)
    is denoted \(\func{f}{A}{B}\).
    \(A\) is the domain, and \(B\) is the codomain.
    The range \(\img f= f(A)\) is the set of all possible outputs of the function,
    namely \(\img f = \{b \in B : \exists a \in A, f(a) = b\}\).
\end{definition}

\begin{definition}
    Suppose \(A' \subset A\) and \(B' \subset B\).
    The image of \(A'\) is the set of all possible outputs
    given elements in \(A'\),
    denoted \(f(A') = \{f(a) \in B : a \in A'\}\).
    The preimage of \(B'\) is the set of all possible inputs
    that result in an element in \(B'\),
    denoted \(f^{-1}(B') = \{x \in A : f(x) \in B'\}\).
\end{definition}

\begin{definition}
    A function is 1-to-1 or injective
    when each point in the range (or codomain)
    only corresponds to (at most) one point in the domain.
    That is, \(\forall x,y \in A, f(x) = f(y) \implies x = y\).
\end{definition}
\begin{definition}
    A function is onto or surjective
    when every point in the codomain
    corresponds to at least one point in the domain,
    i.e.\ the codomain is the range.
    That is, \(\forall y \in B, \exists x \in A, f(x) = y\).
\end{definition}
\begin{definition}
    A function is bijective if it is both injective and surjective.
\end{definition}

\begin{theorem}[Pigeonhole Principle]\label{thm:pigeonhole}
    Suppose two finite sets \(A,B\) have the same cardinality
    \(\abs{A} = \abs{B}\).
    Then \(\func{f}{A}{B}\) is injective
    if and only if it is surjective.
\end{theorem}
\begin{proof}
    Suppose \(f\) is injective but not surjective.
    Then if \(\abs{A} = n\),
    then there are \(n\) elements in \(B\) that are in the range.
    Lack of surjectivity implies that
    there exists at least one element that is not in the range.
    Hence \(\abs{B} > n\), which is a contradiction.

    Suppose \(f\) is surjective but not injective.
    Then if \(\abs{B} = n\),
    then there are at least \(n\) elements in \(A\).
    But no injectivity implies there must be some two elements
    \(x,y \in A\) that map to the same point \(f(x) = f(y)\),
    which so forces us to conclude \(\abs{A} \geq n+1\),
    which is a contradiction.
\end{proof}

\begin{proposition}\label{prop:composite-injection}
    % Suppose \(f = f_n \circ f_{n-1} \circ \cdots \circ f_2 \circ f_1\).
    If \(f \circ g\) is injective, then \(g\) is injective.
\end{proposition}
\begin{proof}
    % Without loss of generality, let \(f = f_2 \circ f_1\).
    Suppose, by way of contradiction, that \(g\) is not injective.
    Then there exists \(x,y\) in the domain of \(g\)
    such that \(g(x) = g(y)\).
    But that implies \((f \circ g)(x) = (f \circ g)(y)\),
    which contradicts that \(f \circ g\) is injective.
\end{proof}
\begin{proposition}\label{prop:composite-surjective}
    % Suppose \(f = f_n \circ f_{n-1} \circ \cdots \circ f_2 \circ f_1\).
    If \(f \circ g\) is surjective, then \(f\) is surjective.
\end{proposition}
\begin{proof}
    % Without loss of generality, let \(f = f_2 \circ f_1\).
    For all \(z\) in the codomain of \(f \circ g\),
    there is some \(x\) in the domain of \(f \circ g\)
    such that \((f \circ g)(x) = z\).
    But \(y = f(x)\) is in the domain of \(f\)
    such that \(f(y) = z\) maps to our desired element.
\end{proof}

\begin{proposition}
    Suppose \(\func{f}{A}{B}\) is a function, and \({\{B_i\}}_{i \in I}\) is a collection of subsets of \(B\).
    Then:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(f^{-1}(\emptyset) = \emptyset\) and \(f^{-1}(B) = A\);
        \item \(f^{-1}(\bigcup B_i) = \bigcup f^{-1}(B_i)\); and
        \item \(f^{-1}(\bigcap B_i) = \bigcap f^{-1}(B_i)\).
    \end{enumerate} 
\end{proposition}
\begin{proof}
    For (a), because every element \(a \in A\) gets mapped to some \(f(a) \in B\),
    \(a \notin f^{-1}(\emptyset)\) and \(a \in f^{-1}(B)\).
    Hence \(f^{-1}(\emptyset)\) is empty, and \(A \subseteq f^{-1}(B)\) (the reverse direction is trivially true).

    For (b), we can see that \(x \in f^{-1}(\bigcup B_i)\)
    means \(f(x) \in \bigcup B_i\), and hence \(f(x) \in B_i\) for some \(i\).
    But that exactly implies that \(x \in f^{-1}(B_i)\) for some \(i\);
    i.e.\ \(x \in \bigcup f^{-1}(B_i)\).
    All the implications here hold bidirectionally.

    For (c), replace the words `for some' with `for all' and we are done.
\end{proof}
\begin{proposition}
    Suppose \(\func{f}{A}{B}\) is a function, and \({\{A_i\}}_{i \in I}\) is a collection of subsets of \(A\).
    Then:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(f(\emptyset) = \emptyset\);
        \item \(f(\bigcup A_i) = \bigcup f(A_i)\); and
        \item \(f(\bigcap A_i) \subseteq \bigcap f(A_i)\).
    \end{enumerate} 
\end{proposition}
\begin{proof}
    Statement (a) is trivially true.

    For (b), we can see that \(y \in f(\bigcup A_i)\)
    means there exists some \(x \in \bigcup A_i\) such that \(f(x) = y\);
    i.e.\ for some \(x\) in some \(A_i\), \(f(x) = y\).
    But that implies \(y \in f(A_i)\) for some \(i\), and \(f(A_i) \subseteq \bigcup f(A_i)\).
    All the implications here hold bidirectionally.

    For (c), replace the words `for some' with `for all' and we are done.
    Notice that we do not have the reverse inclusion
    because the step \(y \in f(A_i)\) for all \(i\)
    does not imply there exists a single \(x \in A_i\) for all \(i\) such that \(f(x) = y\).
    As a result, the reverse inclusion only holds when \(f\) is injective.
\end{proof}


\section{Integers}

\begin{definition}
    Suppose we have integers \(a,b \in \bZ\).
    If there exists \(c \in \bZ\) such that \(b = ac\),
    then we say \(a\) divides \(b\) or \(b\) is a multiple of \(a\),
    often denoted \(a \mid b\).
\end{definition}
\begin{proposition}\label{prop:divisibility-properties}
    We have the following general properties for divisibility:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item if \(a \mid b\), then \((-a) \mid b\) and \(a \mid (-b)\);
        \item \(1 \mid b\), \(a \mid a\), and \(a \mid 0\) for all \(a,b \in \bZ\);
        \item if \(c \neq 0\), then \(a \mid b\) if and only if \(ca \mid cb\); and
        \item \(a \mid b\) and \(a \mid c\) implies \(a \mid (bx+cy)\) for all \(x,y \in \bZ\).
    \end{enumerate}
\end{proposition}
\begin{proof}
    Suppose \(a \mid b\). Then there exists \(c\) such that \(b = ac\).
    Then \(b = (-a)(-c)\) and \(-b = a(-c)\) as desired.
    This proves statement (a).

    \(b = 1 \cdot b\), \(a = a \cdot 1\), and \(0 = a \cdot 0\).
    This proves statement (b).

    If \(a \mid b\), then there exists \(d\) such that \(b = ad\),
    so \(cb = cad\) and \(ca \mid cb\) as desired.
    If \(ca \mid cb\), then there exists \(d\) such that \(cb = cad\),
    and when we cancel out \(c\) on both sides,
    we get \(b = ad\) and \(a \mid b\) as desired.
    This proves statement (c).

    Suppose \(a \mid b\) and \(a \mid c\).
    Then there exists \(d,e\) such that \(b = ad\) and \(c = ae\).
    We have \(bx+cy = axd + aye = a(xd + ye)\).
    This proves statment (d).
\end{proof}

\begin{theorem}[Division Algorithm]\label{thm:division}
    Suppose \(a \in \bN\) and \(b \in \bZ\).
    Then there exists unique \(q,r \in \bZ\)
    such that \(b = aq + r\) and \(0 \leq r < a\).
    \(q\) is called the quotient, and \(r\) is called the remainder.
\end{theorem}
\begin{proof}
    We first prove the existence of such \(q,r\).
    Consider the set \(R = \{b-an: n \in \bZ\} \cap \bZ_{\geq 0}\).
    By the \hyperref[ax:well-order]{well-ordering principle},
    suppose \(r \in R\) is minimal.
    Then \(r = b-aq\) for some \(q\).
    We note that \(r \geq 0\) already.
    Now, if by way of contradiction that \(r \geq a\),
    then \(0 \leq r-a < r\),
    and we see that \(r-a = (b-aq) - a = b - a(1+q) \in R\).
    This contradicts minimality of \(r \in R\),
    so we conclude that \(0 \leq r < a\).

    We now prove uniqueness.
    Suppose, that we have \(b = aq + r = aq' + r'\) for some other \(q',r'\).
    Without loss of generality, let \(0 \leq r' \leq r < a\).
    Then we have \(r-r' = (b-aq) - (b-aq') = a(q'-q)\),
    so this implies \(a \mid (r-r')\).
    But then \(0 \leq r-r' \leq r < a\),
    so if \(r-r' > 0\), notice that \(a \leq r-r'\) because the smaller number divides a bigger one,
    which gives us a contradiction,
    and we must conclude that \(r-r' = 0\).
    Then by simple computation, \(q = q'\).
\end{proof}
\begin{corollary}
    \(a \mid b\) if and only if the remainder is 0.
\end{corollary}
\begin{proof}
    If \(r = 0\), then \(b = aq\) as desired.
    If \(a \mid b\), then by uniqueness, \(b = aq + 0\).
\end{proof}

\begin{definition}
    Suppose we have integers \(a,b \in \bZ\), not both zero.
    The greatest common denominator, or gcd
    is the largest natural number \(n \in \bN\) such that \(n \mid a\) and \(n \mid b\).
    This is often denoted \(\gcd(a,b)\) or simply \((a,b)\).
    The least common multiple, or lcm
    is the smallest natural number \(m \in \bN\) such that \(a \mid m\) and \(b \mid m\).
    This is often denoted \(\lcm(a,b)\) or simply \([a,b]\).
\end{definition}
\begin{theorem}[B\'{e}zout's Identity]\label{thm:bezout}
    Suppose \(a,b \in \bZ\) are not both zero.
    Let \(S = \{ax+by: x,y \in \bZ\}\).
    Then:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(n \in S\) if and only if \((a,b) \mid n\);
        \item \((a,b)\) is the smallest positive element of \(S\); and
        \item if \(c \mid a\) and \(c \mid b\), then \(c \mid (a,b)\).
    \end{enumerate}
\end{theorem}
\begin{proof}
    Since \((a,b) \mid a\) and \((a,b) \mid b\),
    by Proposition~\ref{prop:divisibility-properties},
    \((a,b) \mid (ax+by)\) for any linear combination.
    This proves the forward direction of (a).
    
    Suppose \(m = \min(S \cap \bN)\) is the smallest positive element,
    which can be written as some \(m = au+bv\).
    By the previous claim, \((a,b) \mid m\), so \((a,b) \leq m\).
    Now, by the \hyperref[thm:division]{division algorithm},
    \(a = mq + r\) for some \(0 \leq r < m\).
    If, by way of contradiction, that \(m \nmid a\), then \(r > 0\),
    and we have \(r = a-mq = a - (au+bv)q = a(1-uq) + b(-vq) \in S\),
    which contradicts minimality.
    Hence we conclude that \(m \mid a\),
    and similarly we should have \(m \mid b\).
    Hence \(m\) is a common divisor of \(a,b\),
    which gives us \(m \leq (a,b)\).
    This proves statement (b).

    Now suppose \((a,b) \mid n\).
    Then for some \(c\), \(n = (a,b)c = (au+bv)c = a(uc) + b(vc) \in S\).
    This proves the reverse direction of (a).

    Lastly, if \(c \mid a\) and \(c \mid b\), then \(c \mid (au+bv) = (a,b)\).
    This proves statement (c).
\end{proof}

\begin{proposition}\label{prop:gcd-mult}
    Suppose \(a,b \in \bZ\) are not both zero, and \(m \in \bN\).
    Then \((ma,mb) = m(a,b)\).
\end{proposition}
\begin{proof}
    Let \(g = (a,b)\).
    We see that since \(g \mid a\) and \(g \mid b\),
    we have \(mg \mid ma\) and \(mg \mid mb\),
    so \(mg \mid (ma,mb)\) and \(mg \leq (ma,mb)\).
    But \(mg = (ma)x + (mb)y\) is a positive linear combination,
    so \(mg \geq (ma,mb)\) by \hyperref[thm:bezout]{B\'ezout's identity}.
\end{proof}
\begin{corollary}
    If \(m \mid c\) and \(m \mid d\), then \((c/m,d/m) = (c,d)/m\).
    In particular \((c/(c,d),d/(c,d)) = 1\).
\end{corollary}
\begin{proof}
    Apply the \hyperref[prop:gcd-mult]{proposition above}
    to \(c = ma\) and \(d = mb\).
    Then apply that conclusion to \(m = (c,d)\).
\end{proof}

\begin{definition}
    \(a,b\) are relatively prime or coprime if \((a,b) = 1\).
\end{definition}
\begin{proposition}
    If \((a,n) = (b,n) = 1\), then \((ab,n) = 1\).
\end{proposition}
\begin{proof}
    By \hyperref[thm:bezout]{B\'ezout's identity},
    we have \(1 = (a,n) = au+nv\) and \(1 = (b,n) = bx+ny\).
    Then \(1 = (au+nv)(bx+ny) = ab(ux) + n(vbx+auy+nvy)\)
    is the smallest linear combination of \(ab\) and \(n\),
    so \(1 = (ab,n)\).
\end{proof}
\begin{proposition}
    Suppose \((a,b) = 1\). If \(a \mid c\) and \(b \mid c\), then \(ab \mid c\).
\end{proposition}
\begin{proof}
    Let \(c = am = bn\), and \(ax+by = 1\).
    Then \(c = c(ax+by) = cax + cby = (bn)ax + (am)by = ab(nx+my)\).
\end{proof}
\begin{theorem}\label{thm:coprime-divisors}
    If \(d \mid ab\) and \((d,b) = 1\), then \(d \mid a\).
\end{theorem}
\begin{proof}
    Let \(ab = dn\) and \(du+bv = 1\).
    Then \(dnv = abv = a(1-du) = a - adu\),
    which implies that \(a = dnv + adu = d(nv+au)\).
\end{proof}
\begin{proposition}
    Suppose \(d \mid a\).
    Then for all \(x \in \bZ\),
    \(d \mid b\) if and only if \(d \mid (b+ax)\).
\end{proposition}
\begin{proof}
    The forward direction is obvious.
    Now suppose \(d \mid a\) and \(d \mid (b+ax)\).
    Then \(d \mid (b+ax)+a(-x) = b\).
\end{proof}
\begin{corollary}
    There exists \(x \in \bZ\) such that \((a,b) = (a,b+ax)\).
\end{corollary}
\begin{proof}
    Since the divisors are the same, the greatest one in particular should also be the same.
\end{proof}
\begin{remark}
    This is essentially how the Euclidean algorithm for finding gcd works.
    We know that by subtracting off the other number,
    the gcd does not change,
    which allows us to reduce the problem to a smaller case.
\end{remark}

\begin{definition}
    \(n \in \bZ\) is a unit if it has a multiplicative inverse \(n^{-1} \in \bZ\)
    such that \(n^{-1}n = nn^{-1} = 1\).
    \(n \in \bZ\) is irreducible if there are exactly two positive divisors.
    \(n \in \bZ\) is composite if there are three or more positive divisors.
\end{definition}
\begin{definition}
    \(p \in \bZ\) is prime if \(p \mid ab\) implies \(p \mid a\) or \(p \mid b\).
\end{definition}
\begin{theorem}[Euclid's Lemma]\label{lem:euclid}
    Irreducible elements are prime.
\end{theorem}
\begin{proof}
    \(p \mid ab\) and \(p \nmid b\) implies that \((p,b) = 1\).
    Then \(p \mid a\) using Theorem~\ref{thm:coprime-divisors}.
\end{proof}

\begin{theorem}[Fundamental Theorem of Arithmetic]
    Every integer greater than 1 factors uniquely into a product of primes.
    That is, we can write any integer \(n > 1\) as
    \begin{equation*}
        n = \prod_{i=1}^k p_i^{n_i}
    \end{equation*}
    where \(p_i\) are distinct primes.
\end{theorem}
\begin{proof}
    We first prove such a prime factorization exists.
    We can see that 2 is prime.
    Proceeding by induction, assume all integers between 2 and \(n\)
    have a prime decomposition.
    If \(n\) is prime, we are done.
    If \(n\) is not prime,
    then it must be represented by \(n = ab\),
    where \(a,b\) must both be smaller,
    and by the inductive hypothesis,
    have prime decompositions.

    Now we prove that prime factorization is unique.
    Assume the contrary,
    and let \(n\) be the smallest such integer without unique factorization.
    Then we write \(n = \prod_{i=1}^k p_i = \prod_{i=1}^{k'} q_i\),
    two distinct factorizations.
    By Euclid's lemma we see that \(p_1\) divides some \(q_i\),
    which without loss of generality say this is \(q_1\).
    Then \(p_1 = q_1\).
    So \(n/p_1 = \prod_{i=2}^k p_i = \prod_{i=2}^{k'} q_i\)
    is also an integer without unique factorization,
    which contradicts our assumption that \(n\) is the smallest.
\end{proof}
