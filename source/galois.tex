\chapter{Galois Theory}\label{sec:galois}

\section{Galois Extensions}

\begin{definition}
    A field extension \(K/F\) is a Galois extension
    if it is algebraic, separable and normal.
\end{definition}
\begin{definition}
    The group of automorphisms that fix \(F\)
    is the Galois group \(\Gal(K/F) = \Aut_F(K)\).
\end{definition}

\begin{lemma}
    Suppose \(K = F(S)\) is an extension generated by a set.
    If \(\sigma,\tau \in \Gal(K/F)\) and \(\sigma\vert_S = \tau\vert_S\),
    then \(\sigma  = \tau\).
\end{lemma}
\begin{proof}
    Write element of \(K\) as linear combination,
    apply linearity of automorphism.
\end{proof}
\begin{lemma}\label{lem:gal-permute-roots}
    Let \(\tau \in \Hom_F(K,L)\), and \(\alpha \in K\) an algebraic element.
    If \(f(x) \in F[x]\) and \(f(\alpha) = 0\),
    then \(f(\tau(\alpha)) = 0\).
    Therefore \(\tau\) permutes the roots of \(\min_F(\alpha)\),
    and \(\min_F(\alpha) = \min_F(\tau(\alpha))\).
\end{lemma}
\begin{proof}
    Simply by definition of an \(F\)-homomorphism,
    \(0 = f(\alpha) = \tau(f(\alpha)) = f(\tau(\alpha))\).
\end{proof}
\begin{corollary}
    \([K:F]\) finite implies \(\abs{\Gal(K/F)}\) finite.
\end{corollary}
\begin{proof}
    Suppose \(K = F(\alpha_1,\hdots,\alpha_n)\).
    Then \(\sigma \in \Gal(K/F)\) determined completely by \(\sigma(\alpha_i)\),
    and each one of these have finitely many possible images
    by Lemma~\ref{lem:gal-permute-roots}.
\end{proof}

\begin{definition}
    Suppose \(K\) a field, and \(G\) some group.
    A linear \(K\)-character on \(G\) is a homomorphism \(\func{\chi}{G}{L^\ast}\)
    where \(\chi(gh) = \chi(g)\chi(h)\).
\end{definition}
\begin{theorem}[Linear Independence of Characters]\label{thm:linear-independent-characters}
    Suppose \(K\) a field, and \(G\) some group.
    Let \({\{\chi_i\}}_{i=1}^n\) be distinct characters.
    Then \({\{\chi_i\}}_{i=1}^n\) is linearly independent.
\end{theorem}
\begin{proof}
    Suppose, by way of contradiction,
    that they are linearly dependent.
    Then there exists some minimal number \(m\) of nonzero coefficients \({\{a_i\}}_{i=1}^m\)
    such that
    \begin{equation*}
        a_1\chi_1 + a_2\chi_2 + \cdots + a_m\chi_m = 0
    \end{equation*}
    since without loss of generality we may assume by reordering the characters,
    the first \(m\) characters form our minimal dependent set.
    Then for any \(g \in G\), we have
    \begin{equation*}
        a_1\chi_1(g) + a_2\chi_2(g) + \cdots + a_m\chi_m(g) = 0
    \end{equation*}
    Let \(g_0 \in G\) be an element such that \(\chi_1(g_0) \neq \chi_m(g_0)\).
    Then we have
    \begin{align*}
        a_1\chi_1(g_0g) + a_2\chi_2(g_0g) + \cdots + a_m\chi_m(g_0g) &= 0 \\
        a_1\chi_1(g_0)\chi_1(g) + a_2\chi_2(g_0)\chi_2(g) + \cdots + a_m\chi_m(g_0)\chi_m(g) &= 0
    \end{align*}
    But we can also multiply by \(\chi_m(g_0)\) to get
    \begin{equation*}
        a_1\chi_m(g_0)\chi_1(g) + a_2\chi_m(g_0)\chi_2(g) + \cdots + a_m\chi_m(g_0)\chi_m(g) = 0
    \end{equation*}
    Subtracting the above two equations, we have
    \begin{equation*}
        (\chi_m(g_0)-\chi_1(g_0))a_1\chi_1(g) + (\chi_m(g_0)-\chi_2(g_0))a_2\chi_2(g)
        + \cdots + (\chi_m(g_0)-\chi_{m-1}(g_0))a_m\chi_m(g) = 0
    \end{equation*}
    We have now found a set of \(m-1\) characters that is linearly dependent,
    which contradicts minimality as assumed.
\end{proof}
\begin{corollary}[Dedekind's Lemma]\label{cor:dedekind}\label{lem:dedekind}
    Suppose \(K\) is a field, and \({\{\sigma_i\}}_{i=1}^n \subset \Aut K\).
    Then there does not exist \({\{a_i\}}_{i=1}^n \subset K\) not all zero
    such that \(a_1\sigma_1(u) + \cdots a_n\sigma_n(u) = 0\) for all \(u \in K\).
\end{corollary}
\begin{proof}
    Direct application of the \hyperref[thm:linear-independent-characters]{theorem above}.
\end{proof}

\begin{theorem}\label{thm:galois-group-size}
    \([K:F]\) finite implies \(\abs{\Gal(K/F)} \leq [K:F]\).
\end{theorem}
\begin{proof}
    Let \(n = [K:F]\), and \({\{u_i\}}_{i=1}^n\) be an \(F\)-basis of \(K\).
    Suppose \({\{\sigma_i\}}_{i=1}^{n+1} \subset \Gal(K/F)\) are distinct.
    Then consider a system of \(n\) homogeneous linear equations in \(n+1\) unknowns.
    \begin{align*}
        \sigma_1(u_1)x_1 + \sigma_2(u_1)x_2  + \cdots + \sigma_{n+1}(u_1)x_{n+1} &= 0 \\
        \sigma_1(u_2)x_1 + \sigma_2(u_2)x_2  + \cdots + \sigma_{n+1}(u_2)x_{n+1} &= 0 \\
        &\;\;\vdots \\
        \sigma_1(u_n)x_1 + \sigma_2(u_n)x_2  + \cdots + \sigma_{n+1}(u_n)x_{n+1} &= 0
    \end{align*}
    This system has a nontrivial solution.
    But that exactly contradicts \hyperref[lem:dedekind]{Dedekind's lemma}.
\end{proof}

\begin{definition}
    Suppose \(K/F\) some extension, and \(G \subseteq \Aut_F(K)\) some subset.
    The fixed field of \(G\) is
    \(K^G =\{a \in K: \sigma(u) = u,\; \forall \sigma \in G\}\).
\end{definition}
\begin{proposition}\label{prop:fixed-field-normal}
    Suppose \(F\) is a field of characteristic 0,
    \(K/F\) some finite extension, \(G = \Aut_F(K)\).
    Then \(K/F\) is normal if and only if \(K^G = F\).
\end{proposition}
\begin{proof}
    First suppose that \(K/F\) is normal.
    Observe that \(K^G \supseteq F\) by definition of an \(F\)-automorphism,
    so it suffices to prove that \(K^G \subseteq F\).
    Let \(K\) be the splitting field of a set of polynomials in \(F[x]\)
    (Theorem~\ref{thm:normal-extension-equivalence}),
    and let \(n = [K:F]\) be the degree of the extension.
    If \(n = 1\), then \(K = F\), so we are done.

    We now proceed by strong induction on \(n\),
    and assume that for any pair of fields \([K_1:F_1] < [K:F]\) we have \(K_1^{\Aut_{F_1}(K_1)} = F_1\).
    Consider any element \(\theta \in K^G\).
    Suppose \(f \in F[x]\) is irreducible of degree \(r\),
    splits completely over \(K\), and let \(\alpha\) be a root of \(f(x)\).
    Hence we can write \(f(x) = (x-\alpha)g(x)\), and that \(\deg g(x) < r\).
    Then \([K:F(\alpha)] < n\) and \(K/F(\alpha)\) is a normal extension (Theorem~\ref{thm:normal-upper}).
    By the inductive hypothesis, we have \(K^{\Aut_{F(\alpha)}(K)} = F(\alpha)\).
    Since \(\Aut_{F(\alpha)}(K) \subseteq \Aut_F(K) = G\),
    we know that \(\theta \in K^G \subseteq K^{\Aut_{F(\alpha)}(K)} = F(\alpha)\).

    Let us now write \(\theta = \sum_{i=1}^{r-1} a_i \alpha^i\), where \(a_i \in F\),
    and that the distinct roots of \(f(x)\) be \(\{\alpha=\alpha_1,\alpha_2,\hdots,\alpha_r\}\).
    Consider the polynomial \(q(x) = a_{r-1} x^{r-1} + \cdots + a_1 x + (a_0 - \theta) \in K[x]\),
    such that \(q(\alpha) = 0\) by construction.
    Since \(K/F(\alpha)\) is normal, all the roots of \(q(x)\) are in \(K\).
    By normality of \(K/F\), given any \(\sigma \in \Aut_F(K)\),
    \(\sigma(\alpha) = \alpha_j\) is another root of \(f(x)\).
    Since \(\theta\) is fixed by \(\sigma\),
    we have \(\sigma(\theta) = \theta = \sum_{i=1}^{r-1} a_i \alpha_j^i\);
    but this exactly shows that \(q(\alpha_j) = 0\), and \(q\), a polynomial of degree \(r-1\),
    has \(r\) distinct roots.
    Hence \(q(x) = 0\), and \(a_0 = \theta \in F\), so we are done.

    \medskip

    For the reverse direction, suppose \(K^G = F\).
    From Theorem~\ref{thm:normal-extension-equivalence},
    it suffices to show that \(K\) is the splitting field of a set of polynomials in \(F[x]\).
    Without any loss of generality, suppose \(a \in K\) and \(K = F(a)\),
    and consider the polynomial \(p(x) = \prod_{\sigma \in G} (x - \sigma(a))\).
    Observe that the coefficients of \(p(x)\)
    are expressible in terms of elementary symmetric polynomials of the roots \(\sigma(a)\),
    so the coefficients of \(p(x)\) are fixed by \(G\).
    We can then see that \(K\) indeed splits \(p(x)\), and hence \(K/F\) is normal.
    This argument can be repeated to get the general case of \(K = F(a_1,\hdots,a_n)\).
\end{proof}
\begin{remark}
    If \(F\) is not necessarily of characteristic 0,
    then \(K/F\) is normal if and only if \(\abs{\Aut_F(K)} \leq {[K:F]}_s\).
    We have an inequality because there are possibly repeated roots with inseparable extensions.
    In particular, if \(F\) is perfect, then \(\abs{\Aut_F(K)} \leq {[K:F]}_s\).
\end{remark}

\begin{proposition}
    Suppose \(L/K\) is a finite extension,
    \(\func{\phi}{K}{K'}\) a ring homomorphism,
    and \(L'/K'\) is normal.
    Then there is at least 1, and at most \([L:K]\) ring homomorphisms
    \(\func{\psi}{L}{L'}\) that extend \(\phi\),
    with equality if and only if \(L/K\) separable.
\end{proposition}
\begin{proof}
    Let \(n = [L:K]\) be the degree of the extension, and induct on \(n\).
    Observe that if \(n = 1\), then \(L = K\), so \(\psi = \phi\), and there is nothing to prove.

    Now, proceed via strong induction.
    Consider an element \(\alpha \in L-K\) and its corresponding extension \(M = K(\alpha)\);
    let \([K(\alpha):K] = d\).
    We know that there are at most \(d\) ring homomorphisms \(\func{\pi}{M}{L'}\) extending \(\phi\).
    By induction, we know that there are at most \([L:M]\) ring homomorphisms \(\psi\) extending \(\pi\),
    which gives us at most \(n = [L:M][M:K]\) extensions of \(\phi\).
    Equality then follows from the defintion of the degree of seperability.
\end{proof}
\begin{proposition}\label{prop:galois-group-separable}
    Let \(E/F\) be a finite extension.
    Then \(\abs{\Aut_F(E)} \leq {[E:F]}_s\),
    with equality if and only if \(E/F\) normal.
\end{proposition}
\begin{proof}
    Consider an algebraic closure \(F^\text{alg}\) of \(F\).
    By definition of the separable degree, \({[E:F]}_s = \abs{\Hom_F(E,F^\text{alg})}\).
    Choosing an element \(\sigma_0 \in \Hom_F(E,F^\text{alg})\),
    we have an injective map
    \begin{equation*}
        \varfunc{\sigma_0\circ\cdot}{\Aut_F(E)}{\Hom_F(E,F^\text{alg})}{\tau}{\sigma_0\circ\tau}
    \end{equation*}
    which gives us the inequality.

    Now suppose \(\abs{\Aut_F(E)} = {[E:F]}_s\).
    This implies that the injection above is surjective,
    which is exactly a condition that is given in Theorem~\ref{thm:normal-extension-equivalence},
    meaning \(E/F\) is normal.

    Conversely, suppose \(E/F\) is normal.
    Then Theorem~\ref{thm:normal-extension-equivalence} gives us that
    every homomorphism \(\sigma \in \Hom_F(E,F^\text{alg})\) is an automorphism in \(\Aut_F(E)\),
    so assigning \(\tau = \sigma_0^{-1}\circ\sigma\) gives us a surjective map.
\end{proof}

\begin{theorem}[Artin's Theorem]\label{thm:artin-galois}
    Suppose \(K\) is a field, and \(G \subseteq \Aut K\) some group of automorphisms,
    with order \(\abs{G} = n\).
    Let \(F = K^G\).
    Then \(K/F\) is a finite Galois extension, with \(\Gal(K/F) = G\),
    and \([K:F] = n\).
\end{theorem}
\begin{proof}
    Observe that since every element of \(G\) fixes \(F = K^G\), we have \(G \subseteq \Gal(K/F)\).
    By Theorem~\ref{thm:galois-group-size}, we also have \(n = \abs{G} \leq \abs{\Gal(K/F)} \leq [K:F]\).
    Hence it suffices to prove that \([K:F] \leq n\) to get both statements.

    Consider some \(\alpha \in K\), and some maximal set \({\{\sigma_i\}}_{i=1}^r \subset G\)
    such that the \(\sigma_i(\alpha)\) are all distinct.
    Now consider some \(\tau \in G\).
    The \(\tau\sigma_i(\alpha)\) are also all distinct, because \(\tau\) is a permutation.
    Moreover, if \(\tau\sigma_i(\alpha) \neq \sigma_i(\alpha)\) for all \(j\),
    this will contradict maximality, so in fact
    \({\{\sigma_i(\alpha)\}}_{i=1}^r = {\{\tau\sigma_i(\alpha)\}}_{i=1}^r\) for all \(\tau \in G\).
    Hence \(\alpha\) is a root of the separable polynomial
    \begin{equation*}
        f(x) = \prod_{i=1}^r (x-\sigma_i(\alpha)) = (x-\sigma_1(\alpha)) \cdots (x-\sigma_r(\alpha))
    \end{equation*}
    and this polynomial is fixed under any \(\tau \in G\).
    The coefficients of this polynomial must therefore be in \(F = K^G\).
    Therefore, every element of \(K\) is a root of a separable \(F\)-polynomial of degree at most \(n\).
    \(K\) is separable, normal, and thus Galois over \(F\),
    which by the \hyperref[prop:galois-group-separable]{proposition above} gives us that \([K:F] = n\).
\end{proof}

\begin{theorem}
    Suppose \(K/F\) is a finite field extension.
    Then the following are equivalent:
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item \(K/F\) Galois, i.e.\ \(\abs{\Aut_F(K)} = [K:F] < \infty\).
        \item \(K/F\) is the splitting field of some separable \(f(x) \in F[x]\).
        \item \(F\) is the fixed field of \(\Aut_F(K)\).
        \item \(K/F\) is normal and separable.
    \end{enumerate}
\end{theorem}
\begin{proof}
    (i) \(\iff\) (iv) is by definition.

    (i) \(\iff\) (ii) is a direct consequence of Theorem~\ref{thm:normal-extension-equivalence}.

    (iii) implies (i) is given by the \hyperref[thm:artin-galois]{theorem above}.

    (i) implies (iii) is proven in the \hyperref[lem:galois-injective]{lemma below}.
\end{proof}

\subsection*{Fundamental Theorem of Galois Theory}

\begin{definition}
    Suppose \(K/F\) is a Galois extension,
    and \(F \subseteq E \subseteq K\) is a tower of extensions.
    Let \(G = \Gal(K/F)\) be the corresponding Galois group.
    A subgroup \(H \subseteq G\) is said to belong to a subfield \(E\)
    if it is the fixed field \(E = K^H\).
\end{definition}
\begin{lemma}\label{lem:galois-injective}
    Suppose \(K/F\) is a Galois extension.
    We have that:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item if \(G = \Gal(K/F)\) is a Galois group, then \(F = K^G\);
        \item if \(F \subseteq E \subseteq K\), then \(K/E\) is a Galois extension; and
        \item the map \(E \mapsto \Gal(K/E)\) from the set of intermediate field extensions
            to the set of subgroups of \(G\) is injective.
    \end{enumerate}
\end{lemma}
\begin{proof}
    Consider \(\alpha \in K^G\), and \(\sigma \in G\).
    % and an \(F\)-embedding \(\func{\sigma}{F(\alpha)}{K^\text{sep}}\) to the separable closure.
    % Allow us to extend \(\sigma\) to a \(K\)-embedding \(\func{\sigma}{K}{K_\text{sep}}\).
    Since \(\alpha \in K^G\), \(\sigma\) fixes \(\alpha\),
    so \(\alpha\) must be inseparable,
    which gives us that \(\alpha \in F\) as \(K\) is separable over \(F\).
    This allows us to conclude that \(K^G \subseteq F\), and the reverse inclusion is trivially true,
    proving statement (a).

    Statement (b) is easy, since \(K/E\) is normal by Theorem~\ref{thm:normal-upper},
    and it is also algebraic and separable by transitivity,
    proven in Corollary~\ref{cor:algebraic-extension-stack} and Theorem~\ref{thm:separable-transitive}.

    Consider two intermediate fields \(E,E'\),
    and let \(H = \Gal(K/E)\) and \(H' = \Gal(K/E')\).
    % By statement (c) of the \hyperref[lem:galois-group-properties]{lemma below},
    % we know that \(K^{\Gal(K/E)} = K^{\Gal(K/E')}\).
    % By statement (f) of the \hyperref[lem:galois-group-properties]{lemma below},
    % we have that \(\)
    By statement (a), we have that \(E = K^H\) and \(E' = K^{H'}\).
    Suppose \(H = H'\).
    Then \(E = K^H = K^{H'} = E'\) as desired,
    proving statement (c).
\end{proof}
\begin{lemma}\label{lem:galois-group-properties}
    Suppose \(K/F\) is Galois. Then the following hold:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item If \(L_1 \subseteq L_2\) are subfields of \(K\),
            then \(\Gal(K/L_1) \supseteq \Gal(K/L_2)\).
        \item If \(L\) is a subfield of \(K\),
            then \(L \subseteq K^G\) where \(G = \Gal(K/L)\).
        \item If \(S_1 \subseteq S_2\) are subsets of \(\Aut K\),
            then \(K^{S_1} \supseteq K^{S_2}\).
        \item If \(S\) is a subset of \(\Aut K\),
            then \(S \subseteq \Gal(K/K^S)\).
        \item If \(L = K^S\) for some \(S \subseteq \Aut K\),
            then \(L = K^G\) where \(G = \Gal(K/L)\).
        \item If \(H = \Gal(K/L)\) for some subfield \(L\),
            then \(H = \Gal(K/K^H)\).
    \end{enumerate}
\end{lemma}
\begin{proof}
    Consider any \(\tau \in \Gal(K/L_2)\).
    Since every element of \(L_1\) is in \(L_2\),
    \(\tau\) must fix \(L_1\) because it already fixes \(L_2\).
    Hence \(\tau \in \Gal(K/L_1)\).
    This proves statement (a).

    Statement (b) condenses down to the fact that \(K^G\),
    which is the fixed field of \(\Gal(K/L)\),
    i.e.\ the automorphisms that fix \(L\),
    must contain \(L\).
    This is almost a tautology.

    Consider any element of \(K^{S_2}\).
    It must be fixed by any automorphism \(\tau \in S_2\),
    so in particular it must also be fixed by any automorphism in \(S_1\).
    Hence our element is in \(K^{S_1}\).
    This proves statement (c).

    Consider any automorphism \(\tau \in S\).
    By definition, it fixes elements in its fixed field \(K^S\),
    and hence, it is an automorphism that fixes \(K^S\),
    i.e.\ \(\tau \in \Gal(K/K^S)\).
    This proves statement (d).

    % Let \(G = \langle S \rangle\) be the group generated by \(S\).
    % Let \(L = K^S\) and \(G = \Gal(K/L)\)
    % By (d), we have \(S \subseteq G = \Gal(K/L)\),
    % so by (c), we have \(K^G \subseteq K^S = L\).
    % For the reverse inclusion,
    Since \(K/F\) is Galois, \(K/L\) is Galois by the \hyperref[lem:galois-injective]{lemma above},
    % so \(K^G = L\) (Proposition~\ref{prop:fixed-field-normal}).
    and thus \(G = \Gal(K/L)\) is a Galois group, meaning \(L = K^G\).
    This proves statement (e).

    Now, let \(H = \Gal(K/L)\), by (b) we have \(L \subseteq K^H\),
    so by (a) we have \(H = \Gal(K/L) \supseteq \Gal(K/K^H)\).
    On the other hand, by (d) we have \(H \subseteq \Gal(K/K^H)\),
    giving us equality for statement (f).
\end{proof}
\begin{corollary}\label{cor:galois-correspondence}
    Suppose \(K/F\) is Galois, and \(G = \Gal(K/F)\).
    Then there is an inclusion-reversing correspondence between subgroups of the form \(\Gal(K/L)\)
    and subfields of the form \(K^S\).
\end{corollary}
\begin{proof}
    The mapping is given by
    \begin{align*}
        \Bqty{\text{subgroups}\;\Gal(K/L) \subseteq G} &\leftrightarrow
        \Bqty{\text{subfields}\; K^S \subseteq K} \\
        H &\mapsto K^H \\
        \Gal(K/L) &\mapsfrom L
    \end{align*}
    The inclusion-reversing part is given
    by statements (a) and (c) from the \hyperref[lem:galois-group-properties]{lemma above}.
    % Two-sided inversion is given
    % by statements (e) and (f) from the \hyperref[lem:galois-group-properties]{lemma above}.

    To prove two-sided inversion,
    first consider an intermediate field \(L\) and its Galois group \(H = \Gal(K/L)\);
    by Lemma~\ref{lem:galois-injective}a, we know that \(L = K^H = K^{\Gal(K/L)}\) indeed.
    Now consider a subgroup \(H\);
    since \(K^H\) is a subfield,
    by statement (f) from the \hyperref[lem:galois-group-properties]{lemma above},
    \(H = \Gal(K/K^H)\) as desired.
\end{proof}
\begin{corollary}\label{cor:galois-intersection-product}
    Suppose \(K/F\) is a Galois extension, and \(G = \Gal(K/F)\) its corresponding Galois group.
    If \(E = K^H, E' = K^{H'}\) are intermediate fields belonging to subgroups \(H,H'\),
    then \(H \cap H'\) belongs to \(EE' = K^{H \cap H'}\)
    and \(HH'\) belongs to \(E \cap E' = K^{HH'}\).
\end{corollary}
\begin{proof}
    Let \(a \in E\) and \(a' \in E'\).
    If \(h \in H \cap H'\), then \(h(aa') = h(a)h(a') = aa'\)
    so indeed \(EE' \subseteq K^{H \cap H'}\).
    By statement (a) from the \hyperref[lem:galois-group-properties]{lemma above},
    we know that \(\Gal(K/EE') \supseteq \Gal(K/K^{H \cap H'}) = H \cap H'\).
    % But we also know that \(\Gal(K/EE') \subseteq \Gal(K/K^{\Gal(K/EE')}) = \) from statement (c),
    But given any \(\sigma \in \Gal(K/EE')\), \(\sigma\) fixes \(EE'\),
    so \(\sigma\) must fix \(E\) and \(E'\) individually,
    meaning \(\sigma \in H\) and \(\sigma \in H'\);
    hence we have \(\Gal(K/EE') \subseteq H \cap H'\).

    Similarly, consider \(hh' \in HH'\), and some \(a \in E \cap E'\).
    Then \(hh'(a) = h(a) = a\), so indeed \(E \cap E' \subseteq K^{HH'}\).
    Now, because \(H \subseteq HH'\), we have \(K^{HH'} \subseteq K^H = E\);
    by the exact same logic, we also have \(K^{HH'} \subseteq K^{H'} = E'\),
    which when combined gives us \(K^{HH'} \subseteq E \cap E'\).
\end{proof}

\begin{theorem}\label{thm:normal-galois}
    Suppose \(K/F\) Galois and \(G = \Gal(K/F)\).
    Suppose \(L\) is a subfield with \(F \subseteq L \subseteq K\) and \(H = \Gal(K/L)\).
    Then \(L\) is normal over \(F\) if and only if \(H \subseteq G\) is a normal subgroup.
    If \(L\) is indeed normal over \(F\),
    then restricting \(\Gal(K/F) \to \Gal(L/F)\) via \(\sigma \mapsto \sigma\vert_L\)
    is a surjective homomorphism with kernel \(H\).
    Hence \(\Gal(L/F) \cong G/H\).
\end{theorem}
\begin{proof}
    First suppose \(L/F\) is normal.
    Observe that given any \(\sigma \in \Gal(K/F)\),
    restricting it to \(L\) gives us a \(\eval{\sigma}_L \in \Hom_F(L,K)\),
    and the restriction map is clearly a homomorphism;
    but then Theorem~\ref{thm:normal-extension-equivalence} tells us that
    \(\eval{\sigma}_L \in \Aut_F(K) = \Gal(K/F)\).
    This map is surjective, because field homomorphisms are extendable
    (Proposition~\ref{prop:algebraic-extension-hom}),
    and by the \hyperref[thm:iso-1-group]{first isomorphism theorem},
    the kernel is a normal subgroup \(H\).

    Now suppose that \(L/F\) is not normal.
    By Theorem~\ref{thm:normal-extension-equivalence},
    there exists some \(\lambda \in \Hom_F(L,K)\) that is not an automorphism.
    Since \(K/F\) is Galois and thus normal,
    we extend \(\lambda\) to an automorphism \(\tau\) in \(\Aut_F(K)\)
    (Theorem~\ref{thm:normal-extension-equivalence}).
    Since we know that field homomorphisms are injective, \(\img\lambda \cong L\).
    As \(\Gal(K/L)\) and \(\Gal(K/\img\lambda)\) are isomorphic under conjugation by \(\lambda\),
    we have two distinct subgroups that are conjugate to each other.
    By definition, they cannot be normal subgroups.
\end{proof}

\begin{definition}
    A Galois extension \(K/F\) is abelian/cyclic
    if \(\Gal(K/F)\) is abelian/cyclic.
\end{definition}
\begin{proposition}
    If \(K/F\) is abelian/cyclic, and \(L\) is an intermediate field,
    then \(L/F\) is abelian/cylic.
\end{proposition}
\begin{proof}
    Subgroups and quotient groups of abelian/cyclic groups are abelian/cyclic.
\end{proof}

\begin{theorem}\label{thm:galois-compositum-intersection}
    Suppose \(K/F\) Galois, and \(L/F\) some other extension,
    under the assumption that \(K,L\) are both contained in some extension.
    Then \(KL/L\) and \(K/(K \cap L)\) are Galois.
\end{theorem}
\begin{proof}
    Observe that \(K \cap L\) is a subfield of \(K\).
    Then \(K/(K \cap L)\) is Galois by Lemma~\ref{lem:galois-injective}b.

    We now claim that \(\Gal(KL/L) \cong \Gal(K/F)\).
    Consider the restriction map
    \begin{equation*}
        \varfunc{\eval{}_K}{\Gal(KL/L)}{\Gal(K/K \cap L)}{\sigma}{\eval{\sigma}_K}
    \end{equation*}
    Obviously, because \(K/F\) is Galois, any \(\eval{\sigma}_K\) must be in \(\Aut_F(K)\).
    This is also injective, because if \(\eval{\sigma}_K = \id\) and \(\sigma\) fixes \(L\),
    then it must be also the identity on \(KL\).
    Then it suffices to show that the image fixes exactly \(K \cap L\).
    Obviously, restricting to \(K\) fixes elements in \(L\);
    conversely, if \(a \in K\) with \(\tau(a) = a\) for all \(\tau\) in the image,
    then \(a\) is also fixed under \(\Gal(KL/L)\) in the first place,
    so we must have \(a \in L\).
    Hence the image must fix only \(K \cap L\) as desired.
    % Consider the groups \(G = \Gal(K/F)\) and \(H = \Gal(KL/L)\).
    % Let \(G = \Gal(K/F)\) and \(H = \Gal(L/F)\) (with \(H\) not necessarily Galois).
    % By Corollary~\ref{cor:galois-intersection-product},
    % \(\Gal(KL/F) = G \cap H\).
    % By Theorem~\ref{thm:normal-galois},
    % it suffices to prove that \(H \subseteq G\) is a normal subgroup.
    % Consider any \(\sigma \in \Gal(KL/L)\) and any \(\lambda \in \Gal(K/F)\).
    % We want to first show that \(\img\lambda\sigma\lambda^{-1} = KL\).
\end{proof}
\begin{corollary}\label{cor:galois-compositum-degree}
    Suppose \(K/F\) finite Galois, and \(L/F\) some extension.
    Then \([KL:L] \mid [K:F]\), and \([KL:F] = [K:F][L:F]\) if and only if \(K \cap L = F\).
\end{corollary}
\begin{proof}
    From the \hyperref[thm:galois-compositum-intersection]{theorem above},
    \(\Gal(KL/L) \cong \Gal(K/(K \cap L))\),
    so we can equate their cardinality, meaning \([KL:L] = [K/(K \cap L)] \mid [K:F]\).
    Moreover, we have
    \begin{equation*}
        [KL:F] = [KL:L][L:F] = [K:(K \cap L)][L:F] = \frac{[K:F][L:F]}{[(K \cap L):F]}
    \end{equation*}
\end{proof}
\begin{remark}
    This does not hold if \(K/F\) not Galois.
\end{remark}

\begin{theorem}
    Suppose \(K/F\) and \(L/F\) both finite Galois,
    such that \(K,L \subseteq \overline{F}\).
    Then \[\vfunc{\psi}{\Gal(KL/F)}{\Gal(K/F)\times\Gal(L/F)}{\sigma}{(\sigma\vert_K,\sigma\vert_L)}\]
    is injective,
    and isomorphism if \(K \cap L = F\).
\end{theorem}
\begin{proof}
    Clearly this is a homomorphism.
    Consider the kernel of this map.
    In that case, \(\sigma \in \Gal(KL/F)\) must fix both \(K\) and \(L\),
    so \(\sigma\) must be the identity map in \(KL\).
    We have injectivity.

    Now, suppose \(K \cap L = F\).
    Then by the \hyperref[cor:galois-compositum-degree]{corollary above},
    we have \([KL:F] = [K:F][L:F]\), so the cardinalities matches up,
    and we have an isomorphism.
\end{proof}
\begin{corollary}
    Suppose \(E/F\) finite separable.
    Then there exists an extension \(K \supseteq E\) such that
    \(K/F\) is Galois and minimal with respect to this property.
\end{corollary}
\begin{proof}
    Consider a basis of \(E/F\).
    Note that these elements all have separable minimal polynomials,
    so the composite of all of these extensions must be Galois,
    i.e.\ there exists at least one Galois extension of \(F\) containing \(E\).
    Now, consider the intersection of all Galois extensions of \(F\) containing \(E\).
    This must then be \(K\).
\end{proof}
\begin{definition}
    This minimal extension \(K \supseteq E \supseteq F\) is called the Galois closure of \(E/F\).
\end{definition}

\begin{definition}
    Suppose \(K/F\) Galois, \(G = \Gal(K/F)\),
    \(\sigma \in G\), \(\alpha \in K\).
    \(\{\sigma(\alpha) : \sigma \in G\}\) is the set of Galois conjugates of \(\alpha\).
    If \(F \subseteq E \subseteq K\) tower of extensions,
    then \(\sigma(E)\) is the conjugate of \(E\).
\end{definition}
\begin{proposition}\label{prop:galois-conjugate}
    Suppose \(F \subseteq E \subseteq K\) is a tower of field extensions,
    \(K/F\) finite Galois, and \(G = \Gal(K/F)\).
    Suppose that \(H \subseteq G\) corresponds to \(E\).
    For \(\sigma \in G\),
    the subgroup corresponding to \(\sigma(E)\)
    is the conjugate subgroup \(\sigma H \sigma^{-1} \subseteq G\).
    In particular \([E:F] = [G:H]\).
\end{proposition}
\begin{proof}
    Let \(\sigma(E)\) correspond to the group \(H'\).
    Given any \(a \in E\) and \(h \in H\), observe that
    \begin{equation*}
        (\sigma h \sigma^{-1})(\sigma(a)) = \sigma h(a) = \sigma(a)
    \end{equation*}
    so \(\sigma H \sigma^{-1}\) indeed fixes \(\sigma(E)\),
    giving us \(\sigma H \sigma^{-1} \subseteq H'\).
    But we also know that \(E/F \cong \sigma(E)/F\),
    so \([E:F] = [\sigma(E):F]\) and thus \([K:E] = [K:\sigma(E)]\).
    Therefore we can equate the cardinalities, and we have \(\sigma H \sigma^{-1} = H'\).

    The last statement then follows from the multiplicativity of degrees of field extensions.
\end{proof}

\begin{theorem}[Fundamental Theorem of Galois Theory]\label{thm:ftgt}
    Suppose \(K/F\) is (finite) Galois, and \(G = \Gal(K/F)\).
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item There is an inclusion-reversing bijection
            between intermediate fields \(E\) of \(K/F\) and subgroups \(H \subseteq G\),
            given by associating \(H\) with \(K^H = E\);
        \item \([K:E] = \abs{H}\) and \([K:F] = \abs{G}\);
        \item \(K/E\) is Galois, \(H = \Gal(K/E)\);
        \item embeddings \(E\) to \(\overline{F}\) are in bijection with
            left cosets of \(H\) in \(G\);
        \item \(E/F\) is Galois if and only if \(H \subseteq G\) is a normal subgroup,
            and in that case, \(\Gal(E/F) \cong G/H\);
        \item intersection of subgroups correspond to compositum of fields
            and joins of subgroups correspond to intersection of fields; and
        \item lattice of subgroups of \(G\) correspond under this bijection
            to inverted lattice of intermediate field extensions.
    \end{enumerate}
\end{theorem}
\begin{proof}
    From Corollary~\ref{cor:galois-correspondence}, in order to prove statement (a),
    it suffices to prove that every subfield is a fixed field,
    and every subgroup is a Galois group.
    By \hyperref[thm:artin-galois]{Artin's theorem}, every group is a Galois group.
    Moreover, given any field \(L\), \(K/L\) is Galois,
    so \(H = \Gal(K/L)\) is a Galois group, and hence \(L = K^H\) is a fixed field
    (Lemma~\ref{lem:galois-injective}).
    % Moreover, given any field \(L\), we can find its Galois group

    Statement (b) is a direct result of \hyperref[thm:artin-galois]{Artin's theorem}.

    Statement (c) is a direct result of Lemma~\ref{lem:galois-injective}b and Corollary~\ref{cor:galois-correspondence}.

    Statement (d) follows from Proposition~\ref{prop:galois-conjugate}.
    
    Statement (e) summarize Theorem~\ref{thm:normal-galois}.

    Statement (f) is a restatement of Theorem~\ref{thm:galois-compositum-intersection}.

    Lastly, statement (g) is a restatement of statement (a).
\end{proof}

\begin{theorem}[Newton's theorem]
    Suppose \(F\) a field, and \(K = F(x_1,\hdots,x_n)\)
    the field of rational functions in \(n\) variables.
    Suppose \(S\) is the subfield of symmetric rational functions.
    Then:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \([K:S] = n!\)
        \item \(\Gal(K/S) = S_n\)
        \item Let \({\{s_i\}}_{i=1}^n\) be the elementary symmetric polynomials.
            Then \(S = F(s_1,\hdots,s_n)\).
        \item \(K\) is the splitting field over \(S\) of
            \(t^n - s_1 t^{n-1} + s_2 t^{n-2} - \cdots + {(-1)}^n s_n\).
    \end{enumerate}
\end{theorem}
\begin{proof}
    Permuting the \(x_i\) is clearly an automorphism that fixes symmetric rational functions,
    so \(S_n \subseteq \Gal(K/S)\).
    We obviously have \(S \supseteq F(s_1,\hdots,s_n)\),
    since the elementary symmetric polynomials are all symmetric and rational by definition.
    Hence we have
    \begin{equation*}
        [K:F(s_1,\hdots,s_n)] \geq [K:S] = \abs{\Gal(K/S)} \geq \abs{S_n} = n!
    \end{equation*}
    It suffices to show that \([K:F(s_1,\hdots,s_n)] \leq n!\)
    to get enough equalities to prove the first three statements.

    Consider the polynomial
    \begin{equation*}
        p(t) = t^n - s_1 t^{n-1} + s_2 t^{n-2} - \cdots + {(-1)}^n s_n \in F[s_1,\hdots,s_n][t]
    \end{equation*}
    which factors into
    \begin{equation*}
        p(t) = (t-x_1)(t-x_2)\cdots(t-x_n)
    \end{equation*}
    Clearly this cannot split over any proper subfield of \(K\),
    so \(K\) must be the splitting of field of \(p(t)\),
    and thus is normal over \(F(s_1,\hdots,s_n)\) (Theorem~\ref{thm:normal-extension-equivalence}).
    In particular, extending by \(x_n\) is at most degree \(n\),
    and then sequentially extending by \(x_i\) is at most degree \(i\),
    so \([K:F(s_1,\hdots,s_n)] \leq n!\) as desired.
    Along the way we have also proved statement (d).
\end{proof}

\subsection*{Finite fields}

\begin{theorem}\label{thm:finite-field-cyclic-galois}
    Let \(F = \bF_{p^n}\).
    Then \([F:\bF_p] = n\), \(F/\bF_p\) is cyclic,
    and the Galois group is generated by the Frobenius automorphism.
\end{theorem}
\begin{proof}
    We know that \([F:\bF_p] = n\) and \(F^\times\) is cyclic.
    Moreover, \(F/\bF_p\) is separable and normal, since it is the splitting field of \(x^{p^n} - x\);
    thus it is Galois.
    The fixed field of the Frobenius automorphism is \(\bF_p\),
    so it generates the Galois group.
    Hence \(F/\bF_p\) is cyclic of order \(n\).
\end{proof}
\begin{corollary}\label{cor:cyclic-galois-frobenius}
    Suppose \(K/F\) is an extension of finite fields of characteristic \(p\).
    Then \(K/F\) is cyclic Galois.
    If \(\abs{F} = p^n\), then \(\Gal(K/F) = \langle\tau\rangle\)
    where \(\vfunc{\tau}{K}{K}{a}{a^{p^n}}\).
\end{corollary}
\begin{proof}
    Suppose \([K:\bF_p] = m\).
    Then by the \hyperref[thm:finite-field-cyclic-galois]{theorem above},
    \(G = \Gal(K/\bF_p)\) is cyclic of order \(m\),
    and \(G\) is generated by the Frobenius automorphism \(\Phi\).
    Hence \(H = \Gal(K/F)\) is a subgroup of \(G\), which must also be cyclic.

    Now, if \(\abs{F} = p^n\), observe that \(\Phi^n(a) = a^{p^n}\).
    Since \([F:\bF_p] = n\) and all extensions are Galois, this suffices.
\end{proof}

\begin{theorem}
    Suppose \(\overline{\bF}_p\) an algebraic closure of \(\bF_p\).
    For any \(n \in \bN\),
    there exists a unique subfield of \(\overline{\bF}_p\) of order \(p^n\).
    If \(K,L\) are subfields of \(\overline{\bF}_p\) of orders \(p^m\) and \(p^n\),
    then \(K \subseteq L \iff m \mid n\).
    In that case, \(L/K\) Galois with \(\Gal(L/K)\) cyclic,
    generated by \(\vfunc{\tau}{L}{L}{a}{a^{p^m}}\).
\end{theorem}
\begin{proof}
    Consider an arbitrary \(n \in \bN\).
    The set of roots of \(x^{p^n} - x\) (in \(\overline{\bF}_p\)) has \(p^n\) elements.
    This can be our subfield of order \(p^n\).
    As two such subfields are isomorphic over \(\bF_p\), it must therefore be unique.

    Now consider \(\abs{K} = p^m\) and \(\abs{L} = p^n\).
    Of course \(K \subseteq L\) happens if and only if \(m \mid n\)
    (Proposition~\ref{prop:finite-field-size-inclusion}).
    Then by the \hyperref[cor:cyclic-galois-frobenius]{corollary above},
    we have our \(\tau\) generating the Galois group.
\end{proof}

\begin{remark}
    We shall not prove in detail, but the Galois group \(\Gal(\overline{\bF}_p/\bF_p)\)
    is the inverse limit \(\invlim \Gal(\bF_{p^n}/\bF_p)\) (which we construct in Section~\ref{sec:universality}),
    which is the topological completion \(\hat{\bZ}\) of the integers (which we define in Section~\ref{sec:completions}).
\end{remark}

\subsection*{Simple extensions}

\begin{definition}
    Suppose \(E/F\) is an extension.
    \(E/F\) is simple (algebraic)
    if \(E = F(\alpha)\) (for \(\alpha\) algebraic over \(F\)).
    We call \(\alpha\) a primitive element of \(E/F\).
\end{definition}
\begin{lemma}\label{lem:simple-finite-subfields}
    Suppose \(K/F\) is a finite extensions.
    Then \(K = F(\theta)\) is simple extensions
    if and only if there are only finitely many subfields of \(K\) containing \(F\).
\end{lemma}
\begin{proof}
    First suppose \(K = F(\theta)\) is simple.
    Consider an intermediate field \(E\),
    and let \(f(x) = \min_F(\theta)\) and \(g(x) = \min_E(\theta)\)
    be the minimal polynomials in \(F\) and \(E\) respectively.
    Then \(g \mid f\) inside \(E[x]\).
    Now let \(E'\) be the field generated by the coefficients of \(g\) over \(F\).
    Obviously, \(E' \subseteq E\) and \(g(x) = \min_{E'}(\theta)\),
    so we have \([K:E] = \deg g(x) = [K:E']\), which implies \(E = E'\).
    Hence subfields of \(K\) containing \(F\) must be generated by the coefficients of factors of \(f\),
    and thus wholly determined by the factors of \(f\), of which there are finitely many.

    Now suppose there are only finitely many subfields of \(K\) containing \(F\).
    The case of \(F\) being a finite field is already covered by Corollary~\ref{cor:finite-field-simple},
    so we assume that \(F\) is infinite.
    As \(K\) is finitely generated over \(F\),
    it suffices to show that \(F(a,b)\) is generated by a single element,
    since we can then complete the argument by induction.
    Consider the subfields \(F(a + \lambda b)\) for any \(\lambda \in F\).
    Since there are infinitely many choices of \(\lambda\) but only finitely many subfields that contain \(F\),
    there must exist some \(\lambda \neq \lambda'\) such that \(F(a + \lambda b) = F(a + \lambda' b)\).
    Hence \(a + \lambda b, a + \lambda' b \in F(a + \lambda b)\),
    which implies their difference \((\lambda-\lambda')b \in F(a + \lambda b)\),
    and thus \(b \in F(a + \lambda b)\), which also implies \(a \in F(a + \lambda b)\).
    Therefore we have \(F(a,b) \subseteq F(a + \lambda b)\),
    and the reverse inclusion is trivially true.
    We have equality.
\end{proof}
\begin{theorem}[Primitive Element Theorem]\label{thm:primitive-element-2}
    Suppose \(K/F\) is a finite separable extension.
    Then \(K = F(\theta)\) is simple.
\end{theorem}
\begin{proof}
    Let \(L\) be the Galois closure of \(K/F\).
    Then any subfield of \(K\) containing \(F\) must correspond to a subgroup of \(\Gal(L/F)\)
    by the \hyperref[thm:ftgt]{fundamental theorem of Galois theory},
    and there are only finitely many such subgroups.
    By the \hyperref[lem:simple-finite-subfields]{lemma above},
    there are finitely many subfields of \(K\) containing \(F\),
    so we are done.
\end{proof}


\section{Abelian Extensions}

\begin{definition}
    Suppose \(f(x) \in F[x]\) irreducible and separable,
    and \(K\) splitting field of \(f(x)\).
    Then \(K/F\) is Galois,
    and the Galois group of the polynomial is \(\Gal(f) = \Gal(K/F)\).
\end{definition}
\begin{theorem}
    Suppose \(f(x) \in F[x]\) separable of degree \(n\).
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item If \(f(x)\) is irreducible over \(F\),
            then \(n \mid \abs{\Gal(f)}\).
        \item \(f(x)\) is irreducible over \(F\)
            if and only if \(\Gal(f)\) is a transitive subgroup of \(S_n\).
    \end{enumerate}
\end{theorem}
\begin{proof}
    Suppose \(f\) is irreducible.
    Let \(K\) be the splitting field of \(f\).
    If \(a\) is a root of \(f\), then \(n = [F(a):F] \mid [K:F] = \abs{\Gal(f)}\).
    This proves statement (a).

    Suppose \(f\) is irreducible again.
    Let \(K/F\) be a splitting field, and let \(G = \Gal(f) = \Gal(K/F)\) be its Galois group.
    Consider two roots \(r_i,r_j\) of \(f\).
    There exists some \(\sigma \in \Gal(f)\) such that \(\sigma(r_i) = r_j\),
    since \(f\) is separable by assumption.
    Every root can be reached from any other root, so \(G\) is transitive;
    moreover, these \(\sigma\) are transpositions, so \(G\) is a subgroup of \(S_n\).
    This proves the forward direction of statement (b).

    Now suppose \(f\) is reducible. Then clearly \(n \geq 2\).
    Allow us to factor \(f\) into a product of irreducible factors.
    By separability, each of the irreducible factors have distinct roots from each other.
    Then given any root, the irreducible factor that it belongs to
    forms the minimal polynomial for the root.
    Hence it is impossible for us to get a automorphism \(\sigma \in \Gal(f)\)
    where \(\sigma(r_i) = r_j\) if \(r_i\) and \(r_j\) belong to different factors.
    In that case, \(\Gal(f)\) is not transitive.
    This proves the reverse direction of statement (b).
\end{proof}

\begin{definition}
    Suppose \(K/F\) is finite Galois.
    The norm is a multiplicative homomorphism
    \[\varfunc{N_{K/F}}{K}{F}{a}{\displaystyle\prod_{\sigma\in\Aut_F(K)} \sigma(a)}\]
    and the trace is an additive homomorphism
    \[\varfunc{\tr_{K/F}}{K}{F}{a}{\displaystyle\sum_{\sigma\in\Aut_F(K)} \sigma(a)}\]
\end{definition}
\begin{proposition}
    Suppose \(K/F\) finite Galois,
    \(a \in K\) some element with minimal polynomial \(f(x) = \min_F(a) \in F[x]\).
    Let \(f(x) = \prod_{\sigma\in\Gal(K/F)} (x-\sigma(a)) = x^n + a_1 x^{n-1} + \cdots + a_n\).
    Then \(N_{K/F} = {(-1)}^n a_n\)  and \(\Tr_{K/F} = -a_1\).
\end{proposition}
\begin{proof}
    First observe that
    \begin{equation*}
        a_n = \prod_{\sigma\in\Gal(K/F)} \pqty{-\sigma(a)} = {(-1)}^n \prod_{\sigma\in\Gal(K/F)} \sigma(a)
    \end{equation*}
    Then observe that
    \begin{equation*}
        a_1 = \sum_{\sigma\in\Gal(K/F)} \pqty{-\sigma(a)} = - \sum_{\sigma\in\Gal(K/F)} \sigma(a)
    \end{equation*}
\end{proof}

\begin{theorem}[Hilbert Theorem 90]
    Suppose \(K/F\) is cyclic of degree \(n\), and \(G = \Gal(K/F) = \langle\sigma\rangle\).
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item For any \(b \in K^\times\), \(N_{K/F}(b) = 1\) if and only if \(b = a/\sigma(a)\) for some \(a \in K\).
        \item For any \(b \in K\), \(\Tr_{K/F}(b) = 0\) if and only if \(b = a - \sigma(a)\) for some \(a \in K\).
    \end{enumerate}
\end{theorem}
\begin{proof}
    Let \(n = [K:F]\), and consider a \(b \in K^\times\) with norm 1.
    Consider the map
    \[\varfunc{\Theta}{K}{F}{t}{t + b\sigma(t) + b\sigma(b)\sigma^2(t) + \cdots + b\sigma(b)\cdots\sigma^{n-2}(b)\sigma^{n-1}(t)}\]
    By \hyperref[lem:dedekind]{Dedekind's lemma}, this is a nonzero map,
    so let us find \(c \in K\) such that \(\Theta(c) \neq 0\), and let
    \[a = \Theta(c) = c + b\sigma(c) + b\sigma(b)\sigma^2(c) + \cdots + b\sigma(b)\cdots\sigma^{n-2}(b)\sigma^{n-1}(c)\]
    Then observe that
    \begin{align*}
        b\sigma(a) &= b\sigma(c) + b\sigma(b)\sigma^2(c) + \cdots + b\sigma(b)\cdots\sigma^{n-2}(b)\sigma^{n-1}(c) + b\sigma(b)\cdots\sigma^{n-1}(b)\sigma^n(c) \\
        &= b\sigma(c) + b\sigma(b)\sigma^2(c) + \cdots + b\sigma(b)\cdots\sigma^{n-2}(b)\sigma^{n-1}(c) + N_{K/F}(b) \cdot c \\
        &= c + b\sigma(c) + b\sigma(b)\sigma^2(c) + \cdots + b\sigma(b)\cdots\sigma^{n-2}(b)\sigma^{n-1}(c)
        = a
    \end{align*}
    so indeed \(b = a/\sigma(a)\).
    
    For the reverse direction, let us compute the telescoping product
    \begin{equation*}
        N_{K/F}(b) = \prod_{k=0}^{n-1} \sigma^k(b) = \prod_{k=0}^n \frac{\sigma^k(a)}{\sigma^{k+1}(a)}
        = \frac{\id(a)}{\sigma^n(a)} = \frac{a}{a} = 1
    \end{equation*}
    which proves statement (a).
    
    \medskip
    
    Similarly, consider a \(b \in K\) with zero trace.
    Since the trace map is a sum of the automorphisms,
    by \hyperref[lem:dedekind]{Dedekind's lemma}, it is a nonzero map,
    so let us find a \(c \in K\) such that \(\Tr_{K/F}(c) \neq 0\).
    Now let
    \begin{equation*}
        a = \frac{1}{\Tr(c)}\pqty{bc + (b+\sigma(b))\sigma(c) + \cdots + (b+\sigma(b)+\cdots+\sigma^{n-2}(b))\sigma^{n-2}(c)}
    \end{equation*}
    Notice that
    \begin{align*}
        a - \sigma(a) &= \frac{1}{\Tr(c)}\pqty{bc + (b+\sigma(b))\sigma(c) + \cdots + (b+\sigma(b)+\cdots+\sigma^{n-2}(b))\sigma^{n-2}(c)} \\
        &\quad - \frac{1}{\Tr(c)}\pqty{\sigma(b)\sigma(c) + (\sigma(b)+\sigma^2(b))\sigma^2(c) + \cdots + (\sigma(b)+\sigma^2(b)+\cdots+\sigma^{n-1}(b))\sigma^{n-1}(c)} \\
        &= \frac{1}{\Tr(c)}\pqty{b(c + \sigma(c) + \cdots + \sigma^{n-2}(c) + \sigma^{n-1}(c)) - \Tr(b)\sigma^{n-1}(c)}
        = b \frac{\Tr(c)}{\Tr(c)} = b
    \end{align*}
    and we have the forward direction.

    For the reverse direction, let us compute the telescoping sum
    \begin{equation*}
        \Tr_{K/F}(b) = \sum_{k=0}^{n-1} \sigma^k(b) = \sum_{k=0}^{n-1} \sigma^k(a) - \sigma^{k+1}(a)
        = \id(a) - \sigma^n(a) = a - a = 0
    \end{equation*}
    which proves statement (b).
\end{proof}

\subsection*{Cyclotomic Extensions}

\begin{definition}
    For any \(n \in \bN\), \(\mu_n\) is the group of \(n\)th roots of unity,
    which are roots of \(x^n - 1\).
    A primitive \(n\)th root of unity is \(\zeta_n\),
    with \(\zeta_n^n = 1\) and \(\zeta_n^k \neq 1\) for all \(0 < k < n\).
\end{definition}
\begin{proposition}
    \(\mu_n \cong Z_n\);
    \(d \mid n\) if and only if \(\mu_d \subseteq \mu_n\);
    and \(\langle \zeta_n \rangle = \mu_n\).
\end{proposition}
\begin{proof}
    Obvious.
\end{proof}

\begin{proposition}
    Let \(F\) be a field of characteristic \(p > 0\).
    There does not exist primitive \(np\)th root of unity \(\zeta \in F\)
    for all \(n \in \bN\).
\end{proposition}
\begin{proof}
    \(\zeta\) is an \(np\)th root of unity,
    so \(\zeta^{np} - 1 = 0\).
    But in characteristic \(p\) we have
    \({(\zeta^n)}^p - 1^p = {(\zeta^n - 1)}^p = 0\),
    which allows us to conclude that \(\zeta\) is an \(n\)th root of unity.
    In that case, \(\zeta\) clearly cannot be a primitive \(np\)th root of unity.
\end{proof}

\begin{definition}
    The \(n\)th cyclotomic polynomial is
    \(\Phi_n(x) = \prod_{\text{primitive}\;\zeta\in\mu_n} (x-\zeta)\).
\end{definition}
\begin{lemma}\label{lem:cyclotomic-recursive}
    The cyclotomic polynomials can be recursively computed from
    \[\Phi_n(x) = \frac{x^n - 1}{\prod_{d \mid n,\, d < n} \Phi_d(x)}\]
\end{lemma}
\begin{proof}
    From the definition of coprimality, we know that
    \[x^n - 1 = \prod_{d \mid n} \Phi_d(x)\]
    so reorganizing the terms we have
    \[\Phi_n(x) = \frac{x^n - 1}{\prod_{d \mid n,\, d < n} \Phi_d(x)}\]
\end{proof}
\begin{proposition}
    Suppose \(\Phi_n(x) \in \bQ[x]\).
    Then \(\Phi_n(x)  = \prod_{1 \leq a < n,\; \gcd(a,n) = 1} (x-\zeta_n^a)\),
    and \(\deg \Phi_n(x) = \varphi(n)\),
    where \(\varphi\) is the Euler totient function.
\end{proposition}
\begin{proof}
    By Corollary~\ref{cor:order-cyclic-group},
    for \(\zeta_n^a\) to be a cyclic group, \(a\) must be coprime to \(n\).
    This gives us the product.
    Then, by definition of the totient function, there are \(\varphi(n)\) numbers coprime to \(n\).
\end{proof}
\begin{proposition}
    Consider the cyclotomic polynomial \(\Phi_n(x) \in \bQ[x]\).
    Then \(\Phi_n(x) \in \bZ[x]\), and is monic.
\end{proposition}
\begin{proof}
    Trivially, \(\Phi_1(x) = x - 1 \in \bZ[x]\) and is monic.
    We then proceed by strong induction,
    assuming that \(\Phi_k(x) \in \bZ[x]\) is monic for all \(k < n\).
    Since we know that \(\Phi_n(x)\) is the quotient of two monic polynomials
    (Lemma~\ref{lem:cyclotomic-recursive}),
    \(\Phi_n(x)\) must be monic.

    Moreover, the quotient of two polynomials in \(\bZ[x]\) where the divisor is monic,
    means the quotient and the remainder are both in \(\bZ[x]\).
    This can be seen simply from the Euclidean algorithm (Proposition~\ref{prop:field-polynomial-long-div}),
    where at every step, to determine the coefficient,
    we simply divide by the leading coefficient of the divisor, which is 1.
\end{proof}

\begin{lemma}
    For each \(\sigma \in \Gal(F(\mu_n)/F)\),
    there exists \(k_\sigma \in \bZ\) such that \(\gcd(k_\sigma,n) = 1\)
    and \(\sigma(\zeta) = \zeta^{k_\sigma}\) for all \(\zeta \in \mu_n\).
\end{lemma}
\begin{proof}
    Let \(\mu_n = \aqty{\zeta}\).
    We immediately have \(\zeta^n = 1\) and \(\zeta^k \neq 1\) for \(0 < k < n\);
    we also have \({\sigma(\zeta)}^n = 1\) and \({\sigma(\zeta)}^k\) for \(0 < k < n\),
    hence \(\sigma(\zeta)\) is also a primitive \(n\)th root of unity.
    This immediately tells us that \(\sigma(\zeta) = \zeta^{k_\sigma}\) for some \(k_\sigma\).
    As \(\zeta\) generates \(\mu_n\), this holds for all elements of \(\mu_n\).
\end{proof}
\begin{theorem}\label{thm:cyclotomic-injective}
    The mapping from \(\Gal(F(\mu_n)/F) \to {(\bZ/n\bZ)}^\times\)
    via \(\sigma \mapsto k_\sigma \pmod{n}\)
    where \(\sigma(\zeta) = \zeta^{k_\sigma}\) for all \(\zeta \in \mu_n\),
    is an injective group homomorphism.
\end{theorem}
\begin{proof}
    First off, \(\sigma \mapsto k_\sigma\) is indeed injective,
    because if \(k_\sigma = k_\tau\), then \(\sigma(\zeta) = \zeta^{k_\sigma} = \zeta^{k_\tau} = \tau(\zeta)\).

    Now, we check that it is indeed a group homomorphism.
    % The identity indeed gets mapped \(\id \mapsto 1\).
    \begin{equation*}
        \sigma\tau(\zeta) = \sigma(\zeta^{k_\tau}) = {(\zeta^{k_\tau})}^{k_\sigma} = \zeta^{k_\tau k_\sigma}
    \end{equation*}
    so \(k_{\sigma\tau} = k_\tau k_\sigma\) as desired.
\end{proof}
\begin{theorem}\label{thm:cyclotomic-isomorphism}
    The mapping from \(\Gal(\bQ(\mu_n)/\bQ) \to {(\bZ/n\bZ)}^\times\) is an isomorphism.
\end{theorem}
\begin{proof}
    From the \hyperref[thm:cyclotomic-injective]{theorem above},
    it suffices to prove that this map is surjective,
    which we can show with the \hyperref[thm:pigeonhole]{pigeonhole principle},
    by showing that \(\abs{\Gal(\bQ(\mu_n)/\bQ)} \geq \varphi(n)\).

    Consider \(f(x) = \min_\bQ(\zeta)\) for \(\mu_n = \aqty{\zeta}\).
    Then \(f \mid x^n - 1\), so there exists some \(h \in \bQ[x]\) such that \(x^n - 1 = f(x)h(x)\).
    Using the same argument as above,
    since \(f\) is monic by defintion and \(h\) is monic as a consequence,
    both \(f,h \in \bZ[x]\).

    We now claim that if a prime \(p \nmid n\), then \(\zeta^p\) too is a root of \(f\).
    Notice that \(\zeta^p\) is also a primitive \(n\)th root of unity;
    moreover, as \(\mu_n = \aqty{\zeta^p}\),
    there exists some \(k\) such that \(\zeta = {(\zeta^p)}^k = \zeta^{kp}\).
    Hence there exists \(\sigma \in \Gal(\bQ(\mu_n)/\bQ)\) such that \(\sigma(\zeta) = \zeta^p\)
    Thus \(\zeta^p\) is also a root of \(f\), as \(\sigma\) must permute the roots of a minimal polynomial.
    Repeating this argument tells us that \(\zeta^{kp}\) for all \(k\) are roots of \(f\),
    so there are at least \(\varphi(n)\) roots of \(f\).
    This exactly says that \(\abs{\Gal(\bQ(\mu_n)/\bQ)} = [\bQ(\zeta):\bQ] \geq \varphi(n)\).
\end{proof}
\begin{corollary}
    The cyclotomic polynomial \(\Phi_n(x)\) is irreducible over \(\bQ\).
\end{corollary}
\begin{proof}
    The \hyperref[thm:cyclotomic-isomorphism]{theorem above} also tells us that
    \(\Phi_n(x)\) is the minimal polynomial \(\min_\bQ(\zeta_n)\).
\end{proof}
