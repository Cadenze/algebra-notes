\section{Basic Definitions}

\begin{definition}
    A category \(\mca{C}\) is a collection of objects \(\Ob(\mca{C})\)
    alongisde a collection of morphisms or arrows \(\Mor(\mca{C})\) or \(\Hom(\mca{C})\),
    where
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item given a morphism \(f \in \Mor(\mca{C})\),
            there exists a source and a target object \(X,Y \in \Mor(\mca{C})\)
            such that the morphism maps from the source to the target \(\func{f}{X}{Y}\);
            % a morphism is a mapping from a source object to a target object;
            we define the collection of morphisms from \(X\) to \(Y\)
            as \(\Hom(X,Y) = \{f\in\Mor(\mca{C}) \mid \func{f}{X}{Y}\}\);
        \item composition of morphisms is \(\Hom(X,Y) \times \Hom(Y,Z) \to \Hom(X,Z)\),
            where we send \((f,g) \mapsto g \circ f = gf\);
        \item composition of morphisms is associative, as in \((f \circ g) \circ h = f \circ (g \circ h)\); and
        \item Given any object \(X \in \Ob(\mca{C})\),
            there exists an identity morphism \(\id_X \in \Hom(X,X)\)
            such that \(\id_X \circ g = g,\; f \circ \id_X = f\).
    \end{enumerate}
\end{definition}
\begin{remark}
    \(\Mor\) and \(\Hom\) are often interchangeable in this context.
    % since in algebra, the morphisms that we deal with are usually homomorphisms.
    We shall stick to the convention that the morphisms of a category is \(\Mor\),
    while the morphisms between objects are \(\Hom\).
\end{remark}

\subsection*{Morphisms}

\begin{definition}
    A morphism \(\func{f}{X}{Y}\) is an isomorphism or invertible
    if there exists an inverse morphism \(\func{f^{-1}}{Y}{X}\)
    such that \(f \circ f^{-1} = \id_Y\) and \(f^{-1} \circ f = \id_X\).
\end{definition}
\begin{lemma}
    Inverse morphisms, if they exist, are unique.
\end{lemma}
\begin{proof}
    Suppose \(f\) and two inverses \(g,g'\).
    Then
    \begin{equation*}
        g = g \circ \id = g \circ f \circ g' = \id \circ g' = g'
    \end{equation*}
\end{proof}

\begin{definition}
    A morphism \(\func{f}{X}{Y}\) is monic if it is left-cancellable;
    i.e.\ given any \(\func{g,g'}{Z}{X}\), \(f \circ g = f \circ g'\) implies \(g = g'\).
\end{definition}
\begin{definition}
    A morphism \(\func{f}{X}{Y}\) is epi if it is right-cancellable;
    i.e.\ given any \(\func{g,g'}{Y}{Z}\), \(g \circ f = g' \circ f\) implies \(g = g'\).
\end{definition}

\begin{definition}
    A morphism \(\func{f}{X}{X}\) is idempotent if \(f^2 = f\).
\end{definition}

\begin{definition}
    \(T \in \Ob(\mca{C})\) is a terminal object
    if for every object \(X \in \Ob(\mca{C})\),
    there exists a unique morphism \(X \to T\).
\end{definition}
\begin{definition}
    \(I \in \Ob(\mca{C})\) is an initial object
    if for every object \(X \in \Ob(\mca{C})\),
    there exists a unique morphism \(I \to X\).
\end{definition}
\begin{lemma}
    In a category \(\mca{C}\),
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item if there are two terminal objects \(T,T'\), there exists a unique isomorphism \(T \to T'\); and
        \item if there are two initial objects \(I,I'\), there exists a unique isomorphism \(I \to I'\).
    \end{enumerate}
\end{lemma}
\begin{proof}
    There exists a unique morphism \(T \to T'\), and a unique morphism \(T' \to T\).
    But this composition \(T \to T' \to T\) is the unique morphism \(T \to T\), which must be the identity.
    Repeat this argument for \(T' \to T \to T'\).

    The same argument holds for initial objects, because it is dual to terminal objects.
\end{proof}

\begin{definition}
    \(0 \in \Ob(\mca{C})\) is a null object if it is both initial and terminal.
\end{definition}

\subsection*{Functors}

\begin{definition}
    Suppose \(\mca{C},\mca{D}\) are categories.
    A functor is a map \(\func{T}{\mca{C}}{\mca{D}}\) between categories
    such that:
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item given any object \(X \in \Ob(\mca{C})\), we get an object \(TX \in \Ob(\mca{D})\);
        \item given any morphism \(f \in \Hom(X,Y) \subseteq \Mor(\mca{C})\),
            we get a morphism \(Tf \in \Hom(TX,TY) \subseteq \Mor(\mca{D})\); and
        \item the functor respects composition \(T(fg) = Tf \circ Tg\) and identity \(T\id_X = \id_{TX}\).
    \end{enumerate}
\end{definition}

\begin{definition}
    A functor \(\func{T}{\mca{C}}{\mca{D}}\) is an isomorphism of categories
    if it is a bijection of both objects and morphisms.
    Equivalently, \(T\) is an isomorphism
    if there exists an inverse functor \(\func{T^{-1}}{\mca{D}}{\mca{C}}\)
    such that \(T \circ T^{-1} = \id_\mca{D}\) and \(T^{-1} \circ T = \id_\mca{C}\) are both identity functors.
\end{definition}
\begin{definition}
    A functor \(\func{T}{\mca{C}}{\mca{D}}\) is full
    if given any two objects \(X,Y \in \Ob(\mca{C})\) and any morphism \(g \in \Hom(TX,TY)\),
    there exists some \(f \in \Hom(X,Y)\) such that \(g = Tf\).
    Equivalently, given any two objects \(X,Y \in \Ob(\mca{C})\),
    \(T\) is full if the function on morphisms \(\func{T}{\Hom(X,Y)}{\Hom(TX,TY)}\) is surjective.
    Compositions of full functors are full.
\end{definition}
\begin{definition}
    A functor \(\func{T}{\mca{C}}{\mca{D}}\) is faithful
    if given any two objects \(X,Y \in \Ob(\mca{C})\) and any two morphisms \(f,g \in \Hom(X,Y)\),
    then \(Tf = Tg\) implies \(f = g\).
    Equivalently, given any two objects \(X,Y \in \Ob(\mca{C})\),
    \(T\) is faithful if the function on morphisms \(\func{T}{\Hom(X,Y)}{\Hom(TX,TY)}\) is injective.
    Compositions of faithful functors are faithful.
\end{definition}
\begin{remark}
    A fully faithful functor is `locally bijective' when applied to any single hom-set,
    but need not be an isomorphism, because it does not have to be bijective `globally' for all objects/morphisms.
\end{remark}

\subsection*{Natural Transformations}

\begin{definition}
    Suppose \(\func{S,T}{\mca{C}}{\mca{D}}\) are two functors.
    A natural transformation \(\func{\tau}{S}{T}\)
    induces an arrow \(\tau_X \in \Hom(SX,TX)\) for every object \(X \in \Ob(\mca{C})\).
    It is a `morphism of functors' in the sense that
    if \(X,Y \in \Ob(\mca{C})\) and \(f \in \Hom(X,Y)\),
    \begin{equation*}
        \begin{tikzcd}
            X \arrow{d}{f} & SX \arrow{d}{Sf} \arrow{r}{\tau_X} & TX \arrow{d}{Tf} \\
            Y & SY \arrow{r}{\tau_Y} & TY
        \end{tikzcd}
    \end{equation*}
    is commutative.
\end{definition}
\begin{definition}
    A natural transformation \(\func{\tau}{S}{T}\) is a natural isomorphism
    if every \(\tau_X\) (for all objects \(X\)) is an invertible arrow.
    In that case, \(\func{\tau^{-1}}{T}{S}\) formed by \({(\tau_X)}^{-1}\) is also a natural isomorphism.
\end{definition}


\section{Constructions}

\subsection*{Duality}

\begin{definition}
    Given any statement in category theory,
    we obtain the dual of that statement
    if we reverse all the morphisms and compositions.
\end{definition}
\begin{theorem}[Principle of Duality]
    If any statement in category theory follows from the axioms,
    then the dual of that statement also follows from the axioms.
\end{theorem}
\begin{proof}
    The dual of the axioms are true, and logic does not get altered when taking the dual of any statement.
\end{proof}
\begin{remark}
    Note that when taking the dual of any statement, functors do not get reversed.
\end{remark}

\begin{definition}
    Consider a category \(\mca{C}\).
    The opposite category is \(\mca{C}^\text{op}\)
    consisting of the same objects, but all morphisms reversed.
\end{definition}
\begin{remark}
    There is a functor \(\textbf{Cat} \to \textbf{Cat}\)
    that takes categories \(\mca{C} \mapsto \mca{C}^\text{op}\)
    and any functor between categories \(T \mapsto T^\text{op}\).
\end{remark}

\begin{definition}
    Suppose \(\func{T}{\mca{C}^\text{op}}{\mca{D}}\) is a (covariant) functor.
    Then we can write this as a functor \(\func{\bar{T}}{\mca{C}}{\mca{D}}\)
    where the morphisms are reversed, i.e.\ \(\bar{T}(fg) = (\bar{T}g)(\bar{T}f)\).
\end{definition}
\begin{remark}
    To distinguish between between contravariant functors and the normal ones,
    we call normal functors covariant.
\end{remark}

\subsection*{Universality}

\begin{definition}
    Suppose \(\func{S}{\mca{D}}{\mca{C}}\) is a functor, and \(X \in \Ob(\mca{C})\).
    A universal arrow from the object \(X\) to the functor \(S\) is a tuple \((Y,f)\),
    where \(Y \in \Ob(\mca{D})\) is an object and \(\func{f}{X}{SY}\) is a morphism,
    such that given any tuple of objects and morphisms \((Z,\phi) \in \Ob(\mca{D})\times\Hom(X,SZ)\),
    there exists a unique arrow \(\func{\bar{\phi}}{Y}{Z}\) such that \(S\bar{\phi} \circ f = \phi\).

    This can be represented by the following commutative diagram:
    \begin{equation*}
        \begin{tikzcd}
            X \arrow{r}{f} \arrow{dr}[swap]{\phi} & SY \arrow[dashrightarrow]{d}{S\bar{\phi}} &
            Y \arrow[dashrightarrow]{d}{\bar{\phi}} \\
            & SZ & Z
        \end{tikzcd}
    \end{equation*}
\end{definition}

\begin{definition}
    A special case is when the functor is the identity.
    Suppose \(X\) is an object (perhaps in \(\textbf{Set}\)),
    and consider an equivalence relation \(\sim\).
    The universal arrow \(\func{\pi}{X}{X/{\sim}}\) sending each element into its equivalence class
    has the universal property given by
    \begin{equation*}
        \begin{tikzcd}
            X \arrow{r}{\pi} \arrow{rd}[swap]{\phi} &
            X/{\sim} \arrow[dashrightarrow]{d}{\bar{\phi}} \\
            & Y
        \end{tikzcd}
    \end{equation*}
    for any object \(Y\) and any morphism \(\phi\).
    Coincidentally, this is the commutative diagram on the cover of this set of notes.
\end{definition}

\begin{definition}
    Dually, suppose \(\func{S}{\mca{D}}{\mca{C}}\), and \(X \in \Ob(\mca{C})\).
    A universal arrow from the functor \(S\) to the object \(X\) is a tuple \((Y,f)\),
    where \(Y \in \Ob(\mca{D})\) is an object and \(\func{f}{SY}{X}\) is a morphism,
    such that given any tuple of objects and morphisms \((Z,\phi) \in \Ob(\mca{D})\times\Hom(SZ,X)\),
    there exists a unique arrow \(\func{\bar{\phi}}{Y}{Z}\) such that \(f \circ S\bar{\phi} = \phi\).

    This can be represented by the following commutative diagram:
    \begin{equation*}
        \begin{tikzcd}
            Y \arrow[dashrightarrow]{d}{\bar{\phi}} &
            SY \arrow{r}{f} \arrow[dashrightarrow]{d}{S\bar{\phi}} & X \\
            Z & SZ \arrow{ru}[swap]{\phi}
        \end{tikzcd}
    \end{equation*}
\end{definition}

\begin{definition}
    Suppose \(\mca{D}\) is a category with small hom-sets (as in they are sets in the set-theoretic sense).
    A representation of a functor \(\func{K}{\mca{D}}{\textbf{Set}}\)
    is a tuple \((R,\psi)\), where \(R \in \Ob(\mca{D})\) is an object called the representing object,
    and \(\func{\psi}{\Hom(R,\cdot)}{K}\) is a natural isomorphism.
    \(K\) is a representable functor if such a representation exists.
\end{definition}
\begin{lemma}[Yoneda Lemma]\label{lem:yoneda}
    Suppose \(\mca{D}\) is a category with small hom-sets.
    If \(\func{K}{\mca{D}}{\textbf{Set}}\) is a functor, and \(R \in \Ob(\mca{D})\),
    then there is a bijection
    \begin{equation*}
        \varfunc{Y}{\Hom_{\mca{D}\to\textbf{Set}}(\Hom_\mca{D}(R,\cdot),K)}{KR}{(\func{\alpha}{\Hom(R,\cdot)}{K})}{\alpha_R\id_R}
    \end{equation*}
    between the natural transformations \(\alpha\) and the image of the identity under \(\alpha\),
    which is an element of the set \(KR\).
\end{lemma}
\begin{proof}
    Consider the following commutative diagram:
    \begin{equation*}
        \begin{tikzcd}
            \Hom_\mca{D}(R,R) \arrow{d}[swap]{f_\ast = \Hom(R,f)} \arrow{r}{\alpha_R} & KR \arrow{d}{Kf} & R \arrow{d}{f} \\
            \Hom_\mca{D}(R,Z) \arrow{r}{\alpha_Z} & KZ & Z
        \end{tikzcd}
    \end{equation*}
    
    We construct an inverse.
    Given any element of \(x \in KR\), we need to show that \(\alpha_R \id_R = x\) completely determines \(\alpha\).
    But by the commutative diagram, given any arrow \(\func{f}{R}{Z}\),
    \begin{equation*}
        \alpha_Z f = \alpha_Z(f \circ \id) = \alpha_Z f_\ast(\id_R) = (Kf \circ \alpha_R)(\id_R) = Kf(x)
    \end{equation*}
    is wholly determined by the value of \(x\) as desired.
\end{proof}
\begin{corollary}[Yoneda Lemma]\label{cor:yoneda}
    Suppose \(\mca{D}\) is a category with small hom-sets.
    There is a bijection between the natural transformations
    \(\Hom_\mca{D}(R,\cdot) \to \Hom_\mca{D}(S,\cdot)\)
    and the morphisms \(\Hom_\mca{D}(S,R)\).
\end{corollary}
\begin{proof}
    Conclude with \(K = \Hom_\mca{D}(S,\cdot)\).
\end{proof}

\subsection*{Products and Limits}

% \subsection*{Product}

\begin{definition}
    Suppose \(X,Y\) are two objects.
    We call \(X \times Y\) the product of objects if
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item there exists projections \(\func{\pi_1}{X \times Y}{X}\)
            and \(\func{\pi_2}{X \times Y}{Y}\); and
        \item for any object \(Z\),
            if \(\func{\phi_1}{Z}{X}\) and \(\func{\phi_2}{Z}{Y}\) are homomorphisms,
            then there exists a unique \(\func{\bar{\phi}}{Z}{X \times Y}\)
            such that \(\phi_i = \pi_i\circ\bar{\phi}\) for \(i = 1,2\).
    \end{enumerate}

    This can be represented by the following commutative diagram:
    \begin{center}
        \begin{tikzcd}
            & Z \arrow{ld}[swap]{\phi_1} \arrow{rd}{\phi_2}
            \arrow[dashrightarrow]{d}{\exists!\bar{\phi}} \\
            X & X \times Y \arrow{l}{\pi_1} \arrow{r}[swap]{\pi_2} & Y
        \end{tikzcd}
    \end{center}
\end{definition}

\begin{definition}
    Suppose \({\{X_i\}}_{i \in I}\) is a family of objects.
    We call \(\prod_{i \in I} X_i\) the (infinite) product of objects if
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item there exists projections \(\func{\pi_i}{\prod_{i \in I} X_i}{X_i}\); and
        \item for any object \(Z\),
            if \(\func{\phi_i}{Z}{X_i}\) are homomorphisms,
            then there exists a unique \(\func{\bar{\phi}}{Z}{\prod_{i \in I} X_i}\)
            such that \(\phi_i = \pi_i\circ\bar{\phi}\).
    \end{enumerate}

    This can be represented by the logical statement
    \begin{equation*}
        \forall Z \in \mca{C},\;
        \forall i \in I,\;
        \forall \phi_i \in \Hom(X,V_i),\;
        \exists! \phi \in \Hom\pqty{Z,\prod_{i \in I} X_i},\;
        \phi_i = \pi_i\circ\bar{\phi}
    \end{equation*}
    and the following commutative diagram:
    \begin{center}
        \begin{tikzcd}
            & Z \arrow{ld}[swap]{\phi_i}
            \arrow[dashrightarrow]{d}{\exists!\bar{\phi}} \\
            X_i & \prod_{i \in I} X_i \arrow{l}{\pi_i}
        \end{tikzcd}
    \end{center}
\end{definition}

\subsection*{Coproduct}

\begin{remark}
    The coproduct is the dual to the product.
\end{remark}

\begin{definition}
    Suppose \(X,Y\) are two objects.
    We call \(X \oplus Y\) the coproduct of objects if
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item there exists inclusions \(\func{\iota_1}{X}{X \oplus Y}\)
            and \(\func{\iota_2}{Y}{X \oplus Y}\); and
        \item for any object \(Z\),
            if \(\func{\phi_1}{X}{Z}\) and \(\func{\phi_2}{Y}{Z}\) are homomorphisms,
            then there exists a unique \(\func{\bar{\phi}}{X \oplus Y}{Z}\)
            such that \(\phi_i = \bar{\phi}\circ\iota_i\) for \(i = 1,2\).
    \end{enumerate}

    This can be represented by the following commutative diagram:
    \begin{center}
        \begin{tikzcd}
            & Z \\
            X \arrow{r}[swap]{\iota_1} \arrow{ru}{\phi_1} &
            X \oplus Y \arrow[dashrightarrow]{u}[swap]{\exists!\bar{\phi}} &
            Y \arrow{l}{\iota_2} \arrow{lu}[swap]{\phi_2}
        \end{tikzcd}
    \end{center}
\end{definition}

\begin{definition}
    Suppose \({\{X_i\}}_{i \in I}\) is a family of objects.
    We call \(\bigoplus_{i \in I} X_i\) the (infinite) coproduct of objects if
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item there exists inclusions \(\func{\iota_i}{X}{\bigoplus_{i \in I} X_i}\); and
        \item for any object \(Z\),
            if \(\func{\phi_i}{X_i}{Z}\) are homomorphisms,
            then there exists a unique \(\func{\bar{\phi}}{\bigoplus_{i \in I} X_i}{Z}\)
            such that \(\phi_i = \bar{\phi}\circ\iota_i\).
    \end{enumerate}

    This can be represented by the logical statement
    \begin{equation*}
        \forall Z \in \mca{C},\;
        \forall i \in I,\;
        \forall \phi_i \in \Hom(X_i,Z),\;
        \exists! \phi \in \Hom\pqty{\bigoplus_{i \in I} X_i, Z},\;
        \phi_i = \bar{\phi}\circ\iota_i
    \end{equation*}
    and the following commutative diagram:
    \begin{center}
        \begin{tikzcd}
            & Z \\
            X_i \arrow{r}[swap]{\iota_i} \arrow{ru}{\phi_i} &
            \bigoplus_{i \in I} X_i \arrow[dashrightarrow]{u}[swap]{\exists!\bar{\phi}}
        \end{tikzcd}
    \end{center}
\end{definition}

\subsection*{Pullback}

\begin{definition}
    Suppose \(W,X,Y\) are three objects.
    We call \(X \times_W Y\) the pullback or fibre product with respect to \(W\) if
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item there exists projections \(\func{\pi_X}{X}{W}\) and \(\func{\pi_Y}{Y}{W}\);
        \item there exists projections \(\func{\pi_1}{X \times_W Y}{X}\)
            and \(\func{\pi_2}{X \times_W Y}{Y}\); and
        \item for any object \(Z\),
            if \(\func{\phi_1}{Z}{X}\) and \(\func{\phi_2}{Z}{Y}\) are homomorphisms,
            then there exists a unique \(\func{\bar{\phi}}{Z}{X \times_W Y}\)
            such that the following diagram commutes.
    \end{enumerate}
    \begin{center}
        \begin{tikzcd}
            & Z \arrow{ld}[swap]{\phi_1} \arrow{rd}{\phi_2}
            \arrow[dashrightarrow]{d}{\exists!\bar{\phi}} \\
            X \arrow{rd}[swap]{\pi_X} &
            X \times Y \arrow{l}{\pi_1} \arrow{r}[swap]{\pi_2} &
            Y \arrow{ld}{\pi_Y} \\
            & W
        \end{tikzcd}
    \end{center}
\end{definition}

\subsection*{Pushout}

\begin{remark}
    The pushout is the dual to the pullback.
\end{remark}

\begin{definition}
    Suppose \(W,X,Y\) are three objects.
    We call \(X +_W Y\) the pushout or fibre coproduct with respect to \(W\) if
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item there exists inclusions \(\func{\iota_X}{W}{X}\) and \(\func{\iota_Y}{W}{Y}\);
        \item there exists inclusions \(\func{\iota_1}{X}{X +_W Y}\)
            and \(\func{\iota_2}{Y}{X +_W Y}\); and
        \item for any object \(Z\),
            if \(\func{\phi_1}{X}{Z}\) and \(\func{\phi_2}{Z}{Y}\) are homomorphisms,
            then there exists a unique \(\func{\bar{\phi}}{X +_W Y}{Z}\)
            such that the following diagram commutes.
    \end{enumerate}
    \begin{center}
        \begin{tikzcd}
            & Z \\
            X \arrow{r}[swap]{\iota_1} \arrow{ru}{\phi_1} &
            X +_W Y \arrow[dashrightarrow]{u}[swap]{\exists!\bar{\phi}} &
            Y \arrow{l}{\iota_2} \arrow{lu}[swap]{\phi_2} \\
            & W \arrow{lu}{\iota_X} \arrow{ru}[swap]{\iota_Y}
        \end{tikzcd}
    \end{center}
\end{definition}


\section{Limits}

\subsection*{Inverse Limit}

\begin{definition}
    Suppose \({\{X_i\}}_{i \in I}\) is a family of objects,
    and the index set \(I\) is ordered with the relation \(\leq\).
    An inverse system over \(I\) is when \(i \leq j\),
    let \(\func{f_{ij}}{X_j}{X_i}\) be a homomorphism such that
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item \(f_{ii} = \id_{X_i}\); and
        \item \(f_{ik} = f_{ij} \circ f_{jk}\) for all \(i \leq j \leq k\).
    \end{enumerate}

    This can be represented by the following commutative diagram:
    \begin{center}
        \begin{tikzcd}
            X_i & X_j \arrow{l}{f_{ij}} &
            X_k \arrow{l}{f_{jk}} \arrow[bend right]{ll}[swap]{f_{ik}}
        \end{tikzcd}
    \end{center}
\end{definition}

\begin{definition}
    Suppose \({(X_i,f_{ij})}_{i,j \in I}\) is an inverse system.
    The inverse limit of this system is an object \(\invlim X_i\)
    that satisfies the following universal property:
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item there exists projections \(\func{\pi_i}{\invlim X_i}{X_i}\)
            such that for all \(i \leq j\), \(\pi_i = f_{ij}\circ\pi_j\); and
        \item for any object \(Z\), if \(\func{\phi_i}{Z}{X_i}\) are homomorphisms
            such that for all \(i \leq j\), \(\phi_i = f_{ij}\circ\phi_j\),
            then there exists a unique \(\func{\bar{\phi}}{Z}{\invlim X_i}\)
            such that \(\phi_i = \pi_i\circ\bar{\phi}\).
    \end{enumerate}

    This can be represented by the following commutative diagram:
    \begin{center}
        \begin{tikzcd}
            & Z \arrow[bend right]{ddl}[swap]{\phi_i} \arrow[bend left]{ddr}{\phi_j}
            \arrow[dashrightarrow]{d}{\exists!\bar{\phi}} \\
            & \invlim X_i \arrow{dl}{\pi_i} \arrow{dr}[swap]{\pi_j} \\
            X_i && X_j \arrow{ll}{f_{ij}}
        \end{tikzcd}
    \end{center}
\end{definition}

\subsection*{Direct Limit}

\begin{remark}
    The direct system and direct limit
    is dual to the inverse system and inverse limit.
\end{remark}

\begin{definition}
    Suppose \({\{X_i\}}_{i \in I}\) is a family of objects,
    and the index set \(I\) is ordered with the relation \(\leq\).
    A direct system over \(I\) is when \(i \leq j\),
    let \(\func{f_{ij}}{X_i}{X_j}\) be a homomorphism such that
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item \(f_{ii} = \id_{X_i}\); and
        \item \(f_{ik} = f_{jk} \circ f_{ij}\) for all \(i \leq j \leq k\).
    \end{enumerate}

    This can be represented by the following commutative diagram:
    \begin{center}
        \begin{tikzcd}
            X_i \arrow{r}[swap]{f_{ij}} \arrow[bend left]{rr}{f_{ik}} &
            X_j \arrow{r}[swap]{f_{jk}} & X_k
        \end{tikzcd}
    \end{center}
\end{definition}

\begin{definition}
    Suppose \({(X_i,f_{ij})}_{i,j \in I}\) is a direct system.
    The direct limit of this system is an object \(\dirlim X_i\)
    that satisfies the following universal property:
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item there exists inclusions \(\func{\iota_i}{X_i}{\dirlim X_i}\)
            such that for all \(i \leq j\), \(\iota_i = \iota_j \circ f_{ij}\); and
        \item for any object \(Z\), if \(\func{\phi_i}{X_i}{Z}\) are homomorphisms
            such that for all \(i \leq j\), \(\phi_i = \phi_j \circ f_{ij}\),
            then there exists a unique \(\func{\bar{\phi}}{\dirlim X_i}{Z}\)
            such that \(\phi_i = \bar{\phi}\circ\iota_i\).
    \end{enumerate}

    This can be represented by the following commutative diagram:
    \begin{center}
        \begin{tikzcd}
            & Z \\
            & \dirlim X_i \arrow[dashrightarrow]{u}[swap]{\exists!\bar{\phi}} \\
            X_i \arrow{rr}[swap]{f_{ij}} \arrow{ur}[swap]{\iota_i} \arrow[bend left]{uur}{\phi_i} &&
            X_j \arrow{ul}{\iota_j} \arrow[bend right]{uul}[swap]{\phi_j}
        \end{tikzcd}
    \end{center}
\end{definition}
