\chapter{Commutative Algebra}\label{ch:comm-alg}

\begin{remark}
    We shall assume that all rings,
    unless otherwise noted, are commutative;
    that is, the multiplication \(ab = ba\).
\end{remark}


\section{Fractions}

\subsection*{Ring of Fractions}

\begin{definition}
    Suppose \(R\) is a ring, \(S \subseteq R\) is a multiplicative monoid.
    Then the ring of fractions is \(S^{-1}R = \{r/s : r \in R, s \in S\}/{\sim}\),
    where the equivalence relation is given by \(r_1/s_1 = r_2/s_2\)
    if there exists some nonzero \(s \in S\) such that \(s(r_1s_2 - r_2s_1) = 0\).
\end{definition}
\begin{proposition}
    There exists a ring homomorphism \(\vfunc{\iota}{R}{S^{-1}R}{n}{n/1}\).
\end{proposition}
\begin{proof}
    Observe that the ring homomorphism axioms are satisfied as
    \(\iota(0) = 0/1 = 0\),
    \(\iota(1) = 1/1 = 1\),
    \(\iota(r)\iota(s) = (r/1)(s/1) = rs/1 = \iota(rs)\),
    and \(\iota(r)+\iota(s) = (r/1)+(s/1) = (r+s)/1 = \iota(r+s)\).
\end{proof}
\begin{lemma}
    Suppose \(R\) is a ring, \(S \subseteq R\) is a multiplicative monoid.
    If \(s \in S\), then \(s\) is a unit in \(S^{-1}R\).
\end{lemma}
\begin{proof}
    \(s(1/s) = (s/1)(1/s) = 1\).
\end{proof}
\begin{lemma}
    Suppose \(R\) is a ring, \(S \subseteq R\) is a multiplicative monoid.
    \(r \in R\) is zero in \(S^{-1}R\) if and only if there exists some \(s\)
    such that \(sr = 0\) in \(R\).
\end{lemma}
\begin{proof}
    By definition, \(r = r/1 = 0/1 = 0\) if and only if \(s(r-0) = sr = 0\).
\end{proof}

\begin{theorem}[Universal Property of Ring of Fractions]
    Suppose \(R\) is a ring, \(S \subseteq R\) is a multiplicative monoid.
    The ring of fractions is \(S^{-1}R\), if:
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item there exists a homomorphism \(\func{\iota}{R}{S^{-1}R}\); and
        \item for any ring \(T\), if \(\func{\phi}{R}{T}\) is a homomorphism that sends \(s \in S\) to a unit,
            then there exists a unique \(\func{\bar{\phi}}{S^{-1}R}{T}\)
            such that \(\phi = \bar{\phi}\circ\iota\).
    \end{enumerate}
    
    This is represented by the following commutative diagram:
    \begin{equation*}
        \begin{tikzcd}
            R \arrow{r}{\phi} \arrow{d}{\iota} & T \\
            S^{-1}R \arrow[dashrightarrow]{ur}[swap]{\exists!\bar{\phi}}
        \end{tikzcd}
    \end{equation*}
\end{theorem}
\begin{proof}
    Suppose there exists \(\bar{\phi}\neq\bar{\phi}'\) that both fit the diagram.
    Then they disagree on some \(r/s\).
    But then we have
    \begin{equation*}
        s\bar{\phi}(r/s) = \bar{\phi}(r/1) = (\bar{\phi}\circ\iota)(r)
        = \phi(r) = (\bar{\phi}'\circ\iota)(r) = \bar{\phi}'(r/1) = s\bar{\phi}(r/s)
    \end{equation*}
    which is a contradiction.

    Now, we can simply define \(\bar{\phi}(r/s) = \phi(r){(\phi(s))}^{-1}\) as desired.
\end{proof}

\begin{proposition}
    Suppose \(R\) is a domain, and \(S = R - \{0\}\) is the set of nonzero elements in \(R\).
    Then \(S^{-1}R\) is the field of fractions.
\end{proposition}
\begin{proof}
    See existence of field of fractions in Theorem~\ref{thm:existence-field-of-fractions}.
    In particular, we have \(a/b = c/d\) when \(ad = bc\) is sufficient,
    since \(s(ad - bc) = 0\) is equivalent to \(ad - bc = 0\) since we are in an integral domain.
\end{proof}

\begin{definition}
    Suppose \(R\) is a ring, \(\mfk{p} \subseteq R\) is a prime ideal,
    and \(S = R - \mfk{p}\).
    Then we call \(S^{-1}R = R_\mfk{p}\) to be the localization of \(R\) at \(\mfk{p}\).
\end{definition}
\begin{proposition}
    Suppose \(R\) is a ring, and \(\mfk{p} \subseteq R\) is a prime ideal.
    \((R_\mfk{p},\mfk{p}R_\mfk{p})\) forms a local ring.
\end{proposition}
\begin{proof}
    Consider any \(r/s \in R_\mfk{p} - \mfk{p}R_\mfk{p}\).
    We claim that \(r \notin \mfk{p}\),
    since if it is, \(r/s = r(1/s) \in \mfk{p}R_\mfk{p}\) which is a contradiction.
    Hence \(r \in R - \mfk{p}\), and \(s/r \in R_\mfk{p}\).
    But we have just proven that every element in \(R_\mfk{p} - \mfk{p}R_\mfk{p}\) is a unit,
    so by Proposition~\ref{prop:local-ring-units} \((R_\mfk{p},\mfk{p}R_\mfk{p})\) is a local ring.
\end{proof}

\subsection*{Module of Fractions}

\begin{definition}
    Suppose \(S^{-1}R\) is a ring of fractions, and \(M\) be an \(R\)-module.
    Then \(S^{-1}M = \{m/s : m \in M, s \in S\}/{\sim}\) is an \(S^{-1}R\)-module,
    with the equivalence relation is given by \(m_1/s_1 = m_2/s_2\)
    if there exists some \(s \in S\) such that \(s(m_1s_2 - m_2s_1) = 0\).
\end{definition}
\begin{proposition}
    There exists a module homomorphism \(\vfunc{\iota}{M}{S^{-1}M}{m}{m/1}\),
    where \(\ker\iota = \{m : \exists s \in S, ms = 0\}\).
\end{proposition}
\begin{proof}
    Observe that the module homomorphism axioms are satisfied as
    \(\iota(0) = 0/1 = 0\) and
    \(r\iota(m)+\iota(n) = r(m/1)+(n/1) = (rm)/1+(n/1) = (rm+n)/1 = \iota(rm+n)\).
    Obviously, \(\iota(m) = m/1 = 0\) only if \(s(m1 - 0) = sm = 0\).
\end{proof}

\begin{proposition}
    Suppose \(R\) is a ring, and \(S^{-1}R\) is a ring of fractions.
    Then \(\vfunc{S^{-1}}{\textbf{Mod}_R}{\textbf{Mod}_{S^{-1}R}}{M}{S^{-1}M}\)
    is a functor.
\end{proposition}
\begin{proof}
    We already see that it sends modules to modules from our definition of module of fractions,
    so it suffices to see that it maps module homomorphisms.
    Suppose \(\func{\phi}{M}{N}\) is a module homomorphism.
    Then we can simply define \(\vfunc{\phi}{S^{-1}M}{S^{-1}N}{m/s}{\phi(m)/s}\).
\end{proof}
\begin{theorem}\label{thm:fraction-exact}
    \(S^{-1}\) is an exact functor.
\end{theorem}
\begin{proof}
    Consider the exact sequence \(M \overset{f}{\to} N \overset{g}{\to} P\).
    See that if \((g \circ f)(m) = 0\),
    then obviously \((g \circ f)(m/s) = (g \circ f)(m)/s = 0\).
    This gives image in kernel.

    Now consider \(n/s \in S^{-1}N\) such that \(g(n)/s = 0\).
    But then that implies there exists \(s'\) such that \(s'g(n) = 0\);
    i.e.\ \(s'n \in \ker g\).
    By exactness, there exists \(m \in M\) such that \(f(m) = s'n\),
    and we can get \(f(m/(ss')) = f(m)/ss' = s'n/ss' = n/s\).
    This gives kernel in image.
\end{proof}

\begin{proposition}\label{prop:fraction-properties}
    Suppose \(P\) is an \(R\)-module,
    and \(M,N \subseteq P\) are submodules.
    The following properties hold for modules of fractions:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(S^{-1}(M+N) = S^{-1}M + S^{-1}N \subseteq S^{-1}P\);
        \item \(S^{-1}(M \cap N) = S^{-1}M \cap S^{-1}N \subseteq S^{-1}P\); and
        \item \(S^{-1}(M/N) \cong S^{-1}M/S^{-1}N \subseteq S^{-1}P\).
    \end{enumerate}
\end{proposition}
\begin{proof}
    For (a), there is a natural isomorphism \((m+n)/s \mapsto m/s + n/s\),
    since the inverse map is \(m/s + n/s \mapsto (m+n)/s\) obviously.

    For (b), observe that if \(m \in M \cap N\),
    \(m/s \in S^{-1}M\) and \(m/s S^{-1}N\) as desired.
    On the other hand, consider \(m/s \in S^{-1}M\) and \(n/t \in S^{-1}N\),
    and \(m/s = n/t\).
    then there exists some \(u \in S\) such that \(u(tm-sn)\)

    For (c), consider the exact sequence
    \begin{equation*}
        \begin{tikzcd}
            0 \arrow{r} & N \arrow{r} & M \arrow{r} & M/N \arrow{r} & 0
        \end{tikzcd}
    \end{equation*}
    Since \(S^{-1}\) is exact by Theorem~\ref{thm:fraction-exact},
    we have the exact sequence
    \begin{equation*}
        \begin{tikzcd}
            0 \arrow{r} & S^{-1}N \arrow{r} & S^{-1}M \arrow{r} & S^{-1}(M/N) \arrow{r} & 0
        \end{tikzcd}
    \end{equation*}
    and Proposition~\ref{prop:short-exact-sequence}
    gives us that \(S^{-1}(M/N) \cong S^{-1}M/S^{-1}N\) as desired.
\end{proof}

\begin{theorem}\label{thm:fraction-isomorphism}
    There is an isomorphism \(S^{-1}R \otimes M \cong S^{-1}M\).
\end{theorem}
\begin{proof}
    We prove that \(S^{-1}M\) fits the universal property of \(S^{-1}R \otimes M\).
    Suppose we have the commutative diagram
    \begin{equation*}
        \begin{tikzcd}
            S^{-1}R \times M \arrow{r}{\mu} \arrow{rd}{\phi} &
            S^{-1}M \arrow[dashrightarrow]{d}{\exists!\bar{\phi}} \\
            & P
        \end{tikzcd}
    \end{equation*}
    with \(\mu(r/s,m) = (rm)/s\).
    Suppose there exists \(\bar{\phi}\neq\bar{\phi}'\) that both fit this diagram.
    Then we know they must disagree on some \(\bar{\phi}(m/s)\neq\bar{\phi}'(m/s)\).
    But observe that
    \begin{equation*}
        \bar{\phi}\pqty{\frac{m}{s}} = (\bar{\phi}\circ\mu)\pqty{\frac{1}{s},m}
        = \phi\pqty{\frac{1}{s},m} = (\bar{\phi}'\circ\mu)\pqty{\frac{1}{s},m}
        = \bar{\phi}'\pqty{\frac{m}{s}}
    \end{equation*}
    which is a contraction.
    This gives us uniqueness.
    Now, we can readily define \(\bar{\phi}(m/s) = \phi(1/s,m)\),
    and observe that \(\bar{\phi}(rm/s) = \phi(1/s,rm) = r\phi(1/s,m) = \phi(r/s,m)\)
    indeed by bilinearity.
    This gives us existence.
\end{proof}
\begin{corollary}
    \(S^{-1}R\) is a flat \(R\)-module.
\end{corollary}
\begin{proof}
    The \hyperref[thm:fraction-isomorphism]{theorem above}
    gives us that \(S^{-1} = S^{-1}R \otimes \cdot\) as functors.
    Since \(S^{-1}\) is exact (Theorem~\ref{thm:fraction-exact}),
    \(S^{-1}R \otimes \cdot\) is also exact.
\end{proof}

\begin{proposition}\label{prop:fraction-tensor}
    Suppose \(M,N\) are (appropriate) \(R\)-modules.
    The following properties hold for modules of fractions:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(S^{-1}(M \oplus N) \cong S^{-1}M \oplus S^{-1}N\); and
        \item \(S^{-1}(M \otimes N) \cong S^{-1}M \otimes S^{-1}N\).
    \end{enumerate}
\end{proposition}
\begin{proof}
    For (a), this is a special case of Proposition~\ref{prop:fraction-properties},
    where the sum \(M + N\) is direct if we consider \(M,N \subseteq M \oplus N\).

    For (b), we have
    \begin{equation*}
        S^{-1}M \otimes S^{-1}N \cong (S^{-1}R \otimes M) \otimes (S^{-1}R \otimes N)
        \cong S^{-1}R \otimes S^{-1}R \otimes (M \otimes N)
        \cong S^{-1}R \otimes (M \otimes N)
        \cong S^{-1}(M \otimes N)
    \end{equation*}
\end{proof}
\begin{proposition}\label{prop:localization-flat}
    \(M\) is a flat \(R\)-module implies \(S^{-1}M\) is a flat \(S^{-1}R\)-module.
\end{proposition}
\begin{proof}
    If \(M \otimes \cdot\) is an exact functor,
    then \(M \otimes (S^{-1}R \otimes \cdot) \cong S^{-1}M \otimes \cdot\) is an exact functor.
\end{proof}

\begin{definition}
    Suppose \(M\) is an \(R\)-module, and \(M_\mfk{p},M_\mfk{m}\) some localizations.
    We say a module property is local when
    \(M\) has that property if and only if \(M_\mfk{p}\) for all prime \(\mfk{p} \subseteq R\) has that property.
    We say a module property is max-local when
    \(M\) has that property if and only if \(M_\mfk{m}\) for all maximal \(\mfk{m} \subseteq R\) has that property.
\end{definition}
\begin{lemma}\label{lem:local-max-local}
    Suppose \(M,N\) are \(R\)-modules, and \(\func{\phi}{M}{N}\) is a homomorphism.
    Then the following properties are local and max-local:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(M = 0\);
        \item \(\phi\) injective; and
        \item \(\phi\) surjective.
    \end{enumerate}
\end{lemma}
\begin{proof}
    We will prove the three statements in similar fashion,
    by first proving the forward direction of local,
    and then the reverse direction of max-local.
    Finally, using the fact that \hyperref[thm:maximal-prime-radical]{maximal ideals are prime},
    prime localizations having a property immediately implies
    maximal localizations must also have said property.

    \medskip

    First off, it is obvious that \(M = 0\)
    implies \(M_\mfk{p} = 0\) for all prime \(\mfk{p} \subseteq R\).

    We will prove the reverse of max-locality via contraposition.
    Suppose \(M \neq 0\).
    Consider some nonzero element \(m \in M\)
    and its annihilator \(I = \Ann m \subsetneq (1) = R\).
    By Corollary~\ref{cor:proper-in-maximal},
    \(I \subseteq p\) for some maximal ideal \(p \subseteq R\).
    Then we claim that \(m \in M_p\) is nonzero,
    because if it is, then there is some element \(r \in R - p\) that kills \(m\),
    which contradicts that \(\Ann m \subseteq p\).
    This proves that there is some nonzero \(M_p\),
    and we have proven statement (a).

    \medskip

    Suppose \(0 \to M \overset{\phi}{\to} N\) is exact.
    Since \(S^{-1}\) is exact (Theorem~\ref{thm:fraction-exact}),
    \(0 \to M_\mfk{p} \overset{\phi}{\to} N_\mfk{p}\) is also exact.
    
    Suppose \(0 \to M_\mfk{m} \overset{\phi}{\to} N_\mfk{m}\) is exact for all maximal \(\mfk{m} \subseteq R\).
    Consider the exact sequence \(0 \to \ker\phi \to M \overset{\phi}{\to} N\).
    Since \(S^{-1}\) is exact (Theorem~\ref{thm:fraction-exact}),
    \(0 \to {(\ker\phi)}_\mfk{m} \to M_\mfk{m} \overset{\phi}{\to} N_\mfk{m}\) is also exact,
    which implies \({(\ker\phi)}_\mfk{m} = 0\) for all maximal \(\mfk{m} \subseteq R\),
    and by max-locality in (a), \(\ker\phi = 0\) as desired.
    We have proven statement (b).
    
    \medskip
    
    Suppose \(M \overset{\phi}{\to} N \to 0\) is exact.
    Since \(S^{-1}\) is exact (Theorem~\ref{thm:fraction-exact}),
    \(M_\mfk{p} \overset{\phi}{\to} N_\mfk{p} \to 0\) is also exact.
    
    Suppose \(M_\mfk{m} \overset{\phi}{\to} N_\mfk{m} \to 0\) is exact for all maximal \(\mfk{m} \subseteq R\).
    Consider the exact sequence \(M \overset{\phi}{\to} N \to \coker\phi \to 0\).
    Since \(S^{-1}\) is exact (Theorem~\ref{thm:fraction-exact}),
    \(M_\mfk{m} \overset{\phi}{\to} N_\mfk{m} \to {(\coker\phi)}_\mfk{m} \to 0\) is also exact,
    which implies \({(\coker\phi)}_\mfk{m} = 0\) for all maximal \(\mfk{m} \subseteq R\),
    and by max-locality in (a), \(\coker\phi = 0\) as desired.
    We have proven statement (c).
\end{proof}
\begin{theorem}
    Flatness is local and max-local.
\end{theorem}
\begin{proof}
    We prove in the same manner as the \hyperref[lem:local-max-local]{lemma above}.
    Suppose \(M\) is flat.
    Then \(M_\mfk{p}\) is flat for all prime \(\mfk{p}\) by Proposition~\ref{prop:localization-flat}.

    Using the fact that \hyperref[thm:maximal-prime-radical]{maximal ideals are prime},
    prime localizations being flat immediately implies
    maximal localizations must also be flat.

    Now suppose \(M_\mfk{m}\) is flat for all maximal \(\mfk{m} \subseteq R\).
    If \(N \to P\) is injective,
    then \(N_\mfk{m} \to P_\mfk{m}\) is injective by the \hyperref[lem:local-max-local]{lemma above},
    and by flatness, \(M_\mfk{m} \otimes N_\mfk{m} \to M_\mfk{m} \otimes P_\mfk{m}\) is injective.
    By Proposition~\ref{prop:fraction-tensor},
    this is equivalent to saying \({(M \otimes N)}_\mfk{m} \to {(M \otimes P)}_\mfk{m}\) is injective.
    Using the \hyperref[lem:local-max-local]{lemma above} again,
    \(M \otimes N \to M \otimes P\) is injective,
    which tells us that \(M\) is flat (Lemma~\ref{lem:flat-equivalence}).
\end{proof}

\subsection*{Extensions and Contractions}

\begin{lemma}\label{lem:fraction-extensions}
    Consider the ring homomorphism \(\func{\phi}{R}{S^{-1}R}\).
    If \(\mfk{a} \subseteq R\), its extension is \(\mfk{a}^e = \mfk{a} \otimes S^{-1}R = S^{-1}\mfk{a}\).
    Conversely, every ideal in \(S^{-1}R\) is an extension of some ideal.
\end{lemma}
\begin{proof}
    Suppose \(r \in \mfk{a}\).
    Then for any \(s \in S\), \((1/s)r = r/s \in \mfk{a}^e\),
    so we have \(S^{-1}\mfk{a} \subseteq \mfk{a}^e\).
    Now suppose \(\sum r_i/s_i \in \mfk{a}^e\).
    We can assume \(r_i \in \mfk{a}\),
    since we have \(\sum (t_i/s_i)r_i = \sum (t_i r_i)/s_i \in \mfk{a}^e\),
    and \(t_i r_i \in \mfk{a}R \subseteq \mfk{a}\).
    We write this with a common denominator \(\sum r_i/s_i = r/s\),
    and we can immediately conclude that \(r\) is a sum of \(r_i\) and \(s_i\)s,
    so \(r \in \mfk{a}\).

    Now consider some ideal \(\mfk{b} \in S^{-1}R\).
    By the same logic as above, it suffices to consider \(r/s \in \mfk{b}\),
    and we know that \(s(r/s) = r \in \mfk{b}\),
    so \(r \in \mfk{b}^c\) and \(r/s \in \mfk{b}^{ce}\).
    But we have \(\mfk{b} \subseteq \mfk{b}^{ce} \subseteq \mfk{b}\)
    (Proposition~\ref{prop:extension-contraction-properties}),
    so \(\mfk{b}\) is the extension of \(\mfk{b}^c\).
\end{proof}
\begin{lemma}\label{lem:fraction-contractions}
    Consider the ring homomorphism \(\func{\phi}{R}{S^{-1}R}\).
    If \(\mfk{a} \subseteq R\),
    then \(\mfk{a}^{ec} = \bigcup_{s \in S} (\mfk{a}:s)\).
    Conversely, \(\mfk{a} = \mfk{b}^c\) is a contracted ideal
    if and only if no \(s \in S\) are zero divisors in \(R/\mfk{a}\).
\end{lemma}
\begin{proof}
    Consider \(r \in \mfk{a}^{ec} = {(S^{-1}\mfk{a})}^c\).
    Then \(r = a/s \in S^{-1}\mfk{a}\) for \(a \in \mfk{a}\) and \(s \in S\).
    By definition, there exists some \(s' \in S\) such that \(s'(rs-a) = 0\),
    which implies \(rss' = as' \in \mfk{a}\),
    and \(r \in (\mfk{a}:s)\) for a specific \(s\).
    The reverse direction holds as every step here is bidirectional.

    Now consider any contracted ideal \(\mfk{a} = \mfk{b}^c\).
    Then by Proposition~\ref{prop:extension-contraction-properties}, \(\mfk{a} = \mfk{a}^{ec}\).
    If \(s \in S\) is a zero divisor,
    then there exists \(r \in R\) such that \(s[r] = [0] \in R/\mfk{a}\),
    i.e.\ \(sr \in \mfk{a}\) for some \(r \notin \mfk{a}\).
    But this tells us that \(s \in S \cap \mfk{a}\),
    so \(\mfk{a}^{ec} = (1)\).
    Clearly \((1)\) is not the only contracted ideal,
    so we have a contradiction.
    Now suppose there are no elements of \(S\) that are zero divisors in \(R/\mfk{a}\).
    Then \(sr \in \mfk{a}\) for any \(s \in S\) implies \(r \in \mfk{a}\).
    But that exactly shows that if \(r \in \mfk{a}^{ec}\),
    then \(r\) is the contraction of some element \(r \in \mfk{a}^e\),
    i.e.\ \(r \in \mfk{a}\) in the first place.
    We have shown \(\mfk{a}^{ec} \subseteq \mfk{a}\),
    which suffices for equality.
\end{proof}
\begin{theorem}\label{thm:fraction-correspondence}
    There is a bijective correspondence between 
    the prime ideals of \(R\) that do not intersect with \(S\),
    and the prime ideals of \(S^{-1}R\).
    In that case, \(\mfk{a} \subseteq R\) corresponds to its extension,
    \(\mfk{b} \subseteq S^{-1}R\) corresponds to its contraction,
    and \(s \in S \subseteq R\) corresponds to \((1) \subseteq S^{-1}R\).
\end{theorem}
\begin{proof}
    We already know that if \(\mfk{b} \subseteq S^{-1}R\) is prime,
    then its contraction is also prime,
    so it suffices to prove that if \(\mfk{a} \subseteq R\) prime, then \(S^{-1}\mfk{a}\) prime.
    Suppose \(\mfk{a} \subseteq R\) prime.
    Then \(R/\mfk{a}\) is an integral domain (Theorem~\ref{thm:prime-quotient-domain}).
    Hence \(S^{-1}(R/\mfk{a}) \cong S^{-1}R/S^{-1}\mfk{a}\) (Proposition~\ref{prop:fraction-properties}),
    and we see that this is an integral domain (or 0)
    as \([rr']/s = 0\) implies \([rr'] = 0\),
    and \(R/\mfk{a}\) being a domain follows.
    We can then conclude that \(S^{-1}\mfk{a} \subseteq S^{-1}R\) is prime
    (Theorem~\ref{thm:prime-quotient-domain}).

    By the \hyperref[lem:fraction-contractions]{lemma above},
    \(\mfk{a}\) intersects with \(S\) if and only if \(\mfk{a}^e = (1)\),
    so we cannot have a bijection only in that case,
    and a bijection is achieved in all other cases.
\end{proof}

\begin{proposition}\label{prop:fraction-ring-properties}
    Suppose \(R\) is a ring, and \(\mfk{a},\mfk{b}\) are ideals.
    Then we have:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(S^{-1}(\mfk{a}+\mfk{b}) = S^{-1}\mfk{a} + S^{-1}\mfk{b}\);
        \item \(S^{-1}(\mfk{ab}) = S^{-1}\mfk{a} S^{-1}\mfk{b}\);
        \item \(S^{-1}(\mfk{a}\cap\mfk{b}) = S^{-1}\mfk{a} \cap S^{-1}\mfk{b}\); and
        \item \(S^{-1}\sqrt{\mfk{a}} = \sqrt{S^{-1}\mfk{a}}\).
    \end{enumerate}
\end{proposition}
\begin{proof}
    (a) and (b) follow directly from Proposition~\ref{prop:extension-contraction-properties}.
    (c) follows from Proposition~\ref{prop:fraction-properties}.
    The forward inclusion of (d) follows from Proposition~\ref{prop:extension-contraction-properties}.

    Suppose \(x^n/s^n \in \sqrt{S^{-1}\mfk{a}}\), where \(x/s \in S^{-1}\mfk{a}\).
    Then clearly \(x \in \mfk{a}\), so \(x^n \in \sqrt{\mfk{a}}\).
    Hence \(x^n/s^n \in S^{-1}\sqrt{\mfk{a}}\).
\end{proof}


\section{Primary Decomposition}

\begin{definition}
    Suppose \(R\) is a ring, and \(\mfk{q} \subsetneq R\) is a proper ideal.
    \(\mfk{q}\) is primary if \(xy \in \mfk{q}\) implies \(x \in \mfk{q}\) or \(y^n \in \mfk{q}\) for some \(n\).
\end{definition}
\begin{proposition}
    Suppose \(R\) is a ring, and \(\mfk{q} \subsetneq R\) is a proper ideal.
    \(\mfk{q}\) is primary if and only if every zero divisor in \(R/\mfk{q}\) is nilpotent.
\end{proposition}
\begin{proof}
    Suppose \(\mfk{q}\) is primary.
    Consider a zero divisor \([y] \in R/\mfk{q}\).
    Then there exists some nonzero \([x] \in R/\mfk{q}\) such that \([x][y] = 0\).
    But this exactly implies \(xy \in \mfk{q}\),
    which implies either \(x \in \mfk{q}\) (which we assumed is not the case),
    or \(y^n \in \mfk{q}\), i.e.\ \([y^n] = 0\).
    
    Suppose every zero divisor \([y] \in R/\mfk{q}\) is nilpotent, with \([y^n] = 0\).
    Then there exists some nonzero \([x] \in R/\mfk{q}\) such that \([x][y] = 0\).
    But this exactly implies \(xy \in \mfk{q}\),
    and \(y^n \in \mfk{q}\) as desired.
\end{proof}
\begin{theorem}
    Suppose \(R\) is a ring, and \(\mfk{q} \subsetneq R\) is a proper ideal.
    If \(\mfk{q}\) is primary, then \(\sqrt{\mfk{q}}\) is prime.
\end{theorem}
\begin{proof}
    If \(xy \in \sqrt{\mfk{q}}\),
    then \({(xy)}^m = x^m y^m \in \mfk{q}\).
    But as \(\mfk{q}\) is primary, this implies either \(x^m \in \mfk{q}\) or \(y^{mn} \in \mfk{q}\).
    By definition of the radical, this implies either \(x \in \sqrt{\mfk{q}}\) or \(y \in \sqrt{\mfk{q}}\).
\end{proof}
\begin{definition}
    Suppose \(\mfk{q}\) is primary.
    We say \(\mfk{q}\) is \(\mfk{p}\)-primary if \(\sqrt{\mfk{q}} = \mfk{p}\).
\end{definition}

\begin{lemma}
    Suppose \({\{\mfk{q}_i\}}_{i=1}^n\) is a family of \(\mfk{p}\)-primary ideals.
    Then \(\bigcap_{i=1}^n \mfk{q}_i\) is also \(\mfk{p}\)-primary.
\end{lemma}
\begin{proof}
    We first check that \(\mfk{q} = \bigcap \mfk{q}_i\) is indeed primary.
    Consider \(xy \in \mfk{q}\).
    If \(x \notin \mfk{q}\), and hence \(x \notin \mfk{q}_i\),
    then we have \(y^{m_i} \in \mfk{q}_i\),
    and hence \(y^{\max m_i} \in \mfk{q}\).

    A quick application of Proposition~\ref{prop:radical-properties} then gives
    \(\sqrt{\bigcap \mfk{q}_i} = \bigcap \sqrt{\mfk{q}_i} = \bigcap \mfk{p} = \mfk{p}\).
\end{proof}
\begin{definition}
    Suppose \(\mfk{q} \subseteq R\) is an ideal.
    If \(\mfk{q} = \bigcap_i \mfk{q}_i\), we say \(\mfk{q}\) is decomposable.
    In this case, we may assume \(\sqrt{\mfk{q}_i} = \mfk{p}_i\) are distinct,
    and hence \(\mfk{q}_i\) should not be inside any \(\mfk{q}_j\).
    We call such a decomposition the minimal or irredundant decomposition.
\end{definition}
\begin{remark}
    Note that in general we cannot guarantee the existence of a primary decomposition.
\end{remark}

\begin{lemma}\label{lem:quotient-primary}
    Suppose \(\mfk{q}\) is \(\mfk{p}\)-primary, and \(x \in R\).
    We have the following:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item if \(x \in \mfk{q}\), then \((\mfk{q}:x) = (1)\); and
        \item if \(x \notin \mfk{q}\), then \((\mfk{q}:x)\) is \(\mfk{p}\)-primary.
    \end{enumerate}
\end{lemma}
\begin{proof}
    Almost by definition, \(xR \subset \mfk{q}\),
    so \((\mfk{q}:x) = R\), and we have statement (a).

    Consider \(yz \in (\mfk{q}:x)\), and without loss of generality \(y \notin (\mfk{q}:x)\);
    we wish to prove that \(z^n \in (\mfk{q}:x)\).
    Now, by definition of the quotient we have \(xyz \in \mfk{q}\),
    and as \(x \notin \mfk{q}\) and \(y \notin \mfk{q}\), \(xy \notin \mfk{q}\),
    which forces us to conclude that \(z^n \in \mfk{q} \subseteq (\mfk{q}:x)\)
    (Proposition~\ref{prop:quotient-ideal-properties}).
    This shows that \((\mfk{q}:x)\) is indeed primary.
    We now show that \(\sqrt{(\mfk{q}:x)} = \mfk{p}\).
    Observe that for any element \(y \in (\mfk{q}:x)\), \(xy \in \mfk{q}\).
    As \(x \notin \mfk{q}\), we have \(y^n \in \mfk{q}\), so \(y \in \sqrt{\mfk{p}} = \mfk{p}\).
    We have exactly shown that \((\mfk{q}:x) \subseteq \mfk{p}\).
    But we have \(\mfk{p} = \sqrt{\mfk{q}} \subseteq \sqrt{(\mfk{q}:x)} \subseteq \sqrt{\mfk{p}} = \mfk{p}\),
    and we have equality.
    This proves statement (b).
\end{proof}
\begin{lemma}\label{lem:prime-intersection}
    Suppose a prime ideal \(\mfk{p} = \bigcap_i \mfk{a}_i\) for a family of ideals.
    Then \(\mfk{p} = \mfk{a}_i\) for some \(i\).
\end{lemma}
\begin{proof}
    Suppose, by way of contradiction, that \(\mfk{p} \subsetneq \mfk{a}_i\) for all \(i\).
    For each \(i\), we choose some \(x_i \in \mfk{a}_i - \mfk{p}\).
    Then we have \(\prod x_i \in \prod \mfk{a}_i \subseteq \bigcap \mfk{a}_i\),
    but as each \(x_i \notin \mfk{p}\), this product \(\prod x_i \notin \mfk{p}\).
    This contradicts our assumed equality,
    and we conclude that \(\mfk{a}_i \subseteq \mfk{p}\).
    The reverse inclusion \(\mfk{p} = \bigcap \mfk{a}_i \subseteq \mfk{a}_i\) is given trivially.
    We have equality.
\end{proof}
\begin{theorem}[First Uniqueness Theorem for Primary Decompositions]\label{thm:primary-uniqueness-1}
    Suppose \(\mfk{q}\) is a decomposable ideal with an irredundant decomposition \(\mfk{q} = \bigcap_i \mfk{q}_i\).
    Then the associated primes \(\mfk{p}_i = \sqrt{\mfk{q}_i}\) are unique up to reordering.
\end{theorem}
\begin{proof}
    Consider the set of ideals \({\{\sqrt{(\mfk{q}:x)}\}}_{x \in R}\).
    We prove that the prime ideals in that set is exactly \({\{\mfk{p}_i\}}_i\).
    In the forward inclusion, suppose we have an arbitrary \(x \in R\).
    If \(x \in \mfk{q}\), then \(\sqrt{(\mfk{q}:x)} = \sqrt{(1)} = (1)\).
    If not, by Lemma~\ref{lem:quotient-primary}, then it is the prime ideal
    \begin{equation*}
        \mfk{p} = \sqrt{(\mfk{q}:x)} = \sqrt{\pqty{\bigcap \mfk{q}_i:x}} = \sqrt{\bigcap (\mfk{q}_i:x)}
        = \bigcap\sqrt{(\mfk{q}_i:x)} = \bigcap_{x \notin \mfk{q}_i} \mfk{p}_i
    \end{equation*}
    using Propositions~\ref{prop:radical-properties}~\&~\ref{prop:quotient-ideal-properties}.
    From the \hyperref[lem:prime-intersection]{lemma above},
    we know that \(\mfk{p} = \mfk{p}_i\) for some \(i\).
    Now for the reverse inclusion, consider some \(x \in \bigcap_{j \neq i} \mfk{q}_j - \mfk{q}_i\).
    Then \(x \notin \mfk{q}_i\) and \(x \in \mfk{q}_j\) for every other \(j \neq i\),
    so we have
    \begin{equation*}
        \sqrt{(\mfk{q}:x)} = \sqrt{\pqty{\bigcap \mfk{q}_i:x}} = \sqrt{\bigcap (\mfk{q}_i:x)}
        = \bigcap\sqrt{(\mfk{q}_i:x)} = \bigcap_{x \notin \mfk{q}_j} \mfk{p}_j = \mfk{p}_i
    \end{equation*}
    Then we immediately have that the associated primes are uniquely determined by \(\mfk{q}\).
\end{proof}

\begin{proposition}\label{prop:fraction-primary}
    Suppose \(S \subseteq R\) is a multiplicative monoid,
    and \(\mfk{q}\) is a \(\mfk{p}\)-primary ideal.
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item If \(S\cap\mfk{p} \neq \emptyset\), then \(S^{-1}\mfk{q} = S^{-1}R\); and
        \item if \(S\cap\mfk{p} = \emptyset\), then \(S^{-1}\mfk{q}\) is \(S^{-1}\mfk{p}\)-primary,
            and \(S^{-1}\mfk{q}\) contracts to \(\mfk{q}\).
    \end{enumerate}
\end{proposition}
\begin{proof}
    Consider \(s \in S\cap\mfk{p}\).
    We have some \(s^n \in S\cap\mfk{q}\) due to \(\sqrt{\mfk{q}} = \mfk{p}\).
    Then this element in the extension is \(s^n \in S^{-1}\mfk{q}\),
    so \(s^n/s^n = 1 \in S^{-1}\mfk{q}\),
    showing that \(S^{-1}\mfk{q} = S^{-1}R\).
    This proves statement (a).

    If \(S\cap\mfk{p} = \emptyset\), consider any \(s \in S\).
    Given any \(r\), if \(rs \in \mfk{q}\),
    as \(\mfk{q}\) is primary and \(s^n \notin \mfk{q}\),
    we have \(r \in \mfk{q}\).
    Therefore, by definition of the quotient ideal, \((\mfk{q}:s) \subseteq \mfk{q}\)
    By Lemma~\ref{lem:fraction-contractions},
    \(\mfk{q}^{ec} = \bigcup_{s \in S} (\mfk{q}:s) \subseteq \mfk{q}\),
    and the reverse inclusion is trvially given by Proposition~\ref{prop:extension-contraction-properties};
    we have \(\mfk{q}^{ec} = {(S^{-1}\mfk{q})}^c = \mfk{q}\).

    We can immediately see that \(\sqrt{S^{-1}\mfk{q}} = S^{-1}\sqrt{\mfk{q}} = S^{-1}\mfk{p}\)
    by an application of Proposition~\ref{prop:fraction-ring-properties}.
    We now verify that \(S^{-1}\mfk{q}\) is primary.
    Suppose \((x/t)(y/s) \in S^{-1}\mfk{q}\), and \(x/t \notin S^{-1}\mfk{q}\).
    Then it is easy to see that \(x \notin \mfk{q}\),
    so \(y^n \in \mfk{q}\) for some \(n\).
    Hence \(y^n/s^n \in S^{-1}\mfk{q}\).
    This proves statement (b).
\end{proof}
\begin{proposition}\label{prop:fraction-decomposition}
    Suppose \(S \subseteq R\) is a multiplicative monoid,
    and \(\mfk{q} = \bigcap_i \mfk{q}_i\) has an irredundant decomposition,
    with \(\mfk{p}_i = \sqrt{\mfk{q}_i}\).
    Without loss of generality, suppose \(S \cap \mfk{p}_i = \emptyset\) for the first \(n\) ideals,
    and \(S \cap \mfk{p}_i \neq \emptyset\) for the rest of the ideals.
    Then we an have irredundant decomposition for \(S^{-1}\mfk{q} = \bigcap_{i=1}^n S^{-1}\mfk{q}_i\),
    and for its contraction \({(S^{-1}\mfk{q})}^c = \bigcap_{i=1}^n \mfk{q}_i\).
\end{proposition}
\begin{proof}
    By Proposition~\ref{prop:fraction-ring-properties},
    \(S^{-1}\mfk{q} = S^{-1}\pqty{\bigcap_i \mfk{q}_i} = \bigcap_i S^{-1}\mfk{q}_i\),
    and we know we can toss out everything but the first \(n\) ideals
    because of the \hyperref[prop:fraction-primary]{proposition above}.
    This is indeed a primary decomposition,
    as we know that \(S^{-1}\mfk{q}_i\) are all \(S^{-1}\mfk{p}_i\)-primary
    again from the \hyperref[prop:fraction-primary]{proposition above}.
    This decomposition is also irredundant because \(\mfk{p}_i\) are distinct,
    and therefore \(S^{-1}\mfk{p}_i\) are also distinct.

    We can then take the contraction of everything to get
    \({(S^{-1}\mfk{q})}^c = {(\bigcap_{i=1}^n S^{-1}\mfk{q}_i)}^c
    = \bigcap_{i=1}^n {(S^{-1}\mfk{q}_i)}^c = \bigcap_{i=1}^n \mfk{q}_i\),
    simply by using Proposition~\ref{prop:extension-contraction-operations}
    and the \hyperref[prop:fraction-primary]{proposition above}.
\end{proof}
\begin{definition}
    Suppose \(\mfk{q}\) is a decomposable ideal,
    and \({\{\mfk{p}_i\}}_{i \in S}\) is a subset of the associated prime ideals.
    This set of ideals is isolated if any associated prime \(\mfk{p} \subseteq \mfk{p}_i\)
    is also in this subset \(\mfk{p} \in {\{\mfk{p}_i\}}_{i \in S}\).
\end{definition}
\begin{theorem}[Second Uniqueness Theorem for Primary Decompositions]\label{thm:primary-uniqueness-2}
    Suppose \(\mfk{q}\) is a decomposable ideal with an irredundant decomposition \(\mfk{q} = \bigcap_i \mfk{q}_i\).
    Let \({\{\mfk{p}_{i_k}\}}_{k \in S}\) be an isolated set of associated primes.
    Then \(\bigcap_{i_k} \mfk{q}_{i_k}\) is uniquely determined independent of the decomposition.
\end{theorem}
\begin{proof}
    Consider an ideal \(\mfk{q}\)
    with an irredundant decomposition \(\mfk{q} = \bigcap_i \mfk{q}_i\),
    and that the first \(n\) of the associated primes \(\sqrt{\mfk{q}_i} = \mfk{p}_i\) are isolated.
    Take \(S = R - \bigcup_{i=1}^n \mfk{p}_i\),
    i.e.\ the localization at \(\bigcup_{i=1}^n \mfk{p}_i\).
    Then trivially, for these \(n\) ideals, \(\mfk{p}_i \cap S = \emptyset\),
    and the other ideals, \(\mfk{p}_i \cap S \neq \emptyset\).
    Then immediately from the \hyperref[prop:fraction-decomposition]{proposition above},
    \({(S^{-1}\mfk{q})}^c = \bigcap_{i=1}^n \mfk{q}_i\).
    
    Now suppose \(\mfk{q}\) has a different irredundant decomposition
    \(\mfk{q} = \bigcap_i \mfk{q}'_i\),
    with the same associated primes \(\sqrt{\mfk{q}'_i} = \mfk{p}_i\)
    as determined by the \hyperref[thm:primary-uniqueness-1]{first uniqueness theorem}.
    Then the \hyperref[prop:fraction-decomposition]{proposition above} yields
    \({(S^{-1}\mfk{q})}^c = \bigcap_{i=1}^n \mfk{q}_i = \bigcap_{i=1}^n \mfk{q}'_i\).
\end{proof}
\begin{corollary}
    Suppose \(\mfk{q}\) is a decomposable ideal with an irredundant decomposition \(\mfk{q} = \bigcap_i \mfk{q}_i\),
    and \(\mfk{p}_i\) be a minimal associated prime,
    as in there are no other associated prime ideals \(\mfk{p} \subseteq \mfk{p}_i\).
    Then \(\mfk{q}_i\) is uniquely determined by \(\mfk{q}\).
\end{corollary}
\begin{proof}
    Simply take \(\{\mfk{p}_i\}\) as an isolated set of ideals,
    and apply the \hyperref[thm:primary-uniqueness-2]{theorem above}.
\end{proof}

\begin{definition}
    Suppose \(A \subseteq R\), where \(R\) is an \(A\)-algebra.
    \(R\) is finite over \(A\) if \(R\) is a finitely generated \(A\)-module;
    that is, there is a surjection \(\bigoplus_{i=1}^n x_i A \to R\).
    \(R\) is finite-type over \(A\) if \(R\) is a finitely generated \(A\)-algebra;
    that is, there is a surjection \(A[x_1,\hdots,x_n] \to R\).
\end{definition}
\begin{definition}
    Suppose \(R\) is an \(A\)-algebra, and \(x \in R\).
    We say \(x\) is integral over \(A\) if it satisfies a monic polynomaial
    \(x^n + a_1 x^{n-1} + \cdots + a_n = 0\) for \(a_i \in A\).
    We say \(x\) is algebraic over \(A\) if it satisfies a polynomaial
    \(a_0 x^n + a_1 x^{n-1} + \cdots + a_n = 0\) for \(a_i \in A\).
\end{definition}
\begin{definition}
    \(A \subseteq R\) is integral if every \(x \in R\) is integral over \(A\).
\end{definition}
\begin{lemma}\label{lem:integral-finite}
    Suppose \(R\) is an \(A\)-algebra.
    \(x \in R\) is integral over \(A\) if and only if \(A[x]\) is finite over \(A\).
\end{lemma}
\begin{proof}
    Suppose \(x\) is integral over \(A\),
    with \(x^n + a_1x^{n-1} + \cdots + a_n = 0\).
    Then any element \(f \in A[x]\) can be written as a polynomial of degree at most \(n-1\) over \(x\),
    as any \(x^n\) factors can be reduced to a polynomial of degree at most \(n-1\).

    Suppose \(A[x]\) is finite over \(A\).
    Then consider the map \(\vfunc{\phi}{A[x]}{A[x]}{f}{xf}\) that multiplies by \(x\).
    By the \hyperref[lem:cayley-hamilton]{Cayley-Hamilton Theorem},
    \(0 = \phi^n + a_1\phi^{n-1} + \cdots + a_n = x^n + a_1 x^{n-1} + \cdots + a_n\) as desired.
\end{proof}
\begin{corollary}
    Suppose \(R\) is an \(A\)-algebra.
    If \(x \in R\) is integral over \(A\),
    then \(A[x]\) is integral over \(A\).
    Hence finite extensions are integral.
\end{corollary}
\begin{proof}
    Consider any element \(y \in A[x]\),
    and the map \(\vfunc{\phi}{A[x]}{A[x]}{f}{yf}\) that multiplies by \(y\).
    By the \hyperref[lem:cayley-hamilton]{Cayley-Hamilton Theorem},
    \(0 = \phi^n + a_1\phi^{n-1} + \cdots + a_n = y^n + a_1 y^{n-1} + \cdots + a_n\) as desired.
\end{proof}
\begin{corollary}
    Suppose \(R\) is an \(A\)-algebra,
    and \({\{x_i\}}_{i=1}^n \subset R\) are all integral over \(A\).
    Then \(A[x_1,\hdots,x_n]\) is finite over \(A\).
    Hence extensions are finite if and only if they are both finite-type and integral.
\end{corollary}
\begin{proof}
    Let the \hyperref[lem:integral-finite]{lemma above} be the base case,
    and proceed by induction.
    Let \(A_k = A[x_1,\hdots,x_k]\), and assume that it is finite over \(A\).
    We wish to prove that \(A_{k+1} = A_k[x_{k+1}]\) is finite over \(A\).
    But because \(x_{k+1}\) is integral over \(A\), it is integral over \(A_k\),
    so by the \hyperref[lem:integral-finite]{lemma above},
    \(A_{k+1}\) is finite over \(A_k\), which is then finite over \(A\).
\end{proof}

\begin{proposition}
    Suppose \(A \subseteq B \subseteq R\) are rings.
    We have the following properties:

\begin{definition}
    Suppose \((\Sigma,\leq)\) is a partially ordered set.
    \(\Sigma\) satisfies the ascending chain condition
    if all ascending chains in \(\Sigma\) eventually stabilizes; that is,
    \begin{equation*}
        x_1 \leq x_2 \leq \cdots \implies x_n = x_{n+1} = \cdots
    \end{equation*}
    \(\Sigma\) satisfies the descending chain condition
    if all descending chains in \(\Sigma\) eventually stabilizes; that is,
    \begin{equation*}
        x_1 \geq x_2 \geq \cdots \implies x_n = x_{n+1} = \cdots
    \end{equation*}
\end{definition}
\begin{definition}
