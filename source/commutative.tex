\section{Fractions}

\subsection*{Ring of Fractions}

\begin{definition}
    Suppose \(R\) is a ring, \(S \subseteq R\) is a multiplicative monoid.
    Then the ring of fractions is \(S^{-1}R = \{r/s : r \in R, s \in S\}/{\sim}\),
    where the equivalence relation is given by \(r_1/s_1 = r_2/s_2\)
    if there exists some nonzero \(s \in S\) such that \(s(r_1s_2 - r_2s_1) = 0\).
\end{definition}
\begin{proposition}
    There exists a ring homomorphism \(\vfunc{\iota}{R}{S^{-1}R}{n}{n/1}\).
\end{proposition}
\begin{proof}
    Observe that the ring homomorphism axioms are satisfied as
    \(\iota(0) = 0/1 = 0\),
    \(\iota(1) = 1/1 = 1\),
    \(\iota(r)\iota(s) = (r/1)(s/1) = rs/1 = \iota(rs)\),
    and \(\iota(r)+\iota(s) = (r/1)+(s/1) = (r+s)/1 = \iota(r+s)\).
\end{proof}
\begin{lemma}
    Suppose \(R\) is a ring, \(S \subseteq R\) is a multiplicative monoid.
    If \(s \in S\), then \(s\) is a unit in \(S^{-1}R\).
\end{lemma}
\begin{proof}
    \(s(1/s) = (s/1)(1/s) = 1\).
\end{proof}
\begin{lemma}
    Suppose \(R\) is a ring, \(S \subseteq R\) is a multiplicative monoid.
    \(r \in R\) is zero in \(S^{-1}R\) if and only if there exists some \(s\)
    such that \(sr = 0\) in \(R\).
\end{lemma}
\begin{proof}
    By definition, \(r = r/1 = 0/1 = 0\) if and only if \(s(r-0) = sr = 0\).
\end{proof}

\begin{theorem}[Universal Property of Ring of Fractions]
    Suppose \(R\) is a ring, \(S \subseteq R\) is a multiplicative monoid.
    The ring of fractions is \(S^{-1}R\), if:
    \begin{enumerate}[label={(\roman*)}, itemsep=0mm]
        \item there exists a homomorphism \(\func{\iota}{R}{S^{-1}R}\); and
        \item for any ring \(T\), if \(\func{\phi}{R}{T}\) is a homomorphism that sends \(s \in S\) to a unit,
            then there exists a unique \(\func{\bar{\phi}}{S^{-1}R}{T}\)
            such that \(\phi = \bar{\phi}\circ\iota\).
    \end{enumerate}
    
    This is represented by the following commutative diagram:
    \begin{equation*}
        \begin{tikzcd}
            R \arrow{r}{\phi} \arrow{d}{\iota} & T \\
            S^{-1}R \arrow[dashrightarrow]{ur}[swap]{\exists!\bar{\phi}}
        \end{tikzcd}
    \end{equation*}
\end{theorem}
\begin{proof}
    Suppose there exists \(\bar{\phi}\neq\bar{\phi}'\) that both fit the diagram.
    Then they disagree on some \(r/s\).
    But then we have
    \begin{equation*}
        s\bar{\phi}(r/s) = \bar{\phi}(r/1) = (\bar{\phi}\circ\iota)(r)
        = \phi(r) = (\bar{\phi}'\circ\iota)(r) = \bar{\phi}'(r/1) = s\bar{\phi}(r/s)
    \end{equation*}
    which is a contradiction.

    Now, we can simply define \(\bar{\phi}(r/s) = \phi(r){(\phi(s))}^{-1}\) as desired.
\end{proof}

\begin{proposition}
    Suppose \(R\) is a domain, and \(S = R - \{0\}\) is the set of nonzero elements in \(R\).
    Then \(S^{-1}R\) is the field of fractions.
\end{proposition}
\begin{proof}
    See existence of field of fractions in Theorem~\ref{thm:existence-field-of-fractions}.
    In particular, we have \(a/b = c/d\) when \(ad = bc\) is sufficient,
    since \(s(ad - bc) = 0\) is equivalent to \(ad - bc = 0\) since we are in an integral domain.
\end{proof}

\begin{definition}
    Suppose \(R\) is a ring, \(\mfk{p} \subseteq R\) is a prime ideal,
    and \(S = R - \mfk{p}\).
    Then we call \(S^{-1}R = R_\mfk{p}\) to be the localization of \(R\) at \(\mfk{p}\).
\end{definition}
\begin{proposition}\label{prop:prime-local-ring}
    Suppose \(R\) is a ring, and \(\mfk{p} \subseteq R\) is a prime ideal.
    \((R_\mfk{p},\mfk{p}R_\mfk{p})\) forms a local ring.
\end{proposition}
\begin{proof}
    Consider any \(r/s \in R_\mfk{p} - \mfk{p}R_\mfk{p}\).
    We claim that \(r \notin \mfk{p}\),
    since if it is, \(r/s = r(1/s) \in \mfk{p}R_\mfk{p}\) which is a contradiction.
    Hence \(r \in R - \mfk{p}\), and \(s/r \in R_\mfk{p}\).
    But we have just proven that every element in \(R_\mfk{p} - \mfk{p}R_\mfk{p}\) is a unit,
    so by Proposition~\ref{prop:local-ring-units} \((R_\mfk{p},\mfk{p}R_\mfk{p})\) is a local ring.
\end{proof}

\subsection*{Module of Fractions}

\begin{definition}
    Suppose \(S^{-1}R\) is a ring of fractions, and \(M\) be an \(R\)-module.
    Then \(S^{-1}M = \{m/s : m \in M, s \in S\}/{\sim}\) is an \(S^{-1}R\)-module,
    with the equivalence relation is given by \(m_1/s_1 = m_2/s_2\)
    if there exists some \(s \in S\) such that \(s(m_1s_2 - m_2s_1) = 0\).
\end{definition}
\begin{proposition}
    There exists a module homomorphism \(\vfunc{\iota}{M}{S^{-1}M}{m}{m/1}\),
    where \(\ker\iota = \{m : \exists s \in S, ms = 0\}\).
\end{proposition}
\begin{proof}
    Observe that the module homomorphism axioms are satisfied as
    \(\iota(0) = 0/1 = 0\) and
    \(r\iota(m)+\iota(n) = r(m/1)+(n/1) = (rm)/1+(n/1) = (rm+n)/1 = \iota(rm+n)\).
    Obviously, \(\iota(m) = m/1 = 0\) only if \(s(m1 - 0) = sm = 0\).
\end{proof}

\begin{proposition}
    Suppose \(R\) is a ring, and \(S^{-1}R\) is a ring of fractions.
    Then \(\vfunc{S^{-1}}{\textbf{Mod}_R}{\textbf{Mod}_{S^{-1}R}}{M}{S^{-1}M}\)
    is a functor.
\end{proposition}
\begin{proof}
    We already see that it sends modules to modules from our definition of module of fractions,
    so it suffices to see that it maps module homomorphisms.
    Suppose \(\func{\phi}{M}{N}\) is a module homomorphism.
    Then we can simply define \(\vfunc{\phi}{S^{-1}M}{S^{-1}N}{m/s}{\phi(m)/s}\).
\end{proof}
\begin{theorem}\label{thm:fraction-exact}
    \(S^{-1}\) is an exact functor.
\end{theorem}
\begin{proof}
    Consider the exact sequence \(M \overset{f}{\to} N \overset{g}{\to} P\).
    See that if \((g \circ f)(m) = 0\),
    then obviously \((g \circ f)(m/s) = (g \circ f)(m)/s = 0\).
    This gives image in kernel.

    Now consider \(n/s \in S^{-1}N\) such that \(g(n)/s = 0\).
    But then that implies there exists \(s'\) such that \(s'g(n) = 0\);
    i.e.\ \(s'n \in \ker g\).
    By exactness, there exists \(m \in M\) such that \(f(m) = s'n\),
    and we can get \(f(m/(ss')) = f(m)/ss' = s'n/ss' = n/s\).
    This gives kernel in image.
\end{proof}

\begin{proposition}\label{prop:fraction-properties}
    Suppose \(P\) is an \(R\)-module,
    and \(M,N \subseteq P\) are submodules.
    The following properties hold for modules of fractions:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(S^{-1}(M+N) = S^{-1}M + S^{-1}N \subseteq S^{-1}P\);
        \item \(S^{-1}(M \cap N) = S^{-1}M \cap S^{-1}N \subseteq S^{-1}P\); and
        \item \(S^{-1}(M/N) \cong S^{-1}M/S^{-1}N \subseteq S^{-1}P\).
    \end{enumerate}
\end{proposition}
\begin{proof}
    For (a), there is a natural isomorphism \((m+n)/s \mapsto m/s + n/s\),
    since the inverse map is \(m/s + n/s \mapsto (m+n)/s\) obviously.

    For (b), observe that if \(m \in M \cap N\),
    \(m/s \in S^{-1}M\) and \(m/s S^{-1}N\) as desired.
    On the other hand, consider \(m/s \in S^{-1}M\) and \(n/t \in S^{-1}N\),
    and \(m/s = n/t\).
    then there exists some \(u \in S\) such that \(u(tm-sn)\)

    For (c), consider the exact sequence
    \begin{equation*}
        \begin{tikzcd}
            0 \arrow{r} & N \arrow{r} & M \arrow{r} & M/N \arrow{r} & 0
        \end{tikzcd}
    \end{equation*}
    Since \(S^{-1}\) is exact by Theorem~\ref{thm:fraction-exact},
    we have the exact sequence
    \begin{equation*}
        \begin{tikzcd}
            0 \arrow{r} & S^{-1}N \arrow{r} & S^{-1}M \arrow{r} & S^{-1}(M/N) \arrow{r} & 0
        \end{tikzcd}
    \end{equation*}
    and Proposition~\ref{prop:short-exact-sequence}
    gives us that \(S^{-1}(M/N) \cong S^{-1}M/S^{-1}N\) as desired.
\end{proof}

\begin{theorem}\label{thm:fraction-isomorphism}
    There is an isomorphism \(S^{-1}R \otimes M \cong S^{-1}M\).
\end{theorem}
\begin{proof}
    We prove that \(S^{-1}M\) fits the universal property of \(S^{-1}R \otimes M\).
    Suppose we have the commutative diagram
    \begin{equation*}
        \begin{tikzcd}
            S^{-1}R \times M \arrow{r}{\mu} \arrow{rd}{\phi} &
            S^{-1}M \arrow[dashrightarrow]{d}{\exists!\bar{\phi}} \\
            & P
        \end{tikzcd}
    \end{equation*}
    with \(\mu(r/s,m) = (rm)/s\).
    Suppose there exists \(\bar{\phi}\neq\bar{\phi}'\) that both fit this diagram.
    Then we know they must disagree on some \(\bar{\phi}(m/s)\neq\bar{\phi}'(m/s)\).
    But observe that
    \begin{equation*}
        \bar{\phi}\pqty{\frac{m}{s}} = (\bar{\phi}\circ\mu)\pqty{\frac{1}{s},m}
        = \phi\pqty{\frac{1}{s},m} = (\bar{\phi}'\circ\mu)\pqty{\frac{1}{s},m}
        = \bar{\phi}'\pqty{\frac{m}{s}}
    \end{equation*}
    which is a contraction.
    This gives us uniqueness.
    Now, we can readily define \(\bar{\phi}(m/s) = \phi(1/s,m)\),
    and observe that \(\bar{\phi}(rm/s) = \phi(1/s,rm) = r\phi(1/s,m) = \phi(r/s,m)\)
    indeed by bilinearity.
    This gives us existence.
\end{proof}
\begin{corollary}
    \(S^{-1}R\) is a flat \(R\)-module.
\end{corollary}
\begin{proof}
    The \hyperref[thm:fraction-isomorphism]{theorem above}
    gives us that \(S^{-1} = S^{-1}R \otimes \cdot\) as functors.
    Since \(S^{-1}\) is exact (Theorem~\ref{thm:fraction-exact}),
    \(S^{-1}R \otimes \cdot\) is also exact.
\end{proof}

\begin{proposition}\label{prop:fraction-tensor}
    Suppose \(M,N\) are (appropriate) \(R\)-modules.
    The following properties hold for modules of fractions:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(S^{-1}(M \oplus N) \cong S^{-1}M \oplus S^{-1}N\); and
        \item \(S^{-1}(M \otimes N) \cong S^{-1}M \otimes S^{-1}N\).
    \end{enumerate}
\end{proposition}
\begin{proof}
    For (a), this is a special case of Proposition~\ref{prop:fraction-properties},
    where the sum \(M + N\) is direct if we consider \(M,N \subseteq M \oplus N\).

    For (b), we have
    \begin{equation*}
        S^{-1}M \otimes S^{-1}N \cong (S^{-1}R \otimes M) \otimes (S^{-1}R \otimes N)
        \cong S^{-1}R \otimes S^{-1}R \otimes (M \otimes N)
        \cong S^{-1}R \otimes (M \otimes N)
        \cong S^{-1}(M \otimes N)
    \end{equation*}
\end{proof}
\begin{proposition}\label{prop:localization-flat}
    \(M\) is a flat \(R\)-module implies \(S^{-1}M\) is a flat \(S^{-1}R\)-module.
\end{proposition}
\begin{proof}
    If \(M \otimes \cdot\) is an exact functor,
    then \(M \otimes (S^{-1}R \otimes \cdot) \cong S^{-1}M \otimes \cdot\) is an exact functor.
\end{proof}

\begin{definition}
    Suppose \(M\) is an \(R\)-module, and \(M_\mfk{p},M_\mfk{m}\) some localizations.
    We say a module property is local when
    \(M\) has that property if and only if \(M_\mfk{p}\) for all prime \(\mfk{p} \subseteq R\) has that property.
    We say a module property is max-local when
    \(M\) has that property if and only if \(M_\mfk{m}\) for all maximal \(\mfk{m} \subseteq R\) has that property.
\end{definition}
\begin{lemma}\label{lem:local-max-local}
    Suppose \(M,N\) are \(R\)-modules, and \(\func{\phi}{M}{N}\) is a homomorphism.
    Then the following properties are local and max-local:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(M = 0\);
        \item \(\phi\) injective; and
        \item \(\phi\) surjective.
    \end{enumerate}
\end{lemma}
\begin{proof}
    We will prove the three statements in similar fashion,
    by first proving the forward direction of local,
    and then the reverse direction of max-local.
    Finally, using the fact that \hyperref[thm:maximal-prime-radical]{maximal ideals are prime},
    prime localizations having a property immediately implies
    maximal localizations must also have said property.

    \medskip

    First off, it is obvious that \(M = 0\)
    implies \(M_\mfk{p} = 0\) for all prime \(\mfk{p} \subseteq R\).

    We will prove the reverse of max-locality via contraposition.
    Suppose \(M \neq 0\).
    Consider some nonzero element \(m \in M\)
    and its annihilator \(I = \Ann m \subsetneq (1) = R\).
    By Corollary~\ref{cor:proper-in-maximal},
    \(I \subseteq p\) for some maximal ideal \(p \subseteq R\).
    Then we claim that \(m \in M_p\) is nonzero,
    because if it is, then there is some element \(r \in R - p\) that kills \(m\),
    which contradicts that \(\Ann m \subseteq p\).
    This proves that there is some nonzero \(M_p\),
    and we have proven statement (a).

    \medskip

    Suppose \(0 \to M \overset{\phi}{\to} N\) is exact.
    Since \(S^{-1}\) is exact (Theorem~\ref{thm:fraction-exact}),
    \(0 \to M_\mfk{p} \overset{\phi}{\to} N_\mfk{p}\) is also exact.
    
    Suppose \(0 \to M_\mfk{m} \overset{\phi}{\to} N_\mfk{m}\) is exact for all maximal \(\mfk{m} \subseteq R\).
    Consider the exact sequence \(0 \to \ker\phi \to M \overset{\phi}{\to} N\).
    Since \(S^{-1}\) is exact (Theorem~\ref{thm:fraction-exact}),
    \(0 \to {(\ker\phi)}_\mfk{m} \to M_\mfk{m} \overset{\phi}{\to} N_\mfk{m}\) is also exact,
    which implies \({(\ker\phi)}_\mfk{m} = 0\) for all maximal \(\mfk{m} \subseteq R\),
    and by max-locality in (a), \(\ker\phi = 0\) as desired.
    We have proven statement (b).
    
    \medskip
    
    Suppose \(M \overset{\phi}{\to} N \to 0\) is exact.
    Since \(S^{-1}\) is exact (Theorem~\ref{thm:fraction-exact}),
    \(M_\mfk{p} \overset{\phi}{\to} N_\mfk{p} \to 0\) is also exact.
    
    Suppose \(M_\mfk{m} \overset{\phi}{\to} N_\mfk{m} \to 0\) is exact for all maximal \(\mfk{m} \subseteq R\).
    Consider the exact sequence \(M \overset{\phi}{\to} N \to \coker\phi \to 0\).
    Since \(S^{-1}\) is exact (Theorem~\ref{thm:fraction-exact}),
    \(M_\mfk{m} \overset{\phi}{\to} N_\mfk{m} \to {(\coker\phi)}_\mfk{m} \to 0\) is also exact,
    which implies \({(\coker\phi)}_\mfk{m} = 0\) for all maximal \(\mfk{m} \subseteq R\),
    and by max-locality in (a), \(\coker\phi = 0\) as desired.
    We have proven statement (c).
\end{proof}
\begin{theorem}
    Flatness is local and max-local.
\end{theorem}
\begin{proof}
    We prove in the same manner as the \hyperref[lem:local-max-local]{lemma above}.
    Suppose \(M\) is flat.
    Then \(M_\mfk{p}\) is flat for all prime \(\mfk{p}\) by Proposition~\ref{prop:localization-flat}.

    Using the fact that \hyperref[thm:maximal-prime-radical]{maximal ideals are prime},
    prime localizations being flat immediately implies
    maximal localizations must also be flat.

    Now suppose \(M_\mfk{m}\) is flat for all maximal \(\mfk{m} \subseteq R\).
    If \(N \to P\) is injective,
    then \(N_\mfk{m} \to P_\mfk{m}\) is injective by the \hyperref[lem:local-max-local]{lemma above},
    and by flatness, \(M_\mfk{m} \otimes N_\mfk{m} \to M_\mfk{m} \otimes P_\mfk{m}\) is injective.
    By Proposition~\ref{prop:fraction-tensor},
    this is equivalent to saying \({(M \otimes N)}_\mfk{m} \to {(M \otimes P)}_\mfk{m}\) is injective.
    Using the \hyperref[lem:local-max-local]{lemma above} again,
    \(M \otimes N \to M \otimes P\) is injective,
    which tells us that \(M\) is flat (Lemma~\ref{lem:flat-equivalence}).
\end{proof}

\subsection*{Extensions and Contractions}

\begin{lemma}\label{lem:fraction-extensions}
    Consider the ring homomorphism \(\func{\phi}{R}{S^{-1}R}\).
    If \(\mfk{a} \subseteq R\), its extension is \(\mfk{a}^e = \mfk{a} \otimes S^{-1}R = S^{-1}\mfk{a}\).
    Conversely, every ideal in \(S^{-1}R\) is an extension of some ideal.
\end{lemma}
\begin{proof}
    Suppose \(r \in \mfk{a}\).
    Then for any \(s \in S\), \((1/s)r = r/s \in \mfk{a}^e\),
    so we have \(S^{-1}\mfk{a} \subseteq \mfk{a}^e\).
    Now suppose \(\sum r_i/s_i \in \mfk{a}^e\).
    We can assume \(r_i \in \mfk{a}\),
    since we have \(\sum (t_i/s_i)r_i = \sum (t_i r_i)/s_i \in \mfk{a}^e\),
    and \(t_i r_i \in \mfk{a}R \subseteq \mfk{a}\).
    We write this with a common denominator \(\sum r_i/s_i = r/s\),
    and we can immediately conclude that \(r\) is a sum of \(r_i\) and \(s_i\)s,
    so \(r \in \mfk{a}\).

    Now consider some ideal \(\mfk{b} \in S^{-1}R\).
    By the same logic as above, it suffices to consider \(r/s \in \mfk{b}\),
    and we know that \(s(r/s) = r \in \mfk{b}\),
    so \(r \in \mfk{b}^c\) and \(r/s \in \mfk{b}^{ce}\).
    But we have \(\mfk{b} \subseteq \mfk{b}^{ce} \subseteq \mfk{b}\)
    (Proposition~\ref{prop:extension-contraction-properties}),
    so \(\mfk{b}\) is the extension of \(\mfk{b}^c\).
\end{proof}
\begin{lemma}\label{lem:fraction-contractions}
    Consider the ring homomorphism \(\func{\phi}{R}{S^{-1}R}\).
    If \(\mfk{a} \subseteq R\),
    then \(\mfk{a}^{ec} = \bigcup_{s \in S} (\mfk{a}:s)\).
    Conversely, \(\mfk{a} = \mfk{b}^c\) is a contracted ideal
    if and only if no \(s \in S\) are zero divisors in \(R/\mfk{a}\).
\end{lemma}
\begin{proof}
    Consider \(r \in \mfk{a}^{ec} = {(S^{-1}\mfk{a})}^c\).
    Then \(r = a/s \in S^{-1}\mfk{a}\) for \(a \in \mfk{a}\) and \(s \in S\).
    By definition, there exists some \(s' \in S\) such that \(s'(rs-a) = 0\),
    which implies \(rss' = as' \in \mfk{a}\),
    and \(r \in (\mfk{a}:s)\) for a specific \(s\).
    The reverse direction holds as every step here is bidirectional.

    Now consider any contracted ideal \(\mfk{a} = \mfk{b}^c\).
    Then by Proposition~\ref{prop:extension-contraction-properties}, \(\mfk{a} = \mfk{a}^{ec}\).
    If \(s \in S\) is a zero divisor,
    then there exists \(r \in R\) such that \(s[r] = [0] \in R/\mfk{a}\),
    i.e.\ \(sr \in \mfk{a}\) for some \(r \notin \mfk{a}\).
    But this tells us that \(s \in S \cap \mfk{a}\),
    so \(\mfk{a}^{ec} = (1)\).
    Clearly \((1)\) is not the only contracted ideal,
    so we have a contradiction.
    Now suppose there are no elements of \(S\) that are zero divisors in \(R/\mfk{a}\).
    Then \(sr \in \mfk{a}\) for any \(s \in S\) implies \(r \in \mfk{a}\).
    But that exactly shows that if \(r \in \mfk{a}^{ec}\),
    then \(r\) is the contraction of some element \(r \in \mfk{a}^e\),
    i.e.\ \(r \in \mfk{a}\) in the first place.
    We have shown \(\mfk{a}^{ec} \subseteq \mfk{a}\),
    which suffices for equality.
\end{proof}
\begin{theorem}\label{thm:fraction-correspondence}
    There is a bijective correspondence between 
    the prime ideals of \(R\) that do not intersect with \(S\),
    and the prime ideals of \(S^{-1}R\).
    In that case, \(\mfk{a} \subseteq R\) corresponds to its extension,
    \(\mfk{b} \subseteq S^{-1}R\) corresponds to its contraction,
    and \(s \in S \subseteq R\) corresponds to \((1) \subseteq S^{-1}R\).
\end{theorem}
\begin{proof}
    We already know that if \(\mfk{b} \subseteq S^{-1}R\) is prime,
    then its contraction is also prime,
    so it suffices to prove that if \(\mfk{a} \subseteq R\) prime, then \(S^{-1}\mfk{a}\) prime.
    Suppose \(\mfk{a} \subseteq R\) prime.
    Then \(R/\mfk{a}\) is an integral domain (Theorem~\ref{thm:prime-quotient-domain}).
    Hence \(S^{-1}(R/\mfk{a}) \cong S^{-1}R/S^{-1}\mfk{a}\) (Proposition~\ref{prop:fraction-properties}),
    and we see that this is an integral domain (or 0)
    as \([rr']/s = 0\) implies \([rr'] = 0\),
    and \(R/\mfk{a}\) being a domain follows.
    We can then conclude that \(S^{-1}\mfk{a} \subseteq S^{-1}R\) is prime
    (Theorem~\ref{thm:prime-quotient-domain}).

    By the \hyperref[lem:fraction-contractions]{lemma above},
    \(\mfk{a}\) intersects with \(S\) if and only if \(\mfk{a}^e = (1)\),
    so we cannot have a bijection only in that case,
    and a bijection is achieved in all other cases.
\end{proof}

\begin{proposition}\label{prop:fraction-ring-properties}
    Suppose \(R\) is a ring, and \(\mfk{a},\mfk{b}\) are ideals.
    Then we have:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(S^{-1}(\mfk{a}+\mfk{b}) = S^{-1}\mfk{a} + S^{-1}\mfk{b}\);
        \item \(S^{-1}(\mfk{ab}) = S^{-1}\mfk{a} S^{-1}\mfk{b}\);
        \item \(S^{-1}(\mfk{a}\cap\mfk{b}) = S^{-1}\mfk{a} \cap S^{-1}\mfk{b}\); and
        \item \(S^{-1}\sqrt{\mfk{a}} = \sqrt{S^{-1}\mfk{a}}\).
    \end{enumerate}
\end{proposition}
\begin{proof}
    (a) and (b) follow directly from Proposition~\ref{prop:extension-contraction-properties}.
    (c) follows from Proposition~\ref{prop:fraction-properties}.
    The forward inclusion of (d) follows from Proposition~\ref{prop:extension-contraction-properties}.

    Suppose \(x^n/s^n \in \sqrt{S^{-1}\mfk{a}}\), where \(x/s \in S^{-1}\mfk{a}\).
    Then clearly \(x \in \mfk{a}\), so \(x^n \in \sqrt{\mfk{a}}\).
    Hence \(x^n/s^n \in S^{-1}\sqrt{\mfk{a}}\).
\end{proof}


\section{Primary Decomposition}

\begin{definition}
    Suppose \(R\) is a ring, and \(\mfk{q} \subsetneq R\) is a proper ideal.
    \(\mfk{q}\) is primary if \(xy \in \mfk{q}\) implies \(x \in \mfk{q}\) or \(y^n \in \mfk{q}\) for some \(n\).
\end{definition}
\begin{proposition}
    Suppose \(R\) is a ring, and \(\mfk{q} \subsetneq R\) is a proper ideal.
    \(\mfk{q}\) is primary if and only if every zero divisor in \(R/\mfk{q}\) is nilpotent.
\end{proposition}
\begin{proof}
    Suppose \(\mfk{q}\) is primary.
    Consider a zero divisor \([y] \in R/\mfk{q}\).
    Then there exists some nonzero \([x] \in R/\mfk{q}\) such that \([x][y] = 0\).
    But this exactly implies \(xy \in \mfk{q}\),
    which implies either \(x \in \mfk{q}\) (which we assumed is not the case),
    or \(y^n \in \mfk{q}\), i.e.\ \([y^n] = 0\).
    
    Suppose every zero divisor \([y] \in R/\mfk{q}\) is nilpotent, with \([y^n] = 0\).
    Then there exists some nonzero \([x] \in R/\mfk{q}\) such that \([x][y] = 0\).
    But this exactly implies \(xy \in \mfk{q}\),
    and \(y^n \in \mfk{q}\) as desired.
\end{proof}
\begin{theorem}
    Suppose \(R\) is a ring, and \(\mfk{q} \subsetneq R\) is a proper ideal.
    If \(\mfk{q}\) is primary, then \(\sqrt{\mfk{q}}\) is prime.
\end{theorem}
\begin{proof}
    If \(xy \in \sqrt{\mfk{q}}\),
    then \({(xy)}^m = x^m y^m \in \mfk{q}\).
    But as \(\mfk{q}\) is primary, this implies either \(x^m \in \mfk{q}\) or \(y^{mn} \in \mfk{q}\).
    By definition of the radical, this implies either \(x \in \sqrt{\mfk{q}}\) or \(y \in \sqrt{\mfk{q}}\).
\end{proof}
\begin{definition}
    Suppose \(\mfk{q}\) is primary.
    We say \(\mfk{q}\) is \(\mfk{p}\)-primary if \(\sqrt{\mfk{q}} = \mfk{p}\).
\end{definition}

\begin{lemma}
    Suppose \({\{\mfk{q}_i\}}_{i=1}^n\) is a family of \(\mfk{p}\)-primary ideals.
    Then \(\bigcap_{i=1}^n \mfk{q}_i\) is also \(\mfk{p}\)-primary.
\end{lemma}
\begin{proof}
    We first check that \(\mfk{q} = \bigcap \mfk{q}_i\) is indeed primary.
    Consider \(xy \in \mfk{q}\).
    If \(x \notin \mfk{q}\), and hence \(x \notin \mfk{q}_i\),
    then we have \(y^{m_i} \in \mfk{q}_i\),
    and hence \(y^{\max m_i} \in \mfk{q}\).

    A quick application of Proposition~\ref{prop:radical-properties} then gives
    \(\sqrt{\bigcap \mfk{q}_i} = \bigcap \sqrt{\mfk{q}_i} = \bigcap \mfk{p} = \mfk{p}\).
\end{proof}
\begin{definition}
    Suppose \(\mfk{q} \subseteq R\) is an ideal.
    If \(\mfk{q} = \bigcap_i \mfk{q}_i\), we say \(\mfk{q}\) is decomposable.
    In this case, we may assume \(\sqrt{\mfk{q}_i} = \mfk{p}_i\) are distinct,
    and hence \(\mfk{q}_i\) should not be inside any \(\mfk{q}_j\).
    We call such a decomposition the minimal or irredundant decomposition.
\end{definition}
\begin{remark}
    Note that in general we cannot guarantee the existence of a primary decomposition.
\end{remark}

\begin{lemma}\label{lem:quotient-primary}
    Suppose \(\mfk{q}\) is \(\mfk{p}\)-primary, and \(x \in R\).
    We have the following:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item if \(x \in \mfk{q}\), then \((\mfk{q}:x) = (1)\); and
        \item if \(x \notin \mfk{q}\), then \((\mfk{q}:x)\) is \(\mfk{p}\)-primary.
    \end{enumerate}
\end{lemma}
\begin{proof}
    Almost by definition, \(xR \subset \mfk{q}\),
    so \((\mfk{q}:x) = R\), and we have statement (a).

    Consider \(yz \in (\mfk{q}:x)\), and without loss of generality \(y \notin (\mfk{q}:x)\);
    we wish to prove that \(z^n \in (\mfk{q}:x)\).
    Now, by definition of the quotient we have \(xyz \in \mfk{q}\),
    and as \(x \notin \mfk{q}\) and \(y \notin \mfk{q}\), \(xy \notin \mfk{q}\),
    which forces us to conclude that \(z^n \in \mfk{q} \subseteq (\mfk{q}:x)\)
    (Proposition~\ref{prop:quotient-ideal-properties}).
    This shows that \((\mfk{q}:x)\) is indeed primary.
    We now show that \(\sqrt{(\mfk{q}:x)} = \mfk{p}\).
    Observe that for any element \(y \in (\mfk{q}:x)\), \(xy \in \mfk{q}\).
    As \(x \notin \mfk{q}\), we have \(y^n \in \mfk{q}\), so \(y \in \sqrt{\mfk{p}} = \mfk{p}\).
    We have exactly shown that \((\mfk{q}:x) \subseteq \mfk{p}\).
    But we have \(\mfk{p} = \sqrt{\mfk{q}} \subseteq \sqrt{(\mfk{q}:x)} \subseteq \sqrt{\mfk{p}} = \mfk{p}\),
    and we have equality.
    This proves statement (b).
\end{proof}
\begin{lemma}\label{lem:prime-intersection}
    Suppose a prime ideal \(\mfk{p} = \bigcap_i \mfk{a}_i\) for a family of ideals.
    Then \(\mfk{p} = \mfk{a}_i\) for some \(i\).
\end{lemma}
\begin{proof}
    Suppose, by way of contradiction, that \(\mfk{p} \subsetneq \mfk{a}_i\) for all \(i\).
    For each \(i\), we choose some \(x_i \in \mfk{a}_i - \mfk{p}\).
    Then we have \(\prod x_i \in \prod \mfk{a}_i \subseteq \bigcap \mfk{a}_i\),
    but as each \(x_i \notin \mfk{p}\), this product \(\prod x_i \notin \mfk{p}\).
    This contradicts our assumed equality,
    and we conclude that \(\mfk{a}_i \subseteq \mfk{p}\).
    The reverse inclusion \(\mfk{p} = \bigcap \mfk{a}_i \subseteq \mfk{a}_i\) is given trivially.
    We have equality.
\end{proof}
\begin{theorem}[First Uniqueness Theorem for Primary Decompositions]\label{thm:primary-uniqueness-1}
    Suppose \(\mfk{q}\) is a decomposable ideal with an irredundant decomposition \(\mfk{q} = \bigcap_i \mfk{q}_i\).
    Then the associated primes \(\mfk{p}_i = \sqrt{\mfk{q}_i}\) are unique up to reordering.
\end{theorem}
\begin{proof}
    Consider the set of ideals \({\{\sqrt{(\mfk{q}:x)}\}}_{x \in R}\).
    We prove that the prime ideals in that set is exactly \({\{\mfk{p}_i\}}_i\).
    In the forward inclusion, suppose we have an arbitrary \(x \in R\).
    If \(x \in \mfk{q}\), then \(\sqrt{(\mfk{q}:x)} = \sqrt{(1)} = (1)\).
    If not, by Lemma~\ref{lem:quotient-primary}, then it is the prime ideal
    \begin{equation*}
        \mfk{p} = \sqrt{(\mfk{q}:x)} = \sqrt{\pqty{\bigcap \mfk{q}_i:x}} = \sqrt{\bigcap (\mfk{q}_i:x)}
        = \bigcap\sqrt{(\mfk{q}_i:x)} = \bigcap_{x \notin \mfk{q}_i} \mfk{p}_i
    \end{equation*}
    using Propositions~\ref{prop:radical-properties}~\&~\ref{prop:quotient-ideal-properties}.
    From the \hyperref[lem:prime-intersection]{lemma above},
    we know that \(\mfk{p} = \mfk{p}_i\) for some \(i\).
    Now for the reverse inclusion, consider some \(x \in \bigcap_{j \neq i} \mfk{q}_j - \mfk{q}_i\).
    Then \(x \notin \mfk{q}_i\) and \(x \in \mfk{q}_j\) for every other \(j \neq i\),
    so we have
    \begin{equation*}
        \sqrt{(\mfk{q}:x)} = \sqrt{\pqty{\bigcap \mfk{q}_i:x}} = \sqrt{\bigcap (\mfk{q}_i:x)}
        = \bigcap\sqrt{(\mfk{q}_i:x)} = \bigcap_{x \notin \mfk{q}_j} \mfk{p}_j = \mfk{p}_i
    \end{equation*}
    Then we immediately have that the associated primes are uniquely determined by \(\mfk{q}\).
\end{proof}

\begin{proposition}\label{prop:fraction-primary}
    Suppose \(S \subseteq R\) is a multiplicative monoid,
    and \(\mfk{q}\) is a \(\mfk{p}\)-primary ideal.
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item If \(S\cap\mfk{p} \neq \emptyset\), then \(S^{-1}\mfk{q} = S^{-1}R\); and
        \item if \(S\cap\mfk{p} = \emptyset\), then \(S^{-1}\mfk{q}\) is \(S^{-1}\mfk{p}\)-primary,
            and \(S^{-1}\mfk{q}\) contracts to \(\mfk{q}\).
    \end{enumerate}
\end{proposition}
\begin{proof}
    Consider \(s \in S\cap\mfk{p}\).
    We have some \(s^n \in S\cap\mfk{q}\) due to \(\sqrt{\mfk{q}} = \mfk{p}\).
    Then this element in the extension is \(s^n \in S^{-1}\mfk{q}\),
    so \(s^n/s^n = 1 \in S^{-1}\mfk{q}\),
    showing that \(S^{-1}\mfk{q} = S^{-1}R\).
    This proves statement (a).

    If \(S\cap\mfk{p} = \emptyset\), consider any \(s \in S\).
    Given any \(r\), if \(rs \in \mfk{q}\),
    as \(\mfk{q}\) is primary and \(s^n \notin \mfk{q}\),
    we have \(r \in \mfk{q}\).
    Therefore, by definition of the quotient ideal, \((\mfk{q}:s) \subseteq \mfk{q}\)
    By Lemma~\ref{lem:fraction-contractions},
    \(\mfk{q}^{ec} = \bigcup_{s \in S} (\mfk{q}:s) \subseteq \mfk{q}\),
    and the reverse inclusion is trvially given by Proposition~\ref{prop:extension-contraction-properties};
    we have \(\mfk{q}^{ec} = {(S^{-1}\mfk{q})}^c = \mfk{q}\).

    We can immediately see that \(\sqrt{S^{-1}\mfk{q}} = S^{-1}\sqrt{\mfk{q}} = S^{-1}\mfk{p}\)
    by an application of Proposition~\ref{prop:fraction-ring-properties}.
    We now verify that \(S^{-1}\mfk{q}\) is primary.
    Suppose \((x/t)(y/s) \in S^{-1}\mfk{q}\), and \(x/t \notin S^{-1}\mfk{q}\).
    Then it is easy to see that \(x \notin \mfk{q}\),
    so \(y^n \in \mfk{q}\) for some \(n\).
    Hence \(y^n/s^n \in S^{-1}\mfk{q}\).
    This proves statement (b).
\end{proof}
\begin{proposition}\label{prop:fraction-decomposition}
    Suppose \(S \subseteq R\) is a multiplicative monoid,
    and \(\mfk{q} = \bigcap_i \mfk{q}_i\) has an irredundant decomposition,
    with \(\mfk{p}_i = \sqrt{\mfk{q}_i}\).
    Without loss of generality, suppose \(S \cap \mfk{p}_i = \emptyset\) for the first \(n\) ideals,
    and \(S \cap \mfk{p}_i \neq \emptyset\) for the rest of the ideals.
    Then we an have irredundant decomposition for \(S^{-1}\mfk{q} = \bigcap_{i=1}^n S^{-1}\mfk{q}_i\),
    and for its contraction \({(S^{-1}\mfk{q})}^c = \bigcap_{i=1}^n \mfk{q}_i\).
\end{proposition}
\begin{proof}
    By Proposition~\ref{prop:fraction-ring-properties},
    \(S^{-1}\mfk{q} = S^{-1}\pqty{\bigcap_i \mfk{q}_i} = \bigcap_i S^{-1}\mfk{q}_i\),
    and we know we can toss out everything but the first \(n\) ideals
    because of the \hyperref[prop:fraction-primary]{proposition above}.
    This is indeed a primary decomposition,
    as we know that \(S^{-1}\mfk{q}_i\) are all \(S^{-1}\mfk{p}_i\)-primary
    again from the \hyperref[prop:fraction-primary]{proposition above}.
    This decomposition is also irredundant because \(\mfk{p}_i\) are distinct,
    and therefore \(S^{-1}\mfk{p}_i\) are also distinct.

    We can then take the contraction of everything to get
    \({(S^{-1}\mfk{q})}^c = {(\bigcap_{i=1}^n S^{-1}\mfk{q}_i)}^c
    = \bigcap_{i=1}^n {(S^{-1}\mfk{q}_i)}^c = \bigcap_{i=1}^n \mfk{q}_i\),
    simply by using Proposition~\ref{prop:extension-contraction-operations}
    and the \hyperref[prop:fraction-primary]{proposition above}.
\end{proof}
\begin{definition}
    Suppose \(\mfk{q}\) is a decomposable ideal,
    and \({\{\mfk{p}_i\}}_{i \in S}\) is a subset of the associated prime ideals.
    This set of ideals is isolated if any associated prime \(\mfk{p} \subseteq \mfk{p}_i\)
    is also in this subset \(\mfk{p} \in {\{\mfk{p}_i\}}_{i \in S}\).
\end{definition}
\begin{theorem}[Second Uniqueness Theorem for Primary Decompositions]\label{thm:primary-uniqueness-2}
    Suppose \(\mfk{q}\) is a decomposable ideal with an irredundant decomposition \(\mfk{q} = \bigcap_i \mfk{q}_i\).
    Let \({\{\mfk{p}_{i_k}\}}_{k \in S}\) be an isolated set of associated primes.
    Then \(\bigcap_{i_k} \mfk{q}_{i_k}\) is uniquely determined independent of the decomposition.
\end{theorem}
\begin{proof}
    Consider an ideal \(\mfk{q}\)
    with an irredundant decomposition \(\mfk{q} = \bigcap_i \mfk{q}_i\),
    and that the first \(n\) of the associated primes \(\sqrt{\mfk{q}_i} = \mfk{p}_i\) are isolated.
    Take \(S = R - \bigcup_{i=1}^n \mfk{p}_i\),
    i.e.\ the localization at \(\bigcup_{i=1}^n \mfk{p}_i\).
    Then trivially, for these \(n\) ideals, \(\mfk{p}_i \cap S = \emptyset\),
    and the other ideals, \(\mfk{p}_i \cap S \neq \emptyset\).
    Then immediately from the \hyperref[prop:fraction-decomposition]{proposition above},
    \({(S^{-1}\mfk{q})}^c = \bigcap_{i=1}^n \mfk{q}_i\).
    
    Now suppose \(\mfk{q}\) has a different irredundant decomposition
    \(\mfk{q} = \bigcap_i \mfk{q}'_i\),
    with the same associated primes \(\sqrt{\mfk{q}'_i} = \mfk{p}_i\)
    as determined by the \hyperref[thm:primary-uniqueness-1]{first uniqueness theorem}.
    Then the \hyperref[prop:fraction-decomposition]{proposition above} yields
    \({(S^{-1}\mfk{q})}^c = \bigcap_{i=1}^n \mfk{q}_i = \bigcap_{i=1}^n \mfk{q}'_i\).
\end{proof}
\begin{corollary}
    Suppose \(\mfk{q}\) is a decomposable ideal with an irredundant decomposition \(\mfk{q} = \bigcap_i \mfk{q}_i\),
    and \(\mfk{p}_i\) be a minimal associated prime,
    as in there are no other associated prime ideals \(\mfk{p} \subseteq \mfk{p}_i\).
    Then \(\mfk{q}_i\) is uniquely determined by \(\mfk{q}\).
\end{corollary}
\begin{proof}
    Simply take \(\{\mfk{p}_i\}\) as an isolated set of ideals,
    and apply the \hyperref[thm:primary-uniqueness-2]{theorem above}.
\end{proof}


\section{Integral Dependence}

\begin{definition}
    Suppose \(A \subseteq R\), where \(R\) is an \(A\)-algebra.
    \(R\) is finite over \(A\) if \(R\) is a finitely generated \(A\)-module;
    that is, there is a surjection \(\bigoplus_{i=1}^n x_i A \to R\).
    \(R\) is finite-type over \(A\) if \(R\) is a finitely generated \(A\)-algebra;
    that is, there is a surjection \(A[x_1,\hdots,x_n] \to R\).
\end{definition}
\begin{definition}
    Suppose \(R\) is an \(A\)-algebra, and \(x \in R\).
    We say \(x\) is integral over \(A\) if it satisfies a monic polynomial
    \(x^n + a_1 x^{n-1} + \cdots + a_n = 0\) for \(a_i \in A\).
    We say \(x\) is algebraic over \(A\) if it satisfies a polynomial
    \(a_0 x^n + a_1 x^{n-1} + \cdots + a_n = 0\) for \(a_i \in A\).
\end{definition}
\begin{definition}
    \(A \subseteq R\) is integral if every \(x \in R\) is integral over \(A\).
\end{definition}
\begin{lemma}\label{lem:integral-finite}
    Suppose \(R\) is an \(A\)-algebra.
    \(x \in R\) is integral over \(A\) if and only if \(A[x]\) is finite over \(A\).
\end{lemma}
\begin{proof}
    Suppose \(x\) is integral over \(A\),
    with \(x^n + a_1x^{n-1} + \cdots + a_n = 0\).
    Then any element \(f \in A[x]\) can be written as a polynomial of degree at most \(n-1\) over \(x\),
    as any \(x^n\) factors can be reduced to a polynomial of degree at most \(n-1\).

    Suppose \(A[x]\) is finite over \(A\).
    Then consider the map \(\vfunc{\phi}{A[x]}{A[x]}{f}{xf}\) that multiplies by \(x\).
    By the \hyperref[lem:cayley-hamilton]{Cayley-Hamilton Theorem},
    \(0 = \phi^n + a_1\phi^{n-1} + \cdots + a_n = x^n + a_1 x^{n-1} + \cdots + a_n\) as desired.
\end{proof}
\begin{corollary}
    Suppose \(R\) is an \(A\)-algebra.
    If \(x \in R\) is integral over \(A\),
    then \(A[x]\) is integral over \(A\).
    Hence finite extensions are integral.
\end{corollary}
\begin{proof}
    Consider any element \(y \in A[x]\),
    and the map \(\vfunc{\phi}{A[x]}{A[x]}{f}{yf}\) that multiplies by \(y\).
    By the \hyperref[lem:cayley-hamilton]{Cayley-Hamilton Theorem},
    \(0 = \phi^n + a_1\phi^{n-1} + \cdots + a_n = y^n + a_1 y^{n-1} + \cdots + a_n\) as desired.
\end{proof}
\begin{corollary}\label{cor:finite-type-integral}
    Suppose \(R\) is an \(A\)-algebra,
    and \({\{x_i\}}_{i=1}^n \subset R\) are all integral over \(A\).
    Then \(A[x_1,\hdots,x_n]\) is finite over \(A\).
    Hence extensions are finite if and only if they are both finite-type and integral.
\end{corollary}
\begin{proof}
    Let the \hyperref[lem:integral-finite]{lemma above} be the base case,
    and proceed by induction.
    Let \(A_k = A[x_1,\hdots,x_k]\), and assume that it is finite over \(A\).
    We wish to prove that \(A_{k+1} = A_k[x_{k+1}]\) is finite over \(A\).
    But because \(x_{k+1}\) is integral over \(A\), it is integral over \(A_k\),
    so by the \hyperref[lem:integral-finite]{lemma above},
    \(A_{k+1}\) is finite over \(A_k\), which is then finite over \(A\).
\end{proof}

\begin{definition}
    Suppose \(R\) is an \(A\)-algebra.
    The integral closure of \(A\) in \(R\) is an intermediate ring \(\overline{A}\)
    that consists of all elements in \(R\) that are integral over \(A\).
\end{definition}
\begin{proposition}
    Suppose \(R\) is an \(A\)-algebra.
    \(\overline{A} \subseteq R\) indeed forms a ring.
\end{proposition}
\begin{proof}
    Given any \(x,y \in R\) that is integral over \(A\),
    we need to show that \(x+y\) and \(xy\) are also integral over \(A\).
    But we know from the \hyperref[cor:finite-type-integral]{corollary above}
    that since \(x,y \in R\) are integral over \(A\),
    then \(A[x,y]\) must be finite over \(A\).
    But then \(A \subseteq A[x+y] \subseteq A[x,y]\) are all finite over \(A\),
    so by Lemma~\ref{lem:integral-finite}, \(x+y\) is integral over \(A\).
    Similarly, \(A \subseteq A[xy] \subseteq A[x,y]\) are all finite over \(A\),
    so \(xy\) is integral over \(A\).
\end{proof}
\begin{lemma}\label{lem:integral-stack}
    Suppose \(A \subseteq B \subseteq R\) are all rings.
    If \(A \subseteq B\) and \(B \subseteq R\) are both integral,
    then \(A \subseteq R\) is also integral.
\end{lemma}
\begin{proof}
    Consider any \(x \in R\).
    Since it is integral over \(B\), there exists \(b_i \in B\) such that
    \(x^n + b_1x^{n-1} + \cdots + b_n = 0\).
    But as each of these \(b_i\) are integral over \(A\),
    \(A \subseteq A[b_1,\hdots,b_n] \subseteq A[b_1,\hdots,b_n][x]\) are all finite,
    so by Lemma~\ref{lem:integral-finite}, \(x\) is integral over \(A\).
\end{proof}
\begin{corollary}
    Suppose \(A \subseteq R\) is an extension of rings.
    Then \(\overline{\overline{A}} = \overline{A}\).
\end{corollary}
\begin{proof}
    Because \(A \subseteq \overline{A} \subseteq \overline{\overline{A}}\) are all integral,
    by the \hyperref[lem:integral-stack]{lemma above},
    \(A \subseteq \overline{\overline{A}}\) is integral,
    and any integral extension is contained inside the integral closure,
    giving us \(\overline{\overline{A}} \subseteq \overline{A}\).
\end{proof}

\begin{lemma}\label{lem:integral-ideal-extension}
    Suppose \(A \subseteq R\) is an integral extension of rings.
    Then we have the following two integral extensions:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item if \(\mfk{b} \subseteq R\) is an ideal, and \(\mfk{a} = \mfk{b} \cap A\),
            then \(A/\mfk{a} \subseteq R/\mfk{b}\) is integral; and
        \item if \(S \subseteq A\) is a multiplicative monoid,
            then \(S^{-1}A \subseteq S^{-1}R\) is integral.
    \end{enumerate}
\end{lemma}
\begin{proof}
    Consider \(x \in R\).
    Since it is integral over \(A\),
    there exists \(a_i \in A\) such that \(x^n + a_1x^{n-1} + \cdots + a_n = 0\).
    Taking the entire equation modulo \(\mfk{b}\),
    we can see that \([x] \in R/\mfk{b}\) is integral over \(A/(\mfk{b} \cap A)\) as desired.
    This gives us statement (a).

    Consider \(x/s \in S^{-1}R\).
    The equation above, divided by \(s^n\),
    gives \({(x/s)}^n + (a_1/s){(x/s)}^{n-1} + \cdots + a_n/s^n = 0\).
    Observe that \(a_i/s^i \in S^{-1}A\),
    so \(x/s\) is integral over \(S^{-1}A\) as desired,
    giving us statement (b).
\end{proof}
\begin{theorem}\label{thm:integral-field-extension}
    Suppose \(A \subseteq R\) is an integral extension of rings,
    and \(A,R\) are both integral domains.
    Then \(A\) is a field if and only if \(R\) is a field.
\end{theorem}
\begin{proof}
    Suppose \(A\) is a field.
    We know from \hyperref[sec:algebraic-extensions]{field theory}
    that any algebraic extension (elements that satisfy a monic polynomial)
    \(A \subseteq R\) is a field.

    Suppose \(R\) is a field.
    Consider \(x \in A\).
    Because \(x^{-1} \in R\), it is integral over \(A\),
    so there exists \(a_i \in A\) such that
    \(x^{-n} + a_1 x^{-n+1} + \cdots + a_n = 0\).
    Multiplying by \(x^{n-1}\),
    we get \(x^{-1} + a_1 + \cdots + a_n x^{n-1} = 0\),
    so \(x^{-1} = - a_1 - \cdots - a_n x^{n-1} \in A\).
\end{proof}
\begin{corollary}\label{cor:integral-maximal-ideal}
    Suppose \(A \subseteq R\) is an integral extension of rings.
    If \(\mfk{q} \subseteq R\) is prime, then \(\mfk{p} = \mfk{q} \cap A\) is also prime,
    and \(\mfk{p}\) is maximal if and only if \(\mfk{q}\) is maximal.
\end{corollary}
\begin{proof}
    Observe that by the \hyperref[lem:integral-ideal-extension]{lemma above},
    \(A/\mfk{p} \subseteq R/\mfk{q}\) is an integral extension,
    and according to the \hyperref[thm:integral-field-extension]{theorem above},
    they are fields if and only if the other one is.
    Hence the ideals are maximal if and only if the other one is.

    The preimage of a prime ideal is always prime, so we are done.
\end{proof}
\begin{theorem}\label{thm:integral-prime-ideal}
    Suppose \(A \subseteq R\) is an integral extension of rings,
    and \(\mfk{p} \subseteq A\) is prime.
    Then there exists some prime \(\mfk{q} \subseteq R\)
    such that \(\mfk{p} = \mfk{q} \cap A\).
\end{theorem}
\begin{proof}
    By Lemma~\ref{lem:integral-ideal-extension},
    \(A_\mfk{p} \subseteq R_\mfk{p}\) is integral.
    We claim that we have the commutative diagram
    \begin{equation*}
        \begin{tikzcd}
            A \arrow{r}{\iota} \arrow{d}{\alpha} & R \arrow{d}{\beta} \\
            A_\mfk{p} \arrow{r}{\iota} & R_\mfk{p}
        \end{tikzcd}
    \end{equation*}
    This is obvious, because both \(\iota\) are inclusions (identity),
    and \(\alpha,\beta\) are also natural injections (identity).

    Consider the maximal ideal \(\mfk{n} \subseteq R_\mfk{p}\).
    Then as \(A_\mfk{p}\) is local, \(\mfk{m} = \mfk{n} \cap A\) is the unique maximal ideal.
    But by Proposition~\ref{prop:prime-local-ring},
    the unique maximal ideal is \(\mfk{m} = \mfk{p}A_\mfk{p}\).

    We now claim that \(\mfk{p} = \alpha^{-1}(\mfk{m})\).
    Clearly elements in \(\mfk{m} = \mfk{p}A_\mfk{p}\) must have the form \(p/s\),
    so its preimage must be in the form \(p/1\) as desired.

    Lastly, let \(\mfk{q} = \beta^{-1}(\mfk{n})\).
    This is prime because the preimage of a prime ideal is prime.
    By commutativity of the diagram, we know that \(\mfk{q} \cap A = \mfk{p}\).
\end{proof}
\begin{theorem}[Going Up Theorem]
    Suppose \(A \subseteq R\) is an integral extension of rings,
    \(\mfk{p}_1 \subseteq \mfk{p}_2 \subseteq A\) and \(\mfk{q}_1 \subseteq R\) are prime ideals,
    with \(\mfk{p}_1 = \mfk{q}_1 \cap A\).
    Then there exists a prime ideal \(\mfk{q}_2 \subseteq R\)
    such that \(\mfk{q}_1 \subseteq \mfk{q}_2\) and \(\mfk{p}_2 = \mfk{q}_2 \cap A\).
\end{theorem}
\begin{proof}
    Consider the two rings \(A' = A/\mfk{p}_1\) and \(R' = R/\mfk{q}_1\).
    By Lemma~\ref{lem:integral-ideal-extension}, \(A' \subseteq B'\) is integral.
    Then from the \hyperref[thm:iso-3-ring]{third isomorphism theorem},
    \(\mfk{p}_2 \subseteq A\) corresponds to a prime ideal \(\mfk{p}_2' \subseteq A'\).
    The \hyperref[thm:integral-prime-ideal]{theorem above} guarantees
    the existence of a prime ideal \(\mfk{q}_2' \subseteq R'\)
    such that \(\mfk{p}_2' = \mfk{q}_2' \cap A'\),
    which another application of \hyperref[thm:iso-3-ring]{third isomorphism theorem}
    gives us a correspondence to \(\mfk{q}_2 \subseteq R\).
\end{proof}

\begin{lemma}\label{lem:ideal-integral-closure}
    Suppose \(R\) is an \(A\)-algebra, and \(\overline{A}\) is the integral closure of \(A\) in \(R\).
    Let \(\mfk{a} \subseteq A\) be an ideal, and \(\mfk{a}^e \subseteq \overline{A}\) be its extension.
    Then the integral closure of \(\mfk{a}\) in \(R\) is \(\sqrt{\mfk{a}^e}\).
\end{lemma}
\begin{proof}
    Consider any integral element \(x \in \overline{\mfk{a}} \subseteq R\).
    Then there exists \(a_i \in \mfk{a} \subseteq A\) such that
    \(x^n + a_1 x^{n-1} + \cdots + a_n = 0\).
    Clearly \(x\) is also integral over \(A\), so \(x \in \overline{A}\);
    moreover, \(x^n = - a_1 x^{n-1} - \cdots - a_n \in \mfk{a}^e\),
    as \(a_i x^{n-i} \in \mfk{a}^e\overline{A} \subseteq \mfk{a}^e\).
    But this exactly says that \(x \in \sqrt{\mfk{a}^e}\),
    so \(\overline{\mfk{a}} \subseteq \sqrt{\mfk{a}^e}\).

    For the reverse direction, suppose \(x \in \sqrt{\mfk{a}^e}\).
    Then \(x^n = \sum a_i x_i \in \mfk{a}^e\), with \(a_i \in \mfk{a}\) and \(x_i \in \overline{A}\).
    By definition, \(x_i\) is integral over \(A\),
    so \(M = A[x_1,\hdots,x_k]\) is finite over \(A\) (Corollary~\ref{cor:finite-type-integral}).
    In particular, \(x^n M \subseteq \mfk{a}M\) by definition,
    so if \(\phi\) is the morphism that multiplies by \(x^n\),
    the \hyperref[lem:cayley-hamilton]{Cayley-Hamilton theorem} guarantees a monic polynomial
    \(\phi^m + a_1\phi^{m-1} + \cdots + a_m = x^{nm} + a_1 x^{n(m-1)} + \cdots + a_m = 0\),
    which gives us that \(x\) is integral over \(\mfk{a}\).
    Hence \(\sqrt{\mfk{a}^e} \subseteq \overline{\mfk{a}}\).
\end{proof}
\begin{proposition}\label{prop:integral-minimal-polynomial}
    Suppose \(A = \overline{A} \subseteq R\) are integral domains,
    and let \(x \in R\) be integral over some ideal \(\mfk{a} \subseteq A\).
    Then \(x\) is algebraic over \(K\), the field of fractions of \(A\);
    moreover, if the minimal polynomial is \(\min_K(x) = t^n + a_1 t^{n-1} + \cdots + a_n\),
    then \(a_i \in \sqrt{\mfk{a}}\).
\end{proposition}
\begin{proof}
    Since \(x\) is integral over \(\mfk{a}\),
    there exists \(a_i \in \mfk{a} \subseteq A \subseteq K\) such that
    \(x^n + a_1 x^{n-1} + \cdots + a_n = 0\),
    so \(x\) is algebraic over \(K\).
    Notice the requirement of an integral domain is needed to invoke the field of fractions.

    Now let \({\{x_i\}}\) be the set of roots of \(\min_K(x)\).
    Then immediately we see that \(x_i\) are all integral over \(\mfk{a}\),
    and as \(\min_K(x) = \prod {(t-x_i)}^{k_i} = t^n + a_1 t^{n-1} + \cdots + a_n\),
    each \(a_k\) can be written as a polynomial in \(x_i\)
    and is therefore also integral over \(\mfk{a}\).
    But then by the \hyperref[lem:ideal-integral-closure]{lemma above},
    \(a_k \in \sqrt{\mfk{a}^e} = \sqrt{\mfk{a}}\) as \(A = \overline{A}\).
\end{proof}
\begin{theorem}[Going Down Theorem]
    Suppose \(A = \overline{A} \subseteq R\) is an integral extension of integral domains,
    \(\mfk{p}_1 \subseteq \mfk{p}_2 \subseteq A\) and \(\mfk{q}_2 \subseteq R\) are prime ideals,
    with \(\mfk{p}_2 = \mfk{q}_2 \cap A\).
    Then there exists a prime ideal \(\mfk{q}_1 \subseteq R\)
    such that \(\mfk{q}_1 \subseteq \mfk{q}_2\) and \(\mfk{p}_1 = \mfk{q}_1 \cap A\).
\end{theorem}
\begin{proof}
    Consider the natural embedding \(A \to R \to S^{-1}R = R_{\mfk{q}_2}\).
    It suffices to show that \(\mfk{p}_1\)
    is the contraction of some prime ideal \(\mfk{q}_1' \subseteq R_{\mfk{q}_2}\),
    because Theorem~\ref{thm:fraction-correspondence} dictates that
    \(\mfk{q}_1'\) will be in correspondence with some prime ideal \(\mfk{q}_1 \subseteq R\)
    that does not intersect \(S = R - \mfk{q}_2\), i.e.\ \(\mfk{q}_1 \subseteq \mfk{q}_2\).
    Then since \(\mfk{q}_1' = S^{-1}\mfk{q}_1\) (Lemma~\ref{lem:fraction-extensions}),
    \(\mfk{p}_1 = S^{-1}\mfk{q}_1 \cap A = \mfk{q}_1 \cap A\).

    We claim that \(\mfk{q}_1' = \mfk{p}_1R_{\mfk{q}_2}\).
    Consider any element \(x/s \in \mfk{p}_1R_{\mfk{q}_2}\),
    where \(x \in \mfk{p}_1R\) and \(s \in R - \mfk{q}_2\).
    Notice \(\mfk{p}_1R\) is a prime (and therefore radical) ideal,
    so \(x\) is integral over \(\mfk{p}_1\) by the \hyperref[lem:ideal-integral-closure]{lemma above};
    moreover, if \(K\) is the field of fractions of \(A\),
    by the \hyperref[prop:integral-minimal-polynomial]{proposition above},
    \(\min_K(x) = x^r + u_1 x^{r-1} + \cdots + u_r\), where \(u_i \in \mfk{p}_1\).
    Now, if \(x/s \in A\) also, then \(s = x(s/x)\), and \(s/x \in K\),
    and \(\min_K(s) = s^r + v_1 s^{r-1} + \cdots + v_r\) is obtained
    by dividing the previous polynomial by \({(x/s)}^r\), where \(v_i = u_i{(s/x)}^i\).
    We note that \(v_i{(x/s)}^i = u_i \in \mfk{p}_1\).
    But because \(x/s \in A\), \(s = x/a\) for some \(a\) must also be integral over \(A\),
    and we can apply the \hyperref[prop:integral-minimal-polynomial]{proposition above} again
    to see that \(v_i \in A\).

    Let us now suppose, by way of contradiction, that \(x/s \notin \mfk{p}_1\).
    Then \(v_i{(x/s)}^i \in \mfk{p}_1\) implies that \(v_i \in \mfk{p}_1\),
    and the minimal polynomial tells us that
    \(s^r = - v_1 s^{r-1} - \cdots - v_r \in \mfk{p}_1R \subseteq \mfk{p}_2R \subseteq \mfk{q}_2\).
    But because \(\mfk{q}_2\) is prime (and therefore radical),
    this implies \(s \in \mfk{q}_2\), which contradicts the definition of \(S = R - \mfk{q}_2\).
    Hence \(\mfk{q}_1' \subseteq \mfk{p}_1\),
    and the reverse inclusion is trivially true.
\end{proof}

\begin{proposition}\label{prop:finite-type-algebraic}
    Suppose \(K(x_1,\hdots,x_n)/K\) is a purely transcendental extension.
    Then \(K(x_1,\hdots,x_n)\) cannot be finite-type over \(K\).
\end{proposition}
\begin{proof}
    Let \(E = K(x_1,\hdots,x_n)\).
    Let us note that this is the fraction field of \(R = K[x_1,\hdots,x_n]\).
    Suppose, by way of contradiction, that \(E/K\) finite-type.
    Then it must be generated by a finite set \({\{f_i/g_i\}}_{i=1}^m \subset E\),
    where \(f_i,g_i \in R\) and \(g_i \neq 0\).
    Then any element \(E\) can be written as a sum
    \begin{equation*}
        \sum_{i_1,\hdots,i_m} \alpha_{i_1 \hdots i_m}
        \pqty{\frac{f_1}{g_1}}^{i_1}\hdots\pqty{\frac{f_m}{g_m}}^{i_m}
        = \frac{F}{g_1^{N_1} \hdots g_m^{N_m}}
    \end{equation*}
    after finding the common denominator and pushing the sum of numerators into \(F\).

    We now claim that \(R\) has infinitely many irreducible elements.
    Euclid's classic proof on the infinitude of primes in \(\bZ\) holds here,
    and we shall imitate it.
    Suppose, by way of contradiction,
    that \({\{p_i\}}_{i=1}^n \subset R\) are all the irreducible elements in \(R\).
    Consider the element \(f = p_1 p_2 \hdots p_n + 1\).
    Since \(R = K[x_1,\hdots,x_n]\) is a UFD (Theorem~\ref{thm:polynomial-ring-ufd}),
    \(f\) has a unique decomposition into irreducible primes,
    and let \(p\) be one such prime in this decomposition.
    Clearly \(p \neq p_i\), because if it did, \(p\) would also divide 1.
    We have reached a contradiction.

    Going back to the elements in \(E\),
    we in particular note that for any polynomial \(h \in R\), \(1/h \in E\).
    But the decomposition above tells us that \(h\) can always be decomposed
    into the primes associated with \({\{g_i\}}_{i=1}^m\),
    which notably, of which there are only finitely many.
    This exactly contradicts our claim of infinitude of primes.
\end{proof}
\begin{theorem}\label{thm:finite-type-algebraic}
    Suppose \(K\) is a field, and \(E\) a finitely generated \(K\)-algebra.
    If \(E\) is a field, then \(E/K\) is a finite algebraic extension.
\end{theorem}
\begin{proof}
    Let us first take the \hyperref[thm:noetherian-normalization]{Noetherian normalization}
    to get \(K \subseteq K[x_1,\hdots,x_n] \subseteq E\),
    where \(K[x_1,\hdots,x_n] \subseteq E\) is an integral ring extension,
    i.e.\ every element in \(E\) satisfies a monic polynomial with coefficients in \(K[x_1,\hdots,x_n]\).
    Taking its field of fractions,
    we can see that \(K(x_1,\hdots,x_n) \subseteq E\) is an algebraic field extension,
    because said monic polynomial that \(e \in E\) satisfies
    has coefficients in \(K[x_1,\hdots,x_n] \subseteq K(x_1,\hdots,x_n)\).
    Notably, algebraic field extensions are finite.

    Theorem~\ref{thm:noetherian-finite-type} then establishes that
    as \(K \subseteq E\) is finite-type and \(K(x_1,\hdots,x_n) \subseteq E\) is finite,
    \(K \subseteq K(x_1,\hdots,x_n)\) must also be finite-type.
    But because it is finite-type,
    \hyperref[prop:finite-type-algebraic]{proposition above}
    it cannot be a purely transcendental extension,
    and hence it must be finite algebraic.
\end{proof}

\begin{definition}
    Suppose \(K\) is a field, and \(R = K[x_1,\hdots,x_n]\).
    Consider \({\{f_i\}}_{i \in S} \subset R\) a family of polynomial functions,
    \(\func{f_i}{K^n}{K}\).
    The vanishing locus is the set of points
    \(\Van(\{f_i\}) = \{p \in K^n: \forall i \in S,\,f_i(p) = 0\}\).
    Now consider \(X \subseteq K^n\).
    The ideal of all vanishing polynomials is
    \(\Ide(X) = \{f \in R: \forall x \in X,\,f(x) = 0\}\).
\end{definition}
\begin{definition}
    A subset \(Z \subseteq K^n\) is said to be algebraic
    if it is the vanishing locus \(Z = \Van(S)\)
    of a set \(S \subseteq K[x_1,\hdots,x_n]\).
\end{definition}
\begin{lemma}\label{lem:iv-inclusion-reversing}
    Suppose \(S \subseteq K^n\) and \(J \subseteq K[x_1,\hdots,x_n]\) are subsets.
    Then we have the following properties:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(\Van\) and \(\Ide\) are inclusion reversing; and
        \item \(J \subseteq \Ide(\Van(J))\) and \(S \subseteq \Van(\Ide(S))\).
    \end{enumerate}
\end{lemma}
\begin{proof}
    Suppose \(J' \subseteq J\).
    Consider any point \(p \in \Van(J)\).
    Since all polynomials \(f \in J\) have \(f(p) = 0\),
    in particular any \(f \in J'\) have \(f(p) = 0\),
    so \(p \in \Van(J')\).
    We have \(\Van(J') \supseteq \Van(J)\).

    Suppose \(S' \subseteq S\).
    Consider any \(f \in \Ide(S)\).
    Then \(f\) vanishes on all points in \(S\),
    and in particular it vanishes on all points in \(S'\),
    so \(f \in \Ide(S')\).
    We have \(\Ide(S') \supseteq \Ide(S)\).
    This proves statement (a).

    \medskip

    Now suppose \(f \in J\).
    Then \(f\) vanishes on \(\Van(J)\).
    Since \(\Ide(\Van(J))\) is the set of functions that vanish on \(\Van(J)\),
    we have \(f \in \Ide(\Van(J))\).

    Suppose \(x \in S\).
    Then all functions in \(\Ide(S)\) vanish at \(x\).
    Since \(\Van(\Ide(S))\) is the set of points that all functions in \(\Ide(S)\) vanish at,
    we have \(x \in \Van(\Ide(S))\).
\end{proof}
\begin{lemma}\label{lem:vix-x}
    Suppose \(X \subseteq K^n\) is an algebraic subset.
    Then \(\Van(\Ide(X)) = X\).
\end{lemma}
\begin{proof}
    Trivially by Lemma~\ref{lem:iv-inclusion-reversing},
    \(X \subseteq \Van(\Ide(X))\).

    By definition of an algebraic subset,
    \(X = \Van(\mfk{a})\) for some ideal \(\mfk{a} \subseteq K[x_1,\hdots,x_n]\).
    But then \(\mfk{a} \subseteq \Ide(\Van(\mfk{a}))\),
    so \(\Van(\mfk{a}) \supseteq \Van(\Ide(\Van(\mfk{a})))\).
\end{proof}
\begin{lemma}\label{lem:ix-radical}
    Suppose \(X \subseteq K^n\) is a subset.
    Then \(\Ide(X)\) is a radical ideal.
\end{lemma}
\begin{proof}
    Suppose \(f^n \in \Ide(X)\).
    Then for all \(x \in X\), \(f^n(x) = {(f(x))}^n = 0\),
    which implies \(f(x) = 0\), so \(f \in \Ide(X)\).
\end{proof}

\begin{lemma}[Weak Hilbert's Nullstellensatz]\label{lem:weak-nullstellensatz}
    Suppose \(K\) is algebraically closed.
    Every maximal ideal \(\mfk{m} \subseteq K[x_1,\hdots,x_n]\) is of the form
    \(\mfk{m} = (x_1-a_1,\hdots,x_n-a_n)\) for some \((a_1,\hdots,a_n) \in K^n\).
    Every proper ideal \(\mfk{a} \subsetneq K[x_1,\hdots,x_n]\)
    has a nonempty vanishing locus \(\Van(\mfk{a}) \neq \emptyset\).
\end{lemma}
\begin{proof}
    From our previous statement of the \hyperref[cor:weak-nullstellensatz]{weak Nullstellensatz},
    \(K = K[x_1,\hdots,x_n]/\mfk{m}\).
    Hence there exists a surjecive morphism \(\func{\phi}{K[x_1,\hdots,x_n]}{K}\)
    with \(\ker\phi = \mfk{m}\) by the \hyperref[thm:iso-1-ring]{first isomorphism thoerem}.
    Then if we let \(\phi(x_i) = a_i\),
    then we can see that \(x_i-a_i \in \ker\phi = \mfk{m}\),
    so \((x_1-a_1,\hdots,x_n-a_n) \subseteq \mfk{m}\).
    For the reverse inclusion, we can see that \((x_1-a_1,\hdots,x_n-a_n)\) is a maximal ideal,
    because \(K[x_1,\hdots,x_n]/(x_1-a_1,\hdots,x_n-a_n)\) is a field,
    as it sends all the indeterminates into itself.
    Hence \((x_1-a_1,\hdots,x_n-a_n) = \mfk{m}\).

    Now, any proper ideal \(\mfk{a}\) must be contained within a maximal ideal \(\mfk{m}\)
    (Corollary~\ref{cor:proper-in-maximal}).
    But as we know that \(\Van\) is an inclusion reversing operation
    (Lem~\ref{lem:iv-inclusion-reversing}),
    \((a_1,\hdots,a_n) \in \Van(\mfk{m}) \subseteq \Van(\mfk{a})\).
\end{proof}
\begin{theorem}[Hilbert's Nullstellensatz, `zero-point theorem']\label{thm:nullstellensatz}
    There is a bijection between the algebraic subsets of \(K^n\)
    and the radical ideals of \(K[x_1,\hdots,x_n]\),
    using the operations \(\Van\) and \(\Ide\).
\end{theorem}
\begin{proof}
    We already have, from Lemma~\ref{lem:vix-x}, that \(\Van(\Ide(X)) = X\) for any algebraic subset,
    so it suffices to prove that \(\Ide(\Van(\mfk{a})) = \mfk{a}\) for any radical.
    We shall prove, instead, that for any ideal \(\mfk{a}\),
    \(\Ide(\Van(\mfk{a})) = \sqrt{\mfk{a}}\).
    By Lemma~\ref{lem:iv-inclusion-reversing},
    we can see that \(\mfk{a} \subseteq \Ide(\Van(\mfk{a}))\),
    so \(\sqrt{\mfk{a}} \subseteq \sqrt{\Ide(\Van(\mfk{a}))} = \Ide(\Van(\mfk{a}))\),
    since \(\Ide(\Van(\mfk{a}))\) is radical (Lemma~\ref{lem:ix-radical}).

    For the forward inclusion,
    suppose \(\mfk{a} = (f_1,\hdots,f_n)\) has these \(n\) generators
    (finitely generated due to Corollary~\ref{cor:noetherian-generated}).
    Let \(g \in \Ide(\Van(\mfk{a}))\).
    Consider the ideal \((f_1,\hdots,f_n,1-yg) \subseteq K[x_1,\hdots,x_n,y]\).
    We claim that \(\Van(f_1,\hdots,f_n,1-yg) = \emptyset\),
    so by way of contradiction, suppose \((\vb{a},b) = (a_1,\hdots,a_n,b) \in \Van(f_1,\hdots,f_n,1-yg)\).
    Since each \(f_i\) is only a function of \(\vb{x} = (x_1,\hdots,x_n)\) and not of \(y\),
    we know that \(f_i(\vb{a}) = 0\);
    on the other hand, the last generator in the ideal gives us
    \((1-yg)(\vb{a},b) = 1 - bg(\vb{a}) = 0\).
    But \(f_i(\vb{a}) = 0\) implies that \(\vb{a} \in \Van(\mfk{a})\),
    so since \(g \in \Ide(\Van(\mfk{a}))\), \(g(\vb{a}) = 0\),
    which when substituted into the equation above, yields \(1 = 0\), which is a contradiction.

    Now, since \(\Van(f_1,\hdots,f_n,1-yg) = \emptyset\),
    the \hyperref[lem:weak-nullstellensatz]{weak Nullstellensatz}
    asserts that \((f_1,\hdots,f_n,1-yg) = K[x_1,\hdots,x_n,y]\) is the whole ring.
    In particular, \(1 \in (f_1,\hdots,f_n,1-yg)\),
    so we can write 1 as a linear combination
    \begin{equation*}
        1 = \sum_{i=1}^n h_i f_i + h_0(1-yg)
    \end{equation*}
    for \(h_i \in K[x_1,\hdots,x_n,y]\).
    If we substitute \(y \mapsto 1/g\), we can get that
    \begin{equation*}
        1 = \sum_{i=1}^n h_i\pqty{\vb{x},\frac{1}{g}}f_i(\vb{x}) = \frac{F}{g^N}
    \end{equation*}
    after finding the common denominator and pushing the sum of numerators into \(F\).
    But cancelling out the denominator gives us
    \begin{equation*}
        g^N = \sum_{i=1}^n g^N h_i\pqty{\vb{x},\frac{1}{g}}f_i(\vb{x})
        \in (f_1,\hdots,f_n) = \mfk{a}
    \end{equation*}
    which is exactly the definition of \(g \in \sqrt{\mfk{a}}\).
\end{proof}
\begin{remark}
    The method of proving the full Nullstellensatz from its weak form
    is introduced by Rabinowitsch in 1930,
    and is commonly called the Rabinowitsch trick.
\end{remark}

\section{Filtrations}

\begin{definition}
    A filtration of a ring is a descending sequence of ideals
    \(R = \mfk{a}_0 \supseteq \mfk{a}_1 \supseteq \mfk{a}_2 \supseteq \cdots\)
    satisfying \(\mfk{a}_i\mfk{a}_j \subseteq \mfk{a}_{i+j}\).
    Similarly, a filtration of an \(R\)-module is a descending sequence of \(R\)-modules
    \(M = M_0 \supseteq M_1 \supseteq M_2 \supseteq \cdots\)
    satisfying \(M_i M_j \subseteq M_{i+j}\).
\end{definition}
\begin{definition}
    If \(\mfk{a} \subseteq R\) is an ideal,
    the \(\mfk{a}\)-adic filtration of \(R\) is given by the sequence
    \(R = R \supseteq \mfk{a}R \supseteq \mfk{a}^2R \supseteq \mfk{a}^3R \supseteq \cdots\).
\end{definition}
\begin{definition}
    Suppose \(\mfk{a} \subseteq R\) is an ideal.
    A filtration \(M = M_0 \supseteq M_1 \supseteq M_2 \supseteq \cdots\) is an \(\mfk{a}\)-filtration
    if \(\mfk{a}M_n \subseteq M_{n+1}\) for all \(n\).
    An \(\mfk{a}\)-filtration \(M = M_0 \supseteq M_1 \supseteq M_2 \supseteq \cdots\) is \(\mfk{a}\)-stable
    if there exists some \(N\) such that \(\mfk{a}M_n = M_{n+1}\) for all \(n \geq N\).
\end{definition}

\begin{definition}
    A graded ring \(R\) is a family of additive subgroups \({\{R_n\}}_{n=0}^\infty\) of \(R\)
    such that \(R = \bigoplus_{n=0}^\infty R_n\) and \(R_m R_n \subseteq R_{m+n}\);
    in particular, \(A_0 \subseteq A\) is a subring, and \(A_n\) are \(A_0\)-modules.
\end{definition}
\begin{definition}
    Suppose \(R\) is a graded ring \(R = \bigoplus_{n=0}^\infty R_n\).
    A graded \(R\)-module \(M\) is a family of subgroups \({\{M_n\}}_{n=0}^\infty\) of \(M\)
    % and a family of additive subgroups \({\{R_n\}}_{n=0}^\infty\) of \(R\)
    such that \(M = \bigoplus_{n=0}^\infty M_n\) and \(R_m M_n \subseteq M_{m+n}\).
    Elements that reside in some \(M_n\) are called homogeneous of degree \(n\).
\end{definition}

\begin{proposition}
    Suppose \(R\) is a graded ring with \(R = \bigoplus_{n=0}^\infty R_n\).
    Then \(R\) is a Noetherian ring if and only if \(R_0\) is Noetherian and \(R\) is a finite-type \(R_0\)-algebra.
\end{proposition}
\begin{proof}
    Suppose \(R\) is a Noetherian ring.
    Let \(R_+ = \bigoplus_{n=1}^\infty R_n\), such that \(R_0 \cong R/R_+\).
    Then immediately using Theorem~\ref{thm:noetherian-quotient},
    \(R_0 \cong R/R_+\) is Noetherian.
    % Now, as \(R_+ \subseteq R\) is an ideal, and because of \(R\) Noetherian,
    % we may write \(R_+ = (x_1,\hdots,x_s)\) with a finite number of generators over \(R\).
    We claim that \(R_+\) can be generated by finitely many homogeneous elements.
    Clearly, because \(R_+\) can be generated by the (infinite) set \(S = \bigcup_{n=1}^\infty R_n\),
    consider the set of ideals generated by finite subsets of \(S\).
    Since every chain of ideals is bounded above by the ascending chain condition due to \(R\) being Noetherian,
    there must exist some maximal element, a finitely generated ideal \(\mfk{a} = (x_1,\hdots,x_s)\).
    If \(\mfk{a} \subsetneq R_+\), then there exists some \(x_{s+1} \in S\) such that \(x_{s+1} \notin \mfk{a}\).
    But then you can construct a strictly larger ideal \(\mfk{a} \subsetneq (x_1,\hdots,x_{s+1})\),
    which contradicts maximality of \(\mfk{a}\).
    Hence \(R_+ = \mfk{a} = (x_1,\hdots,x_s)\) is finitely generated by homogeneous elements.

    Considering \(R_0 \subseteq R\) as a subring,
    allow us to denote the subring \(R' = R_0[x_1,\hdots,x_s] \subseteq R\).
    We claim that \(R_n \subseteq R'\) for all \(n\) via induction.
    By definition we have the base case \(R_0 \subseteq R'\).
    Now consider any \(y \in R_n \subseteq R_+\), assuming that for all \(k < n\), \(R_k \subseteq R'\).
    We can write this as a linear combination \(y = \sum_{i=1}^s a_i x_i\).
    We shall, without loss of generality, assume that \(a_i \in R_{n-\deg x_i}\)
    (with the convention that \(a_i = 0\) if \(\deg x_i > n\));
    this is allowed because if we write \(a_i = b_i + c_i\),
    with \(b_i\) being the homogeneous component of degree \(n - \deg x_i\)
    and \(c_i\) being the other homogeneous components,
    we have \(\sum_{i=1}^s c_i x_i = y - \sum_{i=1}^s b_i x_i \in R_n\)
    and hence \(\sum_{i=1}^s c_i x_i = 0\).
    By the inductive hypothesis, \(a_i \in R'\), so \(\sum a_i x_i \in R'\) as desired.
    Hence we can conclude that \(R = R'\) since all the graded components are in \(R'\).

    The reverse direction is trivially given by the \hyperref[cor:hilbert-basis]{Hilbert basis theorem}.
\end{proof}

\subsection*{Blowup Algebra}

\begin{definition}
    Suppose \(\mfk{a} \subseteq R\) is an ideal.
    The blowup algebra of \(\mfk{a}\) in \(R\) is the graded ring
    \begin{equation*}
        \Bl_\mfk{a} R = \bigoplus_{n=0}^\infty \mfk{a}^n = R \oplus \mfk{a} \oplus \mfk{a}^2 \oplus \cdots
        \cong R[t\mfk{a}] \subseteq R[t]
    \end{equation*}
    where \(t\) is a transcendental element.
\end{definition}
\begin{definition}
    Suppose \(\mfk{a} \subseteq R\) is an ideal, \(M\) is an \(R\)-module,
    and \(\mca{A}: M = M_0 \supseteq M_1 \supseteq M_2 \supseteq \cdots\) is an \(\mfk{a}\)-filtration.
    Then we have a graded module over the blowup ring \(\Bl_\mfk{a} R\)
    \begin{equation*}
        \Bl_\mca{A} M = \bigoplus_{n=0}^\infty M_n = M \oplus M_1 \oplus M_2 \oplus \cdots
    \end{equation*}
    where \(\mfk{a}^m M_n \subseteq M_{m+n}\).
\end{definition}
\begin{lemma}\label{lem:stable-blowup}
    Suppose \(\mfk{a} \subseteq R\) is an ideal, \(M\) is a finitely generated \(R\)-module,
    and \(\mca{A}: M = M_0 \supseteq M_1 \supseteq M_2 \supseteq \cdots\) is an \(\mfk{a}\)-filtration,
    with \(M_n\) all finitely generated over \(R\).
    Then \(\Bl_\mca{A} M\) is a finitely generated \(\Bl_\mfk{a} R\)-module
    if and only if \(\mca{A}\) is a stable filtration.
\end{lemma}
\begin{proof}
    Suppose \(\Bl_\mca{A} M\) is finitely generated by \({\{x_i\}}_{i=1}^s\).
    Then the generators \({\{x_i\}}_{i=1}^s \subset \bigoplus_{k=1}^n M_k\) for some \(n\).
    If we let \(x_i = \sum_{j=1}^n x_{ij}\) be the homogeneous components of \(x_i\),
    then \(\Bl_\mca{A} M\) is also finitely generated by these homogeneous components.
    But if we consider the sequence of morphisms
    \begin{equation*}
        \begin{tikzcd}
            \Bl_\mfk{a} R[x_{ij}] \arrow[twoheadrightarrow]{r} &
            \Bl_\mca{A} M = \displaystyle\bigoplus_{k=1}^\infty M_k \arrow[twoheadrightarrow]{r} &
            \displaystyle\bigoplus_{k=n}^\infty M_k
        \end{tikzcd}
    \end{equation*}
    the \(x_{ij}\) with \(j < n\) gets sent to 0 at the end,
    so \(x_{in}\) finitely generates \(\bigoplus_{k=n}^\infty M_k\).
    % In particular \(\bigoplus_{k=n}^\infty M_k \subseteq \Bl_\mca{A} M\),
    % so any element \(y \in M_k\) can be written as \(y = \sum a_{ij}x_{ij}\) for \(a_{ij} \in \Bl_\mfk{a} R\).
    % Obviously we do not have to care about the terms with \(j < n\), since \(a_{ij}x_{ij} \in \)
    % We claim that \(x_{in}\) generates 
    But that means \(M_n\) generates \(\bigoplus_{k=n}^\infty M_k\),
    so immediately \(M_{n+k} \subseteq \mfk{a}^k M_n\) after matching degrees,
    giving us stability.

    For the reverse direction, if \(\mca{A}\) is stable, as in \(M_{n+k} = \mfk{a}^k M_n\) for some \(n\),
    then \(\Bl_\mca{A} M\) can be generated by the finite union of the set of genrators of \({\{M_k\}}_{k=0}^n\).
\end{proof}
\begin{proposition}[Artin-Rees Lemma]\label{prop:artin-rees}
    Suppose \(R\) is a Noetherian ring, \(\mfk{a} \subseteq R\) is an ideal,
    \(M\) is a finitely generated \(R\)-module,
    and \(\mca{A}: M = M_0 \supseteq M_1 \supseteq M_2 \supseteq \cdots\) is a stable \(\mfk{a}\)-filtration.
    If \(N \subseteq M\) is a submodule,
    then \(N\cap\mca{A}: N = N \cap M_0 \supseteq N \cap M_1 \supseteq N \cap M_2 \supseteq \cdots\)
    is a stable \(\mfk{a}\)-filtration;
    that is, there exists some \(n\) such that for all \(k \geq 0\),
    \(N \cap M_{n+k} = \mfk{a}^k(N \cap M_n)\).
\end{proposition}
\begin{proof}
    Obviously \(N\cap\mca{A}\) is an \(\mfk{a}\)-filtration,
    because \(\mfk{a}(N \cap M_n) \subseteq \mfk{a}N \cap \mfk{a}M_n \subseteq N \cap M_n\).
    Hence we have the finitely generated graded \(\Bl_\mfk{a} R\)-module
    \(\Bl_{(N\cap\mca{A})} N \subseteq \Bl_\mca{A} M\) because \(\Bl_\mfk{a} R\) is Noetherian.
    By the \hyperref[lem:stable-blowup]{lemma above},
    we are done.

    The second part can be seen by definition of stability.
\end{proof}

\begin{theorem}[Krull's Intersection Theorem]
    Suppose \(R\) is a Noetherian ring, and \(\mfk{a} \subseteq R\) is an ideal.
    If \(M\) is a finitely generated \(R\)-module,
    then there exists some element \(a \in \mfk{a}\) such that
    \((1-a)\bigcap_{k=1}^\infty \mfk{a}^k M = 0\).
    If \(R\) is additionally a domain or a local ring, and \(\mfk{a} \subsetneq R\) is a proper ideal,
    then \(\bigcap_{k=1}^\infty \mfk{a}^k = 0\).
\end{theorem}
\begin{proof}
    Applying the \hyperref[prop:artin-rees]{Artin-Rees lemma}
    to \(\bigcap_{k=1}^\infty \mfk{a}^k M \subseteq M\) (and the \(\mfk{a}\)-adic filtration of \(M\)),
    there exists some \(n\) such that
    \begin{equation*}
        \bigcap_{k=1}^\infty \mfk{a}^k M = \bigcap_{k=1}^\infty \mfk{a}^k M \cap \mfk{a}^{n+1} M
        = \mfk{a}\pqty{\bigcap_{k=1}^\infty \mfk{a}^k M \cap \mfk{a}^n M}
        = \mfk{a}\pqty{\bigcap_{k=1}^\infty \mfk{a}^k M}
    \end{equation*}
    Then we have the first statement using \hyperref[cor:nakayama]{Nakayama's lemma}.

    Taking \(M = R\), to prove the second statement,
    it suffices to show that \(1-r\) is not a zero divisor.
    In a domain and a proper ideal, \(r \neq 1\) so \(1-r \neq 0\) and we are done.
    If \(R\) is a local ring, then \(\mfk{a} \subseteq \mfk{m}\) is contained inside the maximal (also Jacobson) ideal,
    so by Lemma~\ref{lem:1-xy-unit} \(r \in \mfk{m}\) implies that \(1-r\) is a unit,
    i.e.\ not a zero divisor.
\end{proof}

\subsection*{Associated Graded Ring}

\begin{definition}
    Given an ideal \(\mfk{a} \subseteq R\),
    the associated graded ring of \(R\) with respect to \(\mfk{a}\) is
    \begin{equation*}
        \gr_\mfk{a} R = \bigoplus_{k=0}^\infty \mfk{a}^k/\mfk{a}^{k+1}
        = R/\mfk{a} \oplus \mfk{a}/\mfk{a}^2 \oplus \mfk{a}^2/\mfk{a}^3 \oplus \cdots
    \end{equation*}
    Notice that if \([x] \in \mfk{a}^m/\mfk{a}^{m+1}\) and \([y] \in \mfk{a}^n/\mfk{a}^{n+1}\),
    given any \(x' \in \mfk{a}^{m+1}\) and \(y' \in \mfk{a}^{n+1}\),
    we have \((x+x')(y+y') = xy + xy' + x'y + x'y' \in xy + \mfk{a}^{m+n+1} + \mfk{a}^{m+n+2} \subseteq xy + \mfk{a}^{m+n+1}\),
    so \([x][y] = [xy] \in \mfk{a}^{m+n}/\mfk{a}^{m+n+1}\) is well-defined.
    Alternatively we can express this as a quotient of blowup rings
    \(\gr_\mfk{a} R = \Bl_\mfk{a}R/\mfk{a}\Bl_\mfk{a}R\).
\end{definition}
\begin{definition}
    Given an ideal \(\mfk{a} \subseteq R\), an \(R\)-module \(m\),
    and an \(\mfk{a}\)-filtration \(\mca{A}: M = M_0 \supseteq M_1 \supseteq \cdots\),
    the associated graded module of \(M\) with respect to the \(\mfk{a}\)-filtration \(\mca{A}\) is
    \begin{equation*}
        \gr_{\mca{A}} M = \bigoplus_{k=0}^\infty M_k/M_{k+1}
        = M/M_1 \oplus M_1/M_2 \oplus M_2/M_3 \oplus \cdots
    \end{equation*}
    Because \(\mca{A}\) is an \(\mfk{a}\)-filtration, the multiplication works as the associated graded ring above.
\end{definition}

\begin{proposition}
    Suppose \(\mfk{a} \subseteq R\) is an ideal, and \(M\) is a finitely generated \(R\)-module.
    If \(\mca{A}: M = M_0 \supseteq M_1 \supseteq \cdots\) is an \(\mfk{a}\)-stable filtration,
    with \(M_n\) all finitely generated over \(R\),
    then \(\gr_\mca{A} M\) is a finitely generated \(\gr_\mfk{a} R\)-module.
\end{proposition}
\begin{proof}
    Let \(n\) be the integer such that for all \(k \geq n\), \(\mfk{a}M_k = M_{k+1}\).
    We then have \((\mfk{a}/\mfk{a}^2)(M_k/M_{k+1}) = M_{k+1}/M_{k+2}\) by simple verification.
    Hence the generators of \(M_0/M_1,\hdots,M_n/M_{n+1}\),
    and in particular to generators of \({\{M_k\}}_{k=1}^n\),
    will generate \(\gr_\mca{A} M\).
\end{proof}

\section{Completions}

\begin{definition}
    Suppose \(\mca{A}: R = \mfk{a}_0 \supseteq \mfk{a}_1 \supseteq \mfk{a}_2 \supseteq \cdots\)
    is a descending chain of ideals (resp.\ filtration).
    The completion of \(R\) with respect to \(\mca{A}\) is the inverse limit of the quotients
    \begin{equation*}
        \hat{R}_\mca{A} = \invlim R/\mfk{a}_i
        = \Bqty{g = (g_1,g_2,\hdots) \in \prod_{i=1}^\infty R/\mfk{a}_i: g_j \equiv g_i\;(\text{mod}\,\mfk{m}_i)\;\text{when}\; j > i}
    \end{equation*}
    \(\hat{R}_\mca{A}\) has a descending chain of ideals (resp.\ filtration)
    \begin{equation*}
        \hat{\mfk{a}}_i = \Bqty{g = (g_1,g_2,\hdots) \in \hat{R}_\mca{A}: g_j = 0 \;\text{when}\; j \leq i}
    \end{equation*}
    and \(\hat{R}/\hat{\mfk{a}}_i = R/\mfk{a}_i\).
\end{definition}
\begin{remark}
    Categorically, this is an inverse system indexed by the category \(\mca{I} = \{0 \leftarrow 1 \leftarrow 2 \leftarrow \cdots\}\).
    where given any object, \(F(i) = R/\mfk{a}_i\),
    and given any morphism, \(F(f_{ij}) = (R/\mfk{a}_i \to R/\mfk{a}_j)\) is the canonical projection (when \(i \geq j\)).
    By Theorem~\ref{thm:existence-limit},
    since there are no equalizers in this category,
    and products exist (we proved that countable direct products exist),
    this inverse limit is guaranteed to exist.
\end{remark}
\begin{definition}
    Suppose \(R \supseteq \mfk{a} \supseteq \mfk{a}^2 \supseteq \cdots\)
    is the \(\mfk{a}\)-adic filtration of \(R\).
    The completion of \(R\) with respect to \(\mfk{a}\) is the inverse limit of the quotients
    \(\hat{R}_\mfk{a} = \invlim R/\mfk{a}^i\).
\end{definition}

\begin{lemma}
    Suppose \(\mfk{m} \subseteq R\) is a maximal ideal.
    Then \((\hat{R}_\mfk{m},\hat{\mfk{m}})\) is a local ring.
\end{lemma}
\begin{proof}
    First off, \(\hat{R}_\mfk{m}/\hat{\mfk{m}} = R/\mfk{m}\) is a field,
    so indeed \(\hat{\mfk{m}}\) is a maximal ideal.
    We then claim that every element \(g\) outside of \(\hat{\mfk{m}}\) is a unit (Proposition~\ref{prop:local-ring-units}).
    Obviously if \(g \in \hat{R}_\mfk{m} - \hat{\mfk{m}}\),
    then \(g_1 \neq 0\), and \(g_i \notin \mfk{m}(R/\mfk{m}^i)\), so each \(g_i\) is a unit.
    But \(g^{-1} = (g_1^{-1},g_2^{-1},\hdots)\) is our inverse,
    so \(g\) is indeed a unit.
\end{proof}

\begin{definition}
    \(\hat{R}_\mfk{m}\) is complete with respect to \(\mfk{m}\) if \(R \to \hat{R}_\mfk{m}\) is an isomorphism.
    In particular, \(\hat{R}_\mfk{m}\) is a complete local ring if \(\mfk{m}\) is also a maximal ideal.
\end{definition}

\begin{lemma}\label{lem:completion-properties}
    Suppose \(\mfk{m} \subseteq R\) is an ideal,
    and \(\hat{R} = \hat{R}_\mfk{m}\) is the \(\mfk{m}\)-adic completion of \(R\).
    We have the following properties:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(\hat{R}/\hat{\mfk{m}}_n = R/\mfk{m}^n\);
        \item \(\hat{R} = \invlim \hat{R}/\hat{\mfk{m}}_n\),
            and \(\hat{R}\) is complete with respect to the filtration by \(\hat{\mfk{m}}_n\); and
        \item if \(\gr_{\hat{\mfk{m}}_n} \hat{R}\) denotes the associated graded ring with respect to the filtration by \(\hat{\mfk{m}}_n\),
            then \(R \to \hat{R}\) induces an isomorphism \(\gr_\mfk{m} R \to \gr_{\hat{\mfk{m}}_n} \hat{R}\).
    \end{enumerate}
\end{lemma}
\begin{proof}
    Part (a) follows from the definition,
    and part (b) follows from part (a) immediately.

    For part (c), a useful application of the \hyperref[thm:iso-3-ring]{third isomorphism theorem}
    yields \(R/\mfk{m}^n \cong (R/\mfk{m}^{n+1})/(\mfk{m}^n/\mfk{m}^{n+1}) \cong (\hat{R}/\hat{\mfk{m}}_{n+1})/(\mfk{m}^n/\mfk{m}^{n+1}) \cong \hat{R}/\hat{\mfk{m}}_n\),
    so we have isomorphisms between every graded component.
\end{proof}

\subsection*{Krull Topology}

\begin{lemma}\label{lem:0-closed-hausdorff}
    Suppose \(G\) is a topological (additive) abelian group,
    with the addition and inversion operations
    \begin{equation*}
        \varfunc{+}{G \times G}{G}{(x,y)}{x+y} \qquad \varfunc{-}{G}{G}{x}{-x}
    \end{equation*}
    both continuous.
    If \(\{0\}\) is closed in \(G\), then \(G\) is Hausdorff.
\end{lemma}
\begin{proof}
    Since \((x,y) \mapsto x-y\) is continuous (being the composition of two continuous maps),
    the preimage of \(\{0\}\) under this map is the diagonal in \(G \times G\),
    so the diagonal is closed.
    By Lemma~\ref{lem:diagonal-closed-hausdorff}, \(G\) is Hausdorff.
\end{proof}
\begin{lemma}
    Suppose \(G\) is a topological abelian group, with continuous addition and inversion.
    Given any neighbourhood \(U \ni 0\), \(U + x\) is a neighbourhood of \(x \in G\);
    conversely, any neighbourhood of \(x \in G\) must be of the form \(U + x \ni x\),
    where \(U\) is a neighbourhood of 0.
    In other words, the neighbourhoods of 0 uniquely determine the topology on \(G\).
\end{lemma}
\begin{proof}
    Fix any element \(x \in G\).
    The translation mapping \(g \mapsto g+x\) is continuous;
    moreover, its inverse \(g \mapsto g-x\) is also a translation, so it must be a homeomorphism.
    Hence translations are open maps, so \(U \ni 0\) is a neighbourhood if and only if \(U + x \ni x\) is a neighbourhood.
\end{proof}
\begin{lemma}
    Suppose \(H\) is the intersection of all neighbourhoods of 0.
    Then:
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(H\) is a subgroup;
        \item \(H = \overline{\{0\}}\);
        \item \(G/H\) is Hausdorff; and
        \item \(G\) is Hausdorff if and only if \(H = \{0\}\).
    \end{enumerate}
\end{lemma}
\begin{proof}
    For (a), we see that \(U \ni 0\) is a neighbourhood if and only if \(-U \ni 0\) is a neighbourhood,
    because inversion is continuous.
    Hence if \(x \in H = \bigcap U = \bigcap (-U)\), then \(-x \in H = \bigcap U\) also.
    Now, if \(x,y \in H\), consider any neighbourhood \(U \ni 0\).
    Then there exists some open \(V \subseteq U\), and the preimage of addition is \(+^{-1}(V) \subseteq G \times G\) is open.
    In particular there exists open subsets \(E \times E' \subseteq +^{-1}(V)\).
    But then \(x \in H \subseteq E\) and \(y \in H \subseteq E'\) because it is open,
    and hence \(x+y \in V\).

    For (b), if a nonzero point \(x \in H = \bigcap U\) for all neighbourhoods \(U \ni 0\),
    the same can be said for \(-x\).
    This happens if and only if \(0 \in \bigcap U+x\),
    and these are precisely the neighbourhoods of \(x\).
    % But that is exactly saying that every open set containing \(-x\) contains 0,
    Hence \(x\) is a boundary point of \(\{0\}\),
    since every neighbourhood of \(x\) contains a point in \(\{0\}\) (namely 0),
    but also contains a point (\(x\)) not in \(\{0\}\).
    Therefore \(x \in \overline{\{0\}}\).
    The reverse argument holds, since \(x \notin \{0\}\),
    so we are forced to conclude that \(x \in \partial\{0\}\).

    For (c), since \(H = [0] \in G/H\) is closed, by Lemma~\ref{lem:0-closed-hausdorff} we are done.

    For (d), the reverse direction follows directly from (c).
    For the forward direction, if \(G\) is Hausdorff, then in particular it is \(\text{T}_1\),
    and every singleton is closed;
    in particular, this means \(\{0\} = \overline{\{0\}} = H\).
\end{proof}

\begin{definition}
    Suppose \({\{a_i\}}_{i=1}^\infty \subset G\) is a sequence of elements in a topological abelian group.
    % The sequence converges to \(a \in \hat{R}\) if for every \(n \in \bN\),
    % there exists \(m \in \bN\) such that for all \(i \geq m\), \(a - a_i \in \hat{\mfk{m}}_n\).
    % We call \(a = \lim_{n\to\infty} a_n\) the limit of the sequence.
    The sequence is Cauchy if for every open \(U \ni 0\),
    there exists \(m \in \bN\) such that for all \(i,j \geq m\), \(a_i - a_j \in U\).
    Two Cauchy sequences \(\{a_i\},\{b_i\}\) are equivalent if \(a_i - b_i \to 0 \in G\).
\end{definition}
\begin{definition}
    Let \(\hat{G}\) be the space of all Cauchy sequences in \(G\).
    \(\hat{G}\) is an abelian group, because addition and inversion works element-wise.
\end{definition}
\begin{definition}
    Suppose \(G\) is a topological abelian group,
    with a sequence of subgroups \(G = G_0 \supseteq G_1 \supseteq G_2 \supseteq \cdots\),
    where the only open neighbourhoods of 0 are \(G_n\).
    Just as before, if we define \(\hat{G}_n = \ker(\hat{G} \to G/G_n)\),
    then it is possible to endow a topology on \(\hat{G}\),
    where the only open neighbourhoods of 0 are \(\hat{G}_n\).
    We call this the Krull topology on \(\hat{G}\).
\end{definition}
\begin{proposition}
    The Cauchy sequence construction is identical to the inverse limit construction.
\end{proposition}
\begin{proof}
    Since every Cauchy sequence in \(G\) is has an eventually constant image in \(G/G_n\),
    we can simply define that image as \(g_n\),
    and verify that it satisfies the definition of the inverse limit.
    Indeed, every Cauchy sequence has such a representation in \(g_n\);
    and because we quotient away the equivalent Cauchy sequences,
    such a representation is also unique.
\end{proof}

\subsection*{Completions and Associated Graded Rings}

\begin{proposition}\label{prop:graded-ring-generators}
    Suppose \(R\) is complete with respect to the filtration by ideals \(\mfk{m}_n\),
    and let \(\gr R = \gr_{\mfk{m}_n} R\) be the associated graded ring with respect to this filtration.
    Given any \(a \in R\), let \(l(a)\) the leading term of \(a\);
    that is, find the largest \(n\) such that \(a \in \mfk{m}_n\),
    and \(l(a) = a \pmod{\mfk{m}_{n+1}} \in \mfk{m}_n/\mfk{m}_{n+1} \subseteq \gr R\).
    Suppose \({\{a_i\}}_{i=1}^s \subset \mfk{a} \subseteq R\) are elements in an ideal.
    If \({\{l(a_i)\}}_{i=1}^s\) generate the ideal \(l(\mfk{a}) \subseteq \gr R\),
    then \({\{a_i\}}_{i=1}^s\) generates \(\mfk{a} \subseteq R\).
\end{proposition}
\begin{proof}
    Let \(\mfk{a}' = (a_1,\hdots,a_s) \subseteq \mfk{a}\).
    Without any loss of generality, suppose \(a_i \neq 0\).
    Find a sufficiently large \(d\) such that \(a_i \notin \mfk{m}_d\).
    Given any \(f \in \mfk{a}\), with \(l(f)\) of degree \(r\),
    we write \(l(f) = \sum_{i=1}^s \bar{c}_i l(a_i)\) for some \(\bar{c}_i \in \gr R\),
    and by matching degrees, \(\deg(\bar{c}_i) = r - \deg(l(a_i))\).

    Then if we choose \(c_i \in R\) such that \(l(c_i) = c_i\),
    then \(f_1 = f - \sum_{i=1}^s c_i a_i \in \mfk{m}_{r+1}\), because the leading terms already match up.
    We can repeat this process any number of times,
    finding \(f_{n+1} = f_n - \sum_{i=1}^s c_i^{(n)} a_i \in \mfk{m}_{r+n+1}\).
    Hence we can eventually get \(f - \sum_{i=1}^s c''_i a_i \in \mfk{m}_{d+1}\).
    We can see that \(f \in \mfk{a}'\) if and only if \(f - \sum_{i=1}^s c''_i a_i \in \mfk{a}'\),
    so without any loss of generality, let \(f \in \mfk{m}_{d+1}\).

    Then in \(l(f) = \sum_{i=1}^s \bar{c}_i l(a_i)\),
    the degrees are \(\deg(\bar{c}_i) = r - \deg(l(a_i)) > r-d\).
    Hence we can choose \(c_i \in \mfk{m}_{r-\deg(l(a_i))} \subseteq \mfk{m}_{r-d}\).
    % because adding anything in \(\mfk{m}_{r-\deg(l(a_i))+1} \supseteq \mfk{m}_{r-d}\)
    % does not change its representative \(\bar{c}_i \in \mfk{m}_{r-\deg(l(a_i))}/\mfk{m}_{r-\deg(l(a_i))+1}\).
    Repeating this procedure, we can choose
    \begin{equation*}
        f_{n+1} = f - \sum_{i=1}^s c_i^{(0)} a_i - \sum_{i=1}^s c_i^{(1)} a_i - \cdots - \sum_{i=1}^s c_i^{(n)} a_i
        = f - \sum_{i=1}^s \sum_{k=1}^n c_i^{(k)} a_i \in \mfk{m}_{r+n+1}
    \end{equation*}
    By definition this means \(f - \sum_{i=1}^s \sum_{k=1}^n c_i^{(k)} a_i \to 0\),
    and the two Cauchy sequences are equivalent.
    We are then forced to conclude that \(f = \sum_{i=1}^s \sum_{k=1}^\infty c_i^{(k)} a_i \in \mfk{a}'\) as desired.
\end{proof}
\begin{theorem}\label{thm:completion-noetherian}
    Suppose \(R\) is a Noetherian ring, and \(\mfk{m} \subseteq R\) is an ideal.
    Let \(\hat{R} = \hat{R}_\mfk{m}\) denote the completion.
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item \(\hat{R}\) is a Noetherian ring.
        \item \(\hat{R}/\mfk{m}^k\hat{R} = R/\mfk{m}^k\).
        Therefore \(\hat{R}\) is complete with respect to \(\mfk{m}\hat{R}\),
        and \(\gr_{\mfk{m}\hat{R}} \hat{R} = \gr_\mfk{m} R\).
    \end{enumerate}
\end{theorem}
\begin{proof}
    For part (a), it suffices to prove that every ideal in \(\hat{R}\) is finitely generated (Corollary~\ref{cor:noetherian-generated}).
    We have \(\gr \hat{R} = \gr R\) (graded using their respective filtrations by \(\hat{\mfk{m}}\) and \(\mfk{m}\))
    because every \(\hat{\mfk{m}}_n/\hat{\mfk{m}}_{n+1} = \mfk{m}^n/\mfk{m}^{n+1}\) (Lemma~\ref{lem:completion-properties}).
    Since \(R\) is Noetherian, \(R/\mfk{m}\) is Noetherian (Theorem~\ref{thm:noetherian-quotient}).
    Because \(\ker(R/\mfk{m}^2 \to R/\mfk{m}) = \mfk{m}/\mfk{m}^2\) is an ideal of \(R/\mfk{m}^2\),
    it is finitely generated, so it must be a finite-dimensional vector space over \(R/\mfk{m}\).
    Hence \(R/\mfk{m}[\mfk{m}/\mfk{m}^2] \to \gr R\) is a surjective homomorphism,
    and by the \hyperref[cor:hilbert-basis]{Hilbert basis theorem}, \(\gr R = \gr \hat{R}\) is Noetherian.
    We can therefore conclude using the \hyperref[prop:graded-ring-generators]{proposition above}
    that \(\hat{R}\) is Noetherian.

    For (b), it suffices to prove that \(\mfk{m}_k = \mfk{m}^k\hat{R}\).
    By the \hyperref[prop:graded-ring-generators]{proposition above},
    we only need to prove that \(l(\mfk{m}_k) = l(\mfk{m}^k\hat{R})\).
    But this is obviously true, because they are both generated by all elements of degree greater than or equal to \(n\).
\end{proof}

\begin{proposition}\label{prop:invlim-exact}
    Suppose \(\{A_n\},\{B_n\},\{C_n\}\) are inverse systems indexed by \(\mca{I} = \{0 \leftarrow 1 \leftarrow 2 \leftarrow \cdots\}\).
    and that we have an exact sequence of inverse system in the sense that
    \begin{equation*}
        \begin{tikzcd}
            & \vdots \arrow{d} & \vdots \arrow{d} & \vdots \arrow{d} \\
            0 \arrow{r} & A_{n+1} \arrow{r} \arrow{d}{\pi_{n+1}} & B_{n+1} \arrow{d}{\pi_{n+1}} \arrow{r} & C_{n+1} \arrow{r} \arrow{d}{\pi_{n+1}} & 0 \\
            0 \arrow{r} & A_n \arrow{r} \arrow{d} & B_n \arrow{r} \arrow{d} & C_n \arrow{r} \arrow{d} & 0 \\
            & \vdots & \vdots & \vdots \\
        \end{tikzcd}
    \end{equation*}
    all the horizontal morphisms are exact, and the diagram commutes.
    In this case,
    \begin{equation*}
        \begin{tikzcd}
            0 \arrow{r} & \invlim A_n \arrow{r} & \invlim B_n \arrow{r} & \invlim C_n
        \end{tikzcd}
    \end{equation*}
    is exact; in other words, \(\invlim\) is a left-exact functor.
    Moreover, if every \(\func{\pi_{n+1}}{A_{n+1}}{A_n}\) is surjective, then
    \begin{equation*}
        \begin{tikzcd}
            0 \arrow{r} & \invlim A_n \arrow{r} & \invlim B_n \arrow{r} & \invlim C_n \arrow{r} & 0
        \end{tikzcd}
    \end{equation*}
    is exact.
\end{proposition}
\begin{proof}
    Let us define the morphism
    \begin{equation*}
        \varfunc{d^A}{\prod_{n=1}^\infty A_n}{\prod_{n=1}^\infty A_n}{(x_n)}{(x_n - \pi_{n+1}(x_{n+1}))}
    \end{equation*}
    and similarly define \(d^B,d^C\).
    One can easily see that \(\ker d^A = \invlim A_n\),
    because that is exactly the condition \(x_n = x_{n+1}\) in \(A_n\).
    Then we have the diagram of short exact sequences
    \begin{equation*}
        \begin{tikzcd}
            0 \arrow{r} & \prod_{n=1}^\infty A_n \arrow{r} \arrow{d}{d^A} & \prod_{n=1}^\infty B_n \arrow{r} \arrow{d}{d^B} & \prod_{n=1}^\infty C_n \arrow{r} \arrow{d}{d^C} & 0 \\
            0 \arrow{r} & \prod_{n=1}^\infty A_n \arrow{r} & \prod_{n=1}^\infty B_n \arrow{r} & \prod_{n=1}^\infty C_n \arrow{r} & 0
        \end{tikzcd}
    \end{equation*}
    By the \hyperref[lem:snake]{snake lemma},
    there is an exact sequence
    \begin{equation*}
        \begin{tikzcd}[row sep=tiny]
            0 \arrow{r} & \ker d^A \arrow{r} \arrow[phantom]{d}[sloped]{=} & \ker d^B \arrow{r} \arrow[phantom]{d}[sloped]{=} & \ker d^C \arrow{r} \arrow[phantom]{d}[sloped]{=} &
            \coker d^A \arrow{r} & \coker d^B \arrow{r} & \coker d^C \arrow{r} & 0 \\
            & \invlim A_n & \invlim B_n & \invlim C_n
        \end{tikzcd}
    \end{equation*}
    which gives us left-exactness.

    Now, suppose \(\func{\pi_{n+1}}{A_{n+1}}{A_n}\) are all surjective.
    It suffices for us to show that \(\coker d^A = 0\), i.e.\ \(d^A\) is surjective.
    But that amounts to us solving \(x_n - \pi_{n+1}(x_{n+1}) = a_n\) given \(a_n \in A_n\).
    We can do so inductively, choosing \(x_0 = a_0\),
    and then abusing surjectivity to find \(x_{n+1}\) such that \(\pi_{n+1}(x_{n+1}) = x_n - a_n\).
\end{proof}

\begin{lemma}\label{lem:isomorphism-completions}
    Suppose \(R = \mfk{m}_0 \supseteq \mfk{m}_1 \supseteq \cdots\)
    and \(R = \mfk{n}_0 \supseteq \mfk{n}_1 \supseteq \cdots\) are two distinct filtrations,

    then there is a natural isomorphism \(\hat{R}_\mfk{m} \cong \hat{R}_\mfk{n}\).
\end{lemma}
\begin{proof}
    First suppose \(\mfk{n}_k\) are a subsequence of the \(\mfk{m}_k\).
    % Then we have \(\mfk{n}_k \subseteq \mfk{m}_k\),
    % which induces the exact sequence
    % \begin{equation*}

    %         0 \arrow{r} & \mfk{m}_k/\mfk{n}_k \arrow{r} & R/\mfk{n}_k \arrow{r} & R/\mfk{m}_k \arrow{r} & 0
    %     \end{tikzcd}
    % \end{equation*}
    % Then defining \(d\) as in the \hyperref[prop:invlim-exact]{proposition above}
    % which gives us a natural projection of quotients \(\prod_{k=1}^\infty R/\mfk{m}_k \to \prod_{k=1}^\infty R/\mfk{m}_{j_k} \cong \prod_{k=1}^\infty R/\mfk{n}_k\).
    Then this gives us a natural projection of completions \(\invlim R/\mfk{m}_k \to \invlim R/\mfk{n}_k\),
    because every sequence \((g_1,g_2,\hdots) \in \invlim R/\mfk{n}_k\)
    can be mapped from some sequence \((g_1,\hdots,g_1,g_2,\hdots,g_2,\hdots)\) simply by filling in the gaps.
    It now suffices to prove that this is also an injection.
    Suppose \(g = (g_1,g_2,\hdots)\) and \(h = (h_1,h_2,\hdots)\) map to the same sequence.
    Then because they are both convergent sequences that have subsequences that converge to the same limit,
    both of them converge to the same limit.
    Then they must be equivalent, and must be considered the same point.

    Returning to the general case,
    For each \(\mfk{m}_k\), we can find some \(\alpha(k),\beta(k),\gamma(k)\) such that
    \(\mfk{m}_k \supseteq \mfk{n}_{\alpha(k)} \supseteq \mfk{m}_{\beta(k)} \supseteq \mfk{n}_{\gamma(k)}\)

    These further induce the maps
    \begin{equation*}
        \begin{tikzcd}[row sep=tiny]
            \invlim R/\mfk{n}_{\gamma(k)} \arrow{r} & \invlim R/\mfk{m}_{\beta(k)} \arrow{r} & \invlim R/\mfk{n}_{\alpha(k)} \arrow{r} & \invlim R/\mfk{m}_k \\

        \end{tikzcd}
    \end{equation*}
    But because any set of two consecutive arrows is an isomorphism,
    we have that the middle arrow is an isomorphism.
\end{proof}
\begin{lemma}\label{lem:m-adic-invlim-exact}
    Suppose \(R\) is a Noetherian ring, and \(\mfk{m} \subseteq R\) is an ideal.
    If \(A,B,C\) are finitely generated \(R\)-modules, and

        \begin{tikzcd}
            0 \arrow{r} & A \arrow{r} & B \arrow{r} & C \arrow{r} & 0
        \end{tikzcd}
    \end{equation*}

        \end{tikzcd}
    \end{equation*}
    is exact.
    In other words, the \(\mfk{m}\)-adic completion is exact with respect to finitely generated modules over Noetherian rings.
\end{lemma}
\begin{proof}
    % We first show that \(\invlim B/\mfk{m}^n B \to \invlim C/\mfk{m}^n C\) is surjective;
    % that is, given any \(c = (c_1,c_2,\hdots) \in \prod C/\mfk{m}^n C\) satisfying \(c_j \equiv c_i \pmod{\mfk{m}^j C}\) whenever \(j > i\),

    % and that \(b_n \mapsto c_n \pmod{\mfk{m}^n C}\).
    % We do so inductively; the base case is clear because there is nothing to choose for \(b_0 \mapsto c_0\).
    % Now assume we have already chosen \({\{b_i\}}_{i=1}^n\).
    % Let us choose some arbitrary \(b_{n+1}' \in B\) such that \(b_{n+1}' \mapsto c_{n+1} \pmod{\mfk{m}^{n+1} C}\).
    % Obviously \(b_{n+1}' \mapsto c_n \pmod{\mfk{m}^n C}\) too, because of the condition on \(c_{n+1} \equiv c_n \pmod{\mfk{m}^n C}\).
    Because the tensor product is right-exact,
    \begin{equation*}
        \begin{tikzcd}[row sep=tiny]
            R/\mfk{m}^n \otimes A \arrow{r} & R/\mfk{m}^n \otimes B \arrow{r} & R/\mfk{m}^n \otimes C \arrow{r} & 0 \\
            A/\mfk{m}^n A \arrow[phantom]{u}[sloped]{=} & B/\mfk{m}^n B \arrow[phantom]{u}[sloped]{=} & C/\mfk{m}^n C \arrow[phantom]{u}[sloped]{=}
        \end{tikzcd}
    \end{equation*}
    is an exact sequence;
    % as \(b_n - b_{n+1}' \mapsto 0 \pmod{\mfk{m}_n C}\),
    % we can find \(a_{n+1} \in A\) such that \(a_{n+1} \mapsto b_n - b_{n+1}' \pmod{\mfk{m}_n B}\).
    % Hence if we define \(b_{n+1} = a_{n+1} + b_{n+1}'\),
    % \(b_{n+1}\) maps into \(c_{n+1}\) as desired (because \(a_{n+1}\) is in the kernel)
    % and \(b_{n+1} = a_{n+1} + b_{n+1}' \equiv b_n \pmod{\mfk{m}_n B}\) as desired.
    this gives us surjectivity of \(B/\mfk{m}^n B \to C/\mfk{m}^n C\).
    % Then we have to show that the rest of the sequence
    % \begin{equation*}
    %     \begin{tikzcd}
    %         0 \arrow{r} & \invlim A/\mfk{m}^n A \arrow{r} & \invlim B/\mfk{m}^n B \arrow{r} & \invlim C/\mfk{m}^n C
    %     \end{tikzcd}
    % \end{equation*}
    % is exact.
    Then we see that
    \begin{equation*}
        \begin{tikzcd}
            0 \arrow{r} & A/(A \cap \mfk{m}^n B) \arrow{r} & B/\mfk{m}^n B \arrow{r} & C/\mfk{m}^n C \arrow{r} & 0
        \end{tikzcd}
    \end{equation*}
    is exact because \(A/(A \cap \mfk{m}^n B) \to B/\mfk{m}^n B\) is injective by construction;
    moreover, any \(a \in A\) naturally gets mapped into zero after two arrows due to our initial exact sequence.
    Then by Proposition~\ref{prop:invlim-exact},
    because \(A/(A \cap \mfk{m}^{n+1} B) \to A/(A \cap \mfk{m}^n B)\) is surjective
    by the \hyperref[thm:iso-3-ring]{third isomorphism theorem},
    \begin{equation*}
        \begin{tikzcd}
            0 \arrow{r} & \invlim A/(A \cap \mfk{m}^n B) \arrow{r} & \invlim B/\mfk{m}^n B \arrow{r} & \invlim C/\mfk{m}^n C \arrow{r} & 0
        \end{tikzcd}
    \end{equation*}
    is exact.
    % Given any \(b = (b_1,b_2,\hdots) \in \invlim B/\mfk{m}^n B\) mapping to \(0 = (0,0,\hdots) \in \invlim C/\mfk{m}^n C\),
    % then each individual \(b_n \mapsto 0 \in C/\mfk{m}^n C\).
    % Hence \(b_n \in A/(A \cap \mfk{m}^n B)\), and \(b \in \invlim A/A \cap \mfk{m}^n B\).

    Lastly, we need to show that \(\invlim A/(A \cap \mfk{m}^n B) = \invlim A/\mfk{m}^n A\).
    Because \(\mfk{m}^n A \subseteq A \cap \mfk{m}^n B\),
    using the \hyperref[prop:artin-rees]{Artin-Rees lemma},
    there exists some \(n\) such that for all \(k\), \(A \cap \mfk{m}^{n+k} B = \mfk{m}^k(A \cap \mfk{m}^n B) \subseteq \mfk{m}^k A\).

\end{proof}
\begin{theorem}
    Suppose \(R\) is a Noetherian ring, and \(\mfk{m} \subseteq R\) is an ideal.
    Let \(\hat{R} = \hat{R}_\mfk{m}\) denote the completion.
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item If \(M\) is a finitely generated \(R\)-module,
            then the natural map \(\hat{R} \otimes M \to \hat{M} = \invlim M/\mfk{m}^k M\) is an isomorphism.
            In particular, if \(S\) is a finite ring extension of \(R\),
            then \(\hat{R} \otimes S = \hat{S}_{\mfk{m}S}\).
        \item \(\hat{R}\) is a flat \(R\)-module.
    \end{enumerate}
\end{theorem}
\begin{proof}
    Consider a free presentation \(F \to F' \to M \to 0\),
    as in find a free modules \(F,F'\) such that the sequence is exact
    (which works because \(M\) is finitely generated).
    Since \(\invlim\) commutes with the direct sum,
    and the \hyperref[lem:m-adic-invlim-exact]{lemma above} proves that each component preserves the exact sequence,
    \begin{equation*}
        \begin{tikzcd}
            \hat{F} \arrow{r} & \hat{F}' \arrow{r} & \hat{M} \arrow{r} & 0 \arrow[dashrightarrow]{r} & 0 \\
            \hat{R} \otimes F \arrow{r} \arrow{u} & \hat{R} \otimes F' \arrow{r} \arrow{u} &
            \hat{R} \otimes M \arrow{r} \arrow{u} & 0 \arrow[dashrightarrow]{r} \arrow[dashrightarrow]{u} & 0 \arrow[dashrightarrow]{u}
        \end{tikzcd}
    \end{equation*}
    the first row is exact.
    Moreover, by the exactness of the tensor product (Theorem~\ref{thm:tensor-right-exact}),
    the second row is exact.
    We know that the arrows \(\hat{R} \otimes F \to \hat{F}\) and \(\hat{R} \otimes F' \to \hat{F}'\)
    are isomorphisms by definition of completion of rings,
    so by an application of the \hyperref[lem:five]{five lemma},
    \(\hat{R} \otimes M \to \hat{M}\) is an isomorphism.
    This proves the first part of statement (a).
    The second part of statement (a) then immediately follows because \(\hat{R} \otimes S = \hat{S}\),
    completing with respect to the extended ideals.

    Now, to prove that \(\hat{R}\) is flat,
    it suffices to prove that given finitely generated \(N' \to N\) is injective,
    then \(\hat{R} \otimes N' \to \hat{R} \otimes N\) is injective (Lemma~\ref{lem:flat-equivalence}).
    % Since both \(N',N\) are finitely generated,
    % we can consider a chain of finitely generated \(R\)-modules
    % \(N' = N_0 \subseteq N_1 \subseteq \cdots \subseteq N_n = N\)
    % where \(N_{k+1}\) is generated by one element over \(N_k\).
    % It suffices to show that each \(\hat{R} \otimes N_{k+1} \to \hat{R} \otimes N_k\) is injective,
    % so without any loss of generality, suppose that \(N/N'\) is cyclic (as in it is generated by one element).
    % By the \hyperref[thm:structure-divisor]{structure theorem},
    % we know that \(N/N' \cong R/\mfk{a}\) for some ideal \(\mfk{a} \subseteq R\).
    In that case, because we have an exact sequence
    \begin{equation*}
        \begin{tikzcd}
            0 \arrow{r} & N' \arrow{r} & N \arrow{r} & N/N' \arrow{r} & 0
        \end{tikzcd}
    \end{equation*}
    by the \hyperref[lem:m-adic-invlim-exact]{lemma above},
    \begin{equation*}
        \begin{tikzcd}[row sep=tiny]
            0 \arrow{r} & \invlim N'/\mfk{m}^k N' \arrow{r} & \invlim N/\mfk{m}^k N \arrow{r} &
            \invlim (N/N')/\mfk{m}^k(N/N') \arrow{r} & 0 \\
            & \hat{R} \otimes N' \arrow[phantom]{u}[sloped]{\cong} &
            \hat{R} \otimes N \arrow[phantom]{u}[sloped]{\cong} &
            \hat{R} \otimes N/N' \arrow[phantom]{u}[sloped]{\cong}
        \end{tikzcd}
    \end{equation*}
    is exact as desired.
\end{proof}

\subsection*{Hensel's Lemma}

\begin{definition}
    Suppose \(S\) is a ring, and \(R = S[x_1,\hdots,x_n]\) a polynomial ring over \(S\).
    If \(\mfk{m} = (x_1,\hdots,x_n) \subseteq R\) is an ideal,
    then \(\hat{R}_\mfk{m} = S[[x_1,\hdots,x_n]]\) is the ring of formal power series.
\end{definition}
\begin{lemma}
    The power series ring indeed is the conventional ring of power series that you are thinking of.
\end{lemma}
\begin{proof}
    Consider elements \(a_{i_1,\hdots,i_n} \in R\) called coefficients,
    and associate this with an element \(f = \sum a_{i_1,\hdots,i_n} x_1^{i_1} \cdots x_n^{i_n}\).
    Associate this with an element \((f+\mfk{m},f+\mfk{m}^2,\hdots) \in \hat{R}_\mfk{m}\).
    The inverse map is obtained by sending \((f_1,f_2,\hdots) \in \hat{R}_\mfk{m} \subseteq \prod R/\mfk{m}_i\)
    into the power series \(f\) where the degree \(k\) term is the degree \(k\) term from \(f_{k+1}\).
    Clearly this map is independent of the choice of representative,
    because all the representatives of \(f_k+\mfk{m}^k\) must agree at all terms with degree lower than \(k\).
\end{proof}

\begin{theorem}\label{thm:power-series-completions}
    Suppose \(S\) is an \(R\)-algebra that is complete with respect to some ideal \(\mfk{n} \subseteq S\).
    Given \({\{f_i\}}_{i=1}^n \subset \mfk{n}\),
    \begin{enumerate}[label={(\alph*)}, itemsep=0mm]
        \item there exists a unique \(R\)-algebra homomorphism
            \begin{equation*}
                \varfunc{\phi}{R[[x_1,\hdots,x_n]]}{S}{x_i}{f_i}
            \end{equation*}
            that sends any power series \(g(x_1,\hdots,x_n) \mapsto g(f_1,\hdots,g_n)\);
        \item if the induced map \(R \to S/\mfk{n}\) is surjective and \(\{f_i\}\) generate \(\mfk{n}\),
            then \(\phi\) is also surjective; and
        \item if the induced map
            \begin{equation*}
                \func{\gr\phi}{R[x_1,\hdots,x_n]\cong\gr_{(x_1,\hdots,x_n)} R[[x_1,\hdots,x_n]]}{\gr_\mfk{n} S}
            \end{equation*}
            is injective, then \(\phi\) is also injective.
    \end{enumerate}
\end{theorem}
\begin{proof}
    Since there exists a unique map for each \(k\)
    \begin{equation*}
        \varfunc{\phi_k}{R[[x_1,\hdots,x_n]]/{(x_1,\hdots,x_n)}^k = R[x_1,\hdots,x_n]/{(x_1,\hdots,x_n)}^k}{S/\mfk{n}^k}{x_i}{f_i}
    \end{equation*}
    by the universal property of the inverse limit,
    there exists a unique map
    \begin{equation*}
        \varfunc{\phi}{R[[x_1,\hdots,x_n]]}{S=\hat{S}_\mfk{n}}{x_i}{f_i}
    \end{equation*}
    as desired.
    This proves statement (a).

    Now, since \({\{f_i\}}_{i=1}^n\) generate \(\mfk{n}\),
    \({(x_1,\hdots,x_n)}^k/{(x_1,\hdots,x_n)}^{k+1} \to \mfk{n}/\mfk{n}^2\) is a surjection (for all \(k \geq 1\)),
    so \(\gr_{(x_1,\hdots,x_n)} R[[x_1,\hdots,x_n]] \to \gr_\mfk{n} S\) is also a surjection
    (with the case \(k = 0\) given to us as the given surjection).
    Given some nonzero element \(g \in S\),
    consider the largest \(k\) such that \(g \in \mfk{n}^k\)
    (we are allowed to do so because \(S\) complete implies \(\bigcap \mfk{n}^k = 0\)).
    By surjectivity of \(\gr\phi\), there exists some polynomial \(g_1 \in {(x_1,\hdots,x_n)}^k\)
    such that the leading terms \(l(g_1) \mapsto l(g)\).
    Then \(g-\phi(g_1) \in \mfk{n}^{k+1}\).
    We can now repeat this process indefinitely and obtain \({\{g_n\}}_{n=1}^\infty\),
    and we know that \(g = \sum_{n=1}^\infty \phi(g_n)\).
    But because \(\phi\) is a homomorphism on the ring of power series,
    \(g = \phi(\sum_{n=1}^\infty g_n)\), giving us surjectivity as required.
    This proves statement (b).

    To prove part (c), it suffices to show that given any nonzero \(g \in R[[x_1,\hdots,x_n]]\),
    \(\phi(g) \neq 0\) also.
    In that case, \(l(g)\) has some positive degree \(d\),
    so \(\gr\phi(l(g)) \neq 0\) in the degree \(d\) part of \(\gr_\mfk{n} S\).
    But \(g \equiv l(g) \pmod{{(x_1,\hdots,x_n)}^{d+1}}\),
    so \(\phi(g) \equiv \gr\phi(l(g)) \pmod{\mfk{n}^{d+1}}\) is nonzero as well.
\end{proof}
\begin{corollary}\label{cor:derivative-unit-isomorphism}
    Suppose \(f \in xR[[x]]\) is a power series.
    If we have the endomorphism
    \begin{equation*}
        \varfunc{\phi}{R[[x]]}{R[[x]]}{x}{f}
    \end{equation*}
    then \(\phi\) is an isomorphism if and only if \(f'(0)\) is a unit in \(R\).
\end{corollary}
\begin{proof}
    Suppose \(\phi\) is an isomorphism.
    Observe that the elements in \((x)\) consist of those with a zero constant term,
    Because \(f \in xR[[x]]\) has no constant term,
    we can see that \(\phi(R[[x]]-(x)) = R[[x]] - (x)\) and \(\phi((x)) = (x)\).
    In particular, that implies \(\phi(x) = f\) also generates \((x)\);
    hence \(f + (x^2)\) generates \((x)/(x^2)\).
    Since \(f \equiv f'(0)x \pmod{(x^2)}\), \(f'(0)x\) generates \((x)\), so \(f'(0)\) must be a unit.

    Conversely, suppose that \(f'(0) = u\) is a unit in \(R\).
    Then as \(\gr_{(x)} R[[x]] = R[x]\), and
    \begin{equation*}
        \varfunc{\gr\phi}{R[x]}{R[x]}{x}{ux}
    \end{equation*}
    is an isomorphism because \(u\) is a unit (we wipe away all other degree terms).
    By the \hyperref[thm:power-series-completions]{theorem above},
    \(\phi\) is injective.
    But we can also write \(f = ux + hx^2 = (u+hx)x\) for some \(h \in [[x]]\);
    because \(u+hx\) is a unit in \(R[[x]]\), \(f\) generates \((x)\).
    Hence by the \hyperref[thm:power-series-completions]{theorem above} again,
    \(\phi\) is surjective.
\end{proof}

\begin{theorem}[Hensel's Lemma]
    Supose \(R\) is a complete ring with respect to the ideal \(\mfk{m}\),
    and let \(f \in R[x]\) be a polynomial.
    If \(a \in R\) satisfies \(f(a) \equiv 0 \pmod{{f'(a)}^2\mfk{m}}\),
    then there exists a root \(b\) such that \(f(b) = 0\) and \(b \equiv a \pmod{f'(a)\mfk{m}}\).
    If \(f'(a)\) is not a zero divisor in \(R\), then \(b\) is unique.
\end{theorem}
\begin{proof}
    Suppose \(f'(a) = e\).
    Find \(h \in R[[x]]\) such that
    \begin{equation*}
        f(a+ex) = f(a) + f'(a)ex + h(x){(ex)}^2 = f(a) + e^2(x + h(x)x^2)
    \end{equation*}
    (like Taylor's theorem in analysis).
    Because \(g = x + h(x)x^2 \in xR[[x]]\), and its derivative \(g' = 1 + 2h'(x)x\) gives us \(g'(0) = 1\) is a unit,
    by the \hyperref[cor:derivative-unit-isomorphism]{corollary above},
    there exists an isomorphism \(\phi\) sending \(x \mapsto g\).
    In particular, we can apply \(\phi^{-1}\) to get
    \begin{equation*}
        f(a+e\phi^{-1}(x)) = f(a) + e^2 x
    \end{equation*}
    Since \(f(a) \equiv 0 \pmod{e^2\mfk{m}}\), we can write \(f(a) = e^2 c\) for some \(c \in \mfk{m}\).
    By Theorem~\ref{thm:power-series-completions},
    there exists a homomorphism \(\psi\) such that \(x \mapsto -c\).
    Applying that we have
    \begin{equation*}
        f(a+e(\psi\circ\phi^{-1})(x)) = \psi(e^2 c + e^2 x) = 0
    \end{equation*}
    so we have found some \(b = a + e(\psi\circ\phi^{-1})(x)\) such that \(f(b) = 0\).
    We can check that \(b-a = e(\psi\circ\phi^{-1})(x) \in e\mfk{m}\) indeed.
    Obviously, as we mentioned in the \hyperref[cor:derivative-unit-isomorphism]{corollary above},
    \(\phi^{-1}(x) \in (x)\), so \((\psi\circ\phi^{-1})(x) \in \psi((x)) \subseteq (c) \subseteq \mfk{m}\)
    as desired.

    % Now, we prove the second statement by contraposition.
    Now, let \(e\) not be a zero divisor,
    Suppose there exists \(b, b_1\) that both satisfy our requirements,
    as in \(f(b) = f(b_1) = 0\), and \(b = a+er\) and \(b_1 + a+er_1\) with \(r,r_1 \in \mfk{m}\).
    By Theorem~\ref{thm:power-series-completions},
    there exists homomorphisms \(\beta,\beta_1\) such that \(x \mapsto r\) and \(x \mapsto r_1\) respectively.
    Applying \(f\) over them we get
    \begin{equation*}
        f(a) + e^2(r + h(r)r^2) = f(b) = f(b_1) = f(a) + e^2(r_1 + h(r_1)r_1^2)
    \end{equation*}
    then \((\beta\circ\phi)(x) = r + h(r)r^2 = r_1 + h(r_1)r_1^2 = (\beta_1\circ\phi)(x)\).
    By Theorem~\ref{thm:power-series-completions}, since this homomorphism is unique,
    \(\beta\circ\phi = \beta_1\circ\phi\), so \(r = r_1\) and \(b = b_1\) as desired.
\end{proof}

\begin{definition}
    Suppose \(R\) is a commutative ring, and \(A\) is a (potentially non-commutative) \(R\)-algebra.
    An element \(e \in A\) is idempotent if \(e^2 = e\).
    Two idempotents \(e,e' \in A\) are orthogonal if \(ee' = 0\).
\end{definition}
\begin{lemma}\label{lem:sum-idempotents}
    Suppose \({\{e_i\}}_{i=1}^n\) are orthogonal idempotents.
    Then \(\sum e_i\) is also an idempotent.
\end{lemma}
\begin{proof}
    It suffices to consider the case of two idempotents.
    \begin{equation*}
        {(e_1 + e_2)}^2 = e_1^2 + e_1e_2 + e_2^2 = e_1 + e_2
    \end{equation*}
\end{proof}
\begin{definition}
    Suppose \({\{e_i\}}_{i=1}^n\) are orthogonal idempotents.
    They are complete if \(\sum e_i = 1\).
\end{definition}
\begin{lemma}
    Suppose \({\{e_i\}}_{i=1}^n\) are a set of incomplete orthogonal idempotents.
    Then it can be completed by adjoining \(f = 1 - \sum e_i\).
\end{lemma}
\begin{proof}
    By the \hyperref[lem:sum-idempotents]{lemma above}, \(f\) is an idempotent.
    Furthermore, it is pairwise orthogonal to the rest, because
    \begin{equation*}
        fe_i = \pqty{1 - \sum_j e_j}e_i = e_i - \sum_j e_i e_j = e_i - e_i^2 = e_i - e_i = 0
    \end{equation*}
\end{proof}
